"""Persistent user settings (display, audio, accessibility).

Stored as settings.json next to save_data.json. Loaded once on startup,
saved whenever the user changes an option in the settings menu.
"""
import json
import os

SETTINGS_FILE = os.path.join(os.path.abspath("."), "settings.json")

DEFAULTS = {
    "version": 1,
    # Display
    "display_mode": 0,        # 0 = Windowed, 1 = Fullscreen
    "resolution_idx": 0,      # index into menu.SETTINGS_RESOLUTIONS
    # Audio (0-100 ints, mirroring the existing master slider)
    "master_volume": 80,
    "music_volume": 80,
    "sfx_volume": 100,
    # Accessibility
    "reduced_motion": False,  # disables screen shake + bright flash overlays
}


def load():
    """Return the current settings dict, falling back to defaults."""
    data = dict(DEFAULTS)
    if not os.path.exists(SETTINGS_FILE):
        return data
    try:
        with open(SETTINGS_FILE, "r", encoding="utf-8") as f:
            loaded = json.load(f)
        if isinstance(loaded, dict):
            for key, default_val in DEFAULTS.items():
                if key in loaded and isinstance(loaded[key], type(default_val)):
                    data[key] = loaded[key]
    except Exception as e:
        print(f"Settings load failed, using defaults: {e}")
    return data


def save(data):
    """Persist the given settings dict. Silently ignores write errors."""
    try:
        payload = dict(DEFAULTS)
        payload.update({k: v for k, v in data.items() if k in DEFAULTS})
        with open(SETTINGS_FILE, "w", encoding="utf-8") as f:
            json.dump(payload, f, indent=2)
    except Exception as e:
        print(f"Settings save failed: {e}")
