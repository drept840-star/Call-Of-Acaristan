def _tc(col, row):
    return col * TILE_SIZE + TILE_SIZE // 2, row * TILE_SIZE + TILE_SIZE // 2


def _tile_rect(col, row, w=1, h=1):
    return pygame.Rect(col * TILE_SIZE, row * TILE_SIZE, w * TILE_SIZE, h * TILE_SIZE)


def _zone_world_rect(zone_name):
    tr = ZONE_DEFS[zone_name]["tile_rect"]
    return _tile_rect(tr.x, tr.y, tr.w, tr.h)


ZONE_WORLD_RECTS = {name: _zone_world_rect(name) for name in ZONE_DEFS}


def get_zone_for_world_pos(pos):
    """Return the authored zone name for a world-space point, or None."""
    for name, rect in ZONE_WORLD_RECTS.items():
        if rect.collidepoint(pos):
            return name
    return None

def _beat(phase, event, default=0):
    return SNIPER_BEAT_CLOCK.get(phase, {}).get(event, default)

def _beat_now(phase, pt, event):
    f = SNIPER_BEAT_CLOCK.get(phase, {}).get(event)
    return f is not None and pt == f


def _set_tile(g, col, row, tile):
    if 0 <= col < MAP_W and 0 <= row < MAP_H:
        g[row][col] = tile


def _paint_tiles(g, rect, tile):
    for row in range(rect.y, rect.y + rect.h):
        for col in range(rect.x, rect.x + rect.w):
            _set_tile(g, col, row, tile)


def _outline_building(g, col, row, w, h, interior="O", door_tiles=()):
    for yy in range(row, row + h):
        for xx in range(col, col + w):
            border = xx in (col, col + w - 1) or yy in (row, row + h - 1)
            _set_tile(g, xx, yy, "C" if border else interior)
    for dx, dy in door_tiles:
        _set_tile(g, col + dx, row + dy, interior)


def _generate_map():
    g = [["g" for _ in range(MAP_W)] for _ in range(MAP_H)]

    # Paint authored regions first.  Collidable structure is added after.
    for zone in ZONE_DEFS.values():
        _paint_tiles(g, zone["tile_rect"], zone["floor"])

    # Continuous highway spine and shoulders.  These broad open sightlines are
    # intentional: the sniper ambush later weaponizes this road geometry.
    _paint_tiles(g, pygame.Rect(17, 23, 37, 55), ".")
    _paint_tiles(g, pygame.Rect(12, 54, 42, 13), ".")
    _paint_tiles(g, pygame.Rect(7, 70, 25, 8), ".")
    _paint_tiles(g, pygame.Rect(4, 47, 23, 15), "d")
    _paint_tiles(g, pygame.Rect(45, 38, 23, 15), "s")
    _paint_tiles(g, pygame.Rect(39, 12, 28, 16), "O")

    # World border.
    for x in range(MAP_W):
        g[0][x] = g[MAP_H - 1][x] = "C"
    for y in range(MAP_H):
        g[y][0] = g[y][MAP_W - 1] = "C"

    # Control building and offices.
    _outline_building(g, 51, 9, 17, 17, interior="O", door_tiles=((6, 16), (7, 16), (0, 9)))
    # Smaller security hut / checkpoint utility room.
    _outline_building(g, 41, 18, 9, 8, interior="O", door_tiles=((4, 7),))

    # Toll booths and gantry supports.
    for bx, by in [(23, 32), (30, 32), (44, 32), (51, 32), (24, 37), (45, 37)]:
        for yy in range(by, by + 2):
            for xx in range(bx, bx + 2):
                _set_tile(g, xx, yy, "C")
    for sx, sy in [(56, 25), (59, 25), (56, 27), (59, 27)]:
        _set_tile(g, sx, sy, "C")

    # Barriers / cover.  These break up travel without deleting the long sniper
    # sightline through the center lane.
    barriers = [
        (12, 72), (13, 72), (21, 69), (22, 69),
        (18, 61), (19, 61), (27, 60), (28, 60), (37, 58),
        (10, 51), (11, 51), (20, 47), (21, 47),
        (29, 40), (30, 40), (34, 39), (35, 39), (40, 40),
        (47, 45), (48, 45), (62, 47), (63, 47),
        (44, 27), (45, 27), (49, 28), (50, 28),
        (57, 21), (58, 21), (62, 18), (63, 18),
    ]
    for col, row in barriers:
        _set_tile(g, col, row, "B")

    # Final-build density pass: guardrails, broken dividers, service-road clutter,
    # and ditch obstructions.  These are intentionally staggered so the map has
    # shape without blocking the central sniper sightline.
    for row in range(25, 76, 4):
        if row not in (32, 33, 34, 35, 36, 37, 38, 39):
            _set_tile(g, 16, row, "B")
            _set_tile(g, 54, row + (row // 4) % 2, "B")
    for row in range(55, 66, 3):
        for col in (14, 41, 46, 51):
            _set_tile(g, col, row, "B")
    for col in range(6, 26, 4):
        _set_tile(g, col, 46, "B")
        _set_tile(g, col + 1, 59, "B")
    for col in range(46, 66, 4):
        _set_tile(g, col, 40, "B")
        _set_tile(g, col + 1, 51, "B")
    for col in range(21, 53, 7):
        _set_tile(g, col, 30, "B")
        _set_tile(g, col + 2, 42, "B")
    # Interior partitions in the control building so it no longer reads as one
    # empty rectangle.  Door gaps keep all evidence reachable.
    for col in range(53, 67):
        if col not in (57, 62):
            _set_tile(g, col, 17, "C")
    for row in range(11, 25):
        if row not in (15, 21):
            _set_tile(g, 60, row, "C")

    # Tanker footprint.  The actual collision uses mission2_tanker_rect.
    for row in range(36, 39):
        for col in range(43, 48):
            _set_tile(g, col, row, "T")

    _set_tile(g, 8, 74, "P")
    _set_tile(g, 64, 12, "X")
    return ["".join(row) for row in g]

def _dist(a, b):
    return math.hypot(a[0] - b[0], a[1] - b[1])


def _player_pos():
    if not STATE.player:
        return (0, 0)
    return STATE.player.rect.centerx, STATE.player.rect.centery


def _near(pos, radius=80):
    return STATE.player and _dist(_player_pos(), pos) <= radius


def _camera_offset_for_target(target):
    tx, ty = target
    max_x = max(0, MAP_W * TILE_SIZE - WIDTH)
    max_y = max(0, MAP_H * TILE_SIZE - HEIGHT)
    cam_x = tx - WIDTH // 2
    cam_y = ty - HEIGHT // 2
    cam_x = max(0, min(max_x, cam_x))
    cam_y = max(0, min(max_y, cam_y))
    return cam_x, cam_y


def _snap_camera_to(target=None):
    if target is None:
        target = _player_pos()
    cam_x, cam_y = _camera_offset_for_target(target)
    STATE.camera_x = cam_x
    STATE.camera_y = cam_y
    STATE.cam_kick_x = 0.0
    STATE.cam_kick_y = 0.0

def _radio(text, color=C_VASILE, duration=300, mark=None, allow_when_cut=False):
    """Small radio helper. Used sparingly; cards do most story work."""
    if mark and mark in _radio_marks:
        return
    if mark:
        _radio_marks.add(mark)

    # Sniper subtitles need reading time (~4 seconds minimum).
    if getattr(STATE, "mission2_stage", "") == STAGE_SNIPER:
        duration = max(duration, 240)

    if getattr(STATE, "mission2_comms_cut", False) and not allow_when_cut:
        text = "RADIO: <static>"
        color = (125, 125, 135)
        duration = min(duration, 140)
    # Muffled radio after the near-miss: drop letters, insert ellipses, fade
    # color so it visually reads as "underwater" without needing a real
    # low-pass filter.
    if getattr(STATE, "mission2_radio_muffle_timer", 0) > 0 and not text.startswith("RADIO:"):
        text = _muffle_radio_text(text)
        color = tuple(max(70, int(c * 0.62)) for c in color)
        if getattr(STATE, "mission2_stage", "") == STAGE_SNIPER:
            duration = max(duration, 240)
        else:
            duration = max(160, int(duration * 0.75))
    trigger_radio_chatter(text, color=color, duration=duration)


def _muffle_radio_text(text):
    """Drop syllables and insert ellipses so a radio line reads as ear-ringing.

    Ramps the muffle severity based on mission2_radio_muffle_timer so the
    effect fades out gradually as the player's hearing returns.
    """
    timer = getattr(STATE, "mission2_radio_muffle_timer", 0)
    if timer <= 0:
        return text
    if ":" in text:
        speaker, _, spoken = text.partition(":")
        spoken = spoken.strip().strip('"')
        words = spoken.split()
        if len(words) <= 2:
            return text
        intensity = min(0.6, timer / 180.0)
        keep = []
        for i, w in enumerate(words):
            if i == 0 or random.random() > intensity:
                keep.append(w)
        return f'{speaker}: "{" . . . ".join(keep)}"'
    return text

        STATE.particles.append(p)

def _actor_collision_rect(x, y):
    return pygame.Rect(int(x - 12), int(y - 12), 24, 24)


def _actor_rect_collides(rect):
    for wall in _solid_walls():
        if rect.colliderect(wall):
            return True
    return False


def _try_actor_step(actor, dx, dy):
    if abs(dx) < 0.001 and abs(dy) < 0.001:
        return False
    moved = False
    x, y = actor["pos"]
    # X axis first.
    if abs(dx) > 0.001:
        nr = _actor_collision_rect(x + dx, y)
        if not _actor_rect_collides(nr):
            actor["pos"][0] = x + dx
            x = actor["pos"][0]
            moved = True
    # Y axis second.
    if abs(dy) > 0.001:
        nr = _actor_collision_rect(x, y + dy)
        if not _actor_rect_collides(nr):
            actor["pos"][1] = y + dy
            moved = True
    return moved


def _move_actor_towards(actor, target, rate=0.055, max_step=2.7):
    """Move scripted squad actors while respecting walls/vehicles/solid props.

    This is deliberately not full combat pathfinding.  It is a lightweight
    cinematic/ambient mover with axis collision and a small sidestep fallback so
    actors stop sliding through barriers but can still work around minor clutter.
    """
    if not actor.get("alive", True):
        return
    x, y = actor["pos"]
    dx = target[0] - x
    dy = target[1] - y
    dist = math.hypot(dx, dy)
    if dist < 1:
        return
    step = min(max_step, dist * rate)
    nx = dx / dist * step
    ny = dy / dist * step
    moved = _try_actor_step(actor, nx, ny)
    if not moved and dist > 8:
        # Try sliding around small props/barriers.  Alternating side preference
        # stops all three squadmates choosing the exact same detour.
        side = -1 if (hash(actor.get("name", "")) % 2) else 1
        px = -ny * side
        py = nx * side
        if not _try_actor_step(actor, px, py):
            _try_actor_step(actor, -px, -py)
    if not moved and dist > 8:
        # Broader escape — try multiple angles when stuck on adjacent barriers.
        for slide_angle in (0.5, -0.5, 1.0, -1.0, 1.5, -1.5):
            cs = math.cos(slide_angle)
            sn = math.sin(slide_angle)
            sx = nx * cs - ny * sn
            sy = nx * sn + ny * cs
            if _try_actor_step(actor, sx, sy):
                break


def _nearest_unfound_clue(max_dist=360):
    if not STATE.player:
        return None
    px, py = _player_pos()
    best = None
    best_d = max_dist
    for clue in getattr(STATE, "mission2_clues", []):
        if clue.get("done"):
            continue
        d = math.hypot(clue["pos"][0] - px, clue["pos"][1] - py)
        if d < best_d:
            best = clue
            best_d = d
    return best

def _update_squad_actor_positions():
    squad = getattr(STATE, "mission2_squad", None)
    if not squad or len(squad) < 3:
        return

    gabi, seby, darius = squad[0], squad[1], squad[2]
    seby_activity = (seby.get("activity") or "").lower()
    darius_activity = (darius.get("activity") or "").lower()
    pt = int(getattr(STATE, "mission2_sniper_phase_timer", 0))

    if "sprint" in seby_activity or "drag" in seby_activity or "kneel" in seby_activity:
        if "_sprint_origin" not in seby:
            seby["_sprint_origin"] = tuple(seby["pos"])
        if "_corpse_origin" not in gabi:
            gabi["_corpse_origin"] = tuple(gabi["pos"])

        s_ox, s_oy = seby["_sprint_origin"]
        g_ox, g_oy = gabi["_corpse_origin"]

        kneel_x, kneel_y = g_ox - 22, g_oy - 4
        cover_x, cover_y = g_ox - 70, g_oy + 18

        if "sprint" in seby_activity:
            start = _beat("movement_two", "seby_breaks", 80)
            end = _beat("movement_two", "seby_arrives", 180)
            t = max(0.0, min(1.0, (pt - start) / max(1, end - start)))
            t = 1.0 - (1.0 - t) ** 2
            seby["pos"] = [s_ox + (kneel_x - s_ox) * t,
                           s_oy + (kneel_y - s_oy) * t]

        elif "drag" in seby_activity or "kneel" in seby_activity:
            # Use cover-lerp during the actual dragging phase.
            # _drag_lerp was only used for the earlier sprint-to-Gabi phase.
            t = max(0.0, min(1.0, float(seby.get("_cover_lerp", seby.get("_drag_lerp", 0.0)))))
            start = _beat("movement_two", "seby_arrives", 180)
            end = _beat("movement_two", "seby_drag_to_cover_end", 340)
            t = max(0.0, min(1.0, (pt - start) / max(1, end - start)))
            t = t * t * (3.0 - 2.0 * t)
            seby["pos"] = [kneel_x + (cover_x + 18 - kneel_x) * t,
                           kneel_y + (cover_y - 4 - kneel_y) * t]
            gabi["pos"] = [g_ox + (cover_x - g_ox) * t,
                           g_oy + (cover_y - g_oy) * t]

            if pt % 5 == 0:
                STATE.blood_decals.append([
                    gabi["pos"][0] + random.randint(-4, 4),
                    gabi["pos"][1] + random.randint(-2, 2),
                    90,
                    random.randint(8, 14),
                ])
    else:
        seby.pop("_sprint_origin", None)
        gabi.pop("_corpse_origin", None)

    if "reach" in darius_activity:
        if "_reach_origin" not in darius:
            darius["_reach_origin"] = tuple(darius["pos"])
        ox, oy = darius["_reach_origin"]
        darius["pos"] = [ox + 18, oy - 8]
    elif "_reach_origin" in darius and "down" not in darius_activity:
        ox, oy = darius["_reach_origin"]
        darius["pos"] = [ox, oy]

def _midpoint(a, b, t=0.5):
    return (
        a[0] + (b[0] - a[0]) * t,
        a[1] + (b[1] - a[1]) * t,
    )


def _avg_points(points):
    if not points:
        return KILL_ZONE
    return (
        sum(p[0] for p in points) / len(points),
        sum(p[1] for p in points) / len(points),
    )

    return _avg_points(pts) if pts else KILL_ZONE


def _sniper_camera_target(t, phase):
    """Anchor-based camera target — the camera holds fixed compositions."""
    if phase == "movement_one":
        if t < _beat("movement_one", "gabi_crack") - 25:
            return CAM_SQUAD_WIDE
        if t < _beat("movement_one", "gantry_hold"):
            return CAM_GABI_BODY          # hold completely still on the body
        if t < _beat("movement_one", "seby_react"):
            return CAM_GANTRY_REVEAL      # hold for ~200 frames on the gantry
        return CAM_SEBY_DRAG              # settle on Seby for the transition

    elif phase == "movement_two":
        if t < _beat("movement_two", "seby_crack") + 60:
            return CAM_SEBY_DRAG          # hold during the entire drag
        if t < _beat("movement_two", "darius_crack") + 40:
            return CAM_DARIUS_RADIO       # hold on Darius
        return CAM_DARIUS_RADIO

    elif phase == "movement_three":
        if STATE.player:
            return (STATE.player.rect.centerx, STATE.player.rect.centery - 20)
        return KILL_ZONE

    return KILL_ZONE

def _player_exposed_to_sniper():
    if (
        not STATE.player
        or not _sniper_alive()
        or getattr(STATE, "mission2_sniper_suppressed", False)
        or getattr(STATE, "mission2_stage", "") != STAGE_COMBAT
    ):
        return False

    px, py = _player_pos()
    sx, sy = SNIPER_NEST_POS

    dist = math.hypot(px - sx, py - sy)

    # The sniper should not shoot across the entire giant map,
    # but if you are anywhere near the checkpoint/gantry road and visible,
    # you are in danger.
    if dist > SNIPER_MAX_LOS_RANGE:
        return False

    # If nothing blocks line-of-sight, the sniper sees you.
    # No tiny killbox rectangle. No magic safety just because you walked past it.
    return not _wall_blocks_line(sx, sy, px, py, _solid_walls())

def _update_sniper_pressure():
    if (
        STATE.mission2_stage != STAGE_COMBAT
        or not _sniper_alive()
        or getattr(STATE, "mission2_sniper_suppressed", False)
        or not STATE.player
        or STATE.player_dead
    ):
        return

    if getattr(STATE, "mission2_grace_timer", 0) > 0:
        STATE.mission2_grace_timer -= 1
        return

    px, py = _player_pos()
    sx, sy = SNIPER_NEST_POS
    dist_to_sniper = math.hypot(px - sx, py - sy)
    exposed = _player_exposed_to_sniper()

    # Already aiming/firing
    if STATE.mission2_sniper_pending_timer > 0:
        STATE.mission2_sniper_pending_timer -= 1

        # Keep laser snapping to the player's current position
        if exposed and STATE.mission2_sniper_pending_timer in (28, 18, 8, 4):
            tx, ty = _player_pos()
            STATE.mission2_sniper_target = (
                tx + random.randint(-10, 10),
                ty + random.randint(-10, 10),
            )
            _add_sniper_laser(STATE.mission2_sniper_target, life=24)

        if STATE.mission2_sniper_pending_timer <= 0:
            target = STATE.mission2_sniper_target or _player_pos()

            if exposed:
                too_close = dist_to_sniper <= SNIPER_POINT_BLANK_RANGE

                if too_close:
                    damage = SNIPER_EXECUTION_DAMAGE
                    msg = "SNIPER EXECUTION"
                else:
                    damage = SNIPER_DAMAGE
                    msg = "SNIPER HIT"

                STATE.player.hp -= damage
                _burst(STATE.player.rect.centerx, STATE.player.rect.centery, C_BLOOD, 18, speed_mul=1.4)

                STATE.floating_texts.append([
                    STATE.player.rect.centerx,
                    STATE.player.rect.centery - 48,
                    msg,
                    (255, 55, 35),
                    80,
                ])

                _sniper_fire_at(_player_pos(), body=True, trace_life=38, shake=34 if too_close else 26)

            else:
                # Player reached cover before the shot landed
                cover = _nearest_killbox_cover()
                _sniper_fire_at(cover, concrete=True, trace_life=32, shake=18)

            STATE.mission2_sniper_pressure_timer = 120
            STATE.mission2_sniper_target = None
            return

    # Not currently aiming
    if exposed:
        STATE.mission2_sniper_pressure_timer -= 1

        # If the player walks close to the sniper, near-instant kill.
        if dist_to_sniper <= SNIPER_POINT_BLANK_RANGE:
            STATE.mission2_sniper_pending_timer = SNIPER_CLOSE_AIM_FRAMES
            STATE.mission2_sniper_target = _player_pos()
            _add_sniper_laser(STATE.mission2_sniper_target, life=18)
            audio.play("rifle_handle", volume=0.35)
            STATE.floating_texts.append([
                px,
                py - 58,
                "TOO CLOSE — SNIPER HAS YOU",
                (255, 60, 40),
                60,
            ])
            return

        # Normal long-range sniper acquisition
        if STATE.mission2_sniper_pressure_timer <= 0:
            STATE.mission2_sniper_pending_timer = SNIPER_AIM_FRAMES
            STATE.mission2_sniper_target = (
                px + random.randint(-12, 12),
                py + random.randint(-12, 12),
            )
            _add_sniper_laser(STATE.mission2_sniper_target, life=44)
            audio.play("rifle_handle", volume=0.30)
            STATE.floating_texts.append([
                px,
                py - 58,
                "SNIPER LINE!",
                (255, 110, 60),
                58,
            ])

    else:
        # If covered, keep pressure timer from instantly firing the moment player peeks.
        STATE.mission2_sniper_pressure_timer = max(45, STATE.mission2_sniper_pressure_timer)


def _update_post_sniper_gameplay():
    if STATE.mission2_stage != STAGE_COMBAT:
        return
    phase = getattr(STATE, "mission2_post_sniper_phase", "")
    if not phase:
        return
    STATE.mission2_post_sniper_timer += 1
    _update_sniper_pressure()

    if phase == "get_cover":
        STATE.mission_phase = "m2_get_cover"
        if STATE.mission2_post_sniper_timer == 1:
            _spawn_initial_ipa_team()
        if STATE.mission2_post_sniper_timer == 15:
            _set_objective("GET TO COVER — break the sniper line")
            _radio('VASILE: "Concrete blocks! Left side! Move now!"', color=(255, 170, 80), mark="post_cover")
        if _player_in_meaningful_cover() and STATE.mission2_post_sniper_timer > 45:
            STATE.mission2_post_sniper_phase = "survive"
            STATE.mission2_post_sniper_timer = 0
            STATE.mission2_sniper_pressure_timer = 70
            _radio('VASILE: "Hold there. Watch the glint. Move only between shots."', color=C_VASILE, mark="survive_killbox")
            # Player committed to cover -> IPA pours in from the flanks to
            # punish the static position.  First wave is immediate and a bit
            # bigger; follow-up waves are scheduled below in the survive phase.
            if not getattr(STATE, "mission2_reinforcement_triggered", False):
                STATE.mission2_reinforcement_triggered = True
                _spawn_reinforcement_wave(count=4, prefer_far=True)
                STATE.mission2_reinforcement_waves = 1
                STATE.mission2_reinforcement_timer = 360
                _radio('VASILE: "They saw you stop! Contacts moving in from both flanks!"',
                       color=(255, 130, 75), mark="reinforce_warn")
        elif STATE.mission2_post_sniper_timer > 360:
            # Fail-soft: if the player is confused but alive, advance to survival
            # instead of trapping the mission in a tutorial objective.
            STATE.mission2_post_sniper_phase = "survive"
            STATE.mission2_post_sniper_timer = 0
    elif phase == "survive":
        STATE.mission_phase = "m2_survive_killbox"
        _set_objective("SURVIVE THE KILL-BOX — stay behind cover")
        # Keep IPA pressure up while the player holds cover.  Waves taper
        # off in size, and stop entirely once a cap is reached so the fight
        # can actually be cleared.
        if getattr(STATE, "mission2_reinforcement_triggered", False):
            STATE.mission2_reinforcement_timer = max(0, STATE.mission2_reinforcement_timer - 1)
            if (STATE.mission2_reinforcement_timer <= 0
                    and STATE.mission2_reinforcement_waves < 3
                    and len(STATE.enemies) < 7):
                STATE.mission2_reinforcement_waves += 1
                wave_size = 3 if STATE.mission2_reinforcement_waves == 2 else 2
                _spawn_reinforcement_wave(count=wave_size, prefer_far=False)
                STATE.mission2_reinforcement_timer = 420
        if not getattr(STATE, "mission2_late_reveal_played", False) and STATE.mission2_post_sniper_timer > 480:
            STATE.mission2_late_reveal_played = True
            _radio('VASILE: "Biribel — the tanker manifest — it lists a second cargo..."', color=C_VASILE, mark="late_reveal")
        if STATE.mission2_post_sniper_timer > 520 or not _sniper_alive():
            STATE.mission2_post_sniper_phase = "flank"
            STATE.mission2_post_sniper_timer = 0
            _radio('VASILE: "Center lane is death. Flank through the ditch or service road."', color=(255, 170, 80), mark="flank_sniper")
    elif phase == "flank":
        STATE.mission_phase = "m2_flank_sniper"
        _set_objective("FLANK THE SNIPER LINE — reach side cover")
        px, py = _player_pos()

def _smoothstep(t):
    return t * t * (3 - 2 * t) if 0 < t < 1 else max(0, min(1, t))

def _ease_out_cubic_m2(t):
    return 1 - pow(1 - max(0, min(1, t)), 3)

def _pulse01(t, period=60):
    """Sawtooth 0→1 over `period` frames, resets at every multiple."""
    return max(0, min(1, (t % period) / period))

def _draw_sniper_callout(surface, fcx, fcy):

    sx, sy = _world(callout["pos"], fcx, fcy)
    life = callout.get("life", 0)
    max_life = max(1, callout.get("max_life", 1))
    t = max(0.0, min(1.0, life / max_life))
    color = callout.get("color", (255, 70, 45))
    alpha = int(255 * t)

        pygame.draw.line(m, (*color, alpha),

def _update_sniper_sequence():
    pt = int(getattr(STATE, "mission2_sniper_phase_timer", 0))
    phase = _sniper_phase_name()
    new_target = _sniper_camera_target(pt, phase)
    existing = STATE.mission2_camera_focus
    if existing is None:
        STATE.mission2_camera_focus = new_target
    else:
        cx, cy = existing
        nx, ny = new_target
        if math.hypot(nx - cx, ny - cy) > 4:
            STATE.mission2_camera_focus = new_target
    _sniper_phase_activity()
    if phase == "regroup":           _m0_regroup(pt)
    elif phase == "movement_one":    _m1_regroup_wrong_scan(pt)
    elif phase == "movement_two":    _m2_seby_darius_die(pt)
    elif phase == "movement_three":  _m3_near_miss_handoff(pt)
    _update_squad_actor_positions()
    STATE.mission2_sniper_phase_timer = pt + 1
    if STATE.mission2_sniper_phase_timer >= _sniper_phase().get("duration", 120):
        _advance_sniper_phase()

def _spawn_reinforcement_wave(count=3, prefer_far=True):
    """Pour in IPA reinforcements from kill-box perimeter spawn anchors.

    Triggered when the player commits to cover during the kill-box.  The wave
    picks spawn anchors farther from the player so it reads as enemies closing
    in from the flanks rather than materializing on top of them.  Each spawn
    gets a small dust/spark burst so it doesn't pop in silently.
    """
    if not STATE.player:
        return
    px, py = _player_pos()
    anchors = list(KILLBOX_REINFORCEMENT_SPAWNS)
    # Sort by distance; pick from the far half if prefer_far, else random.
    anchors.sort(key=lambda a: math.hypot(_tc(*a)[0] - px, _tc(*a)[1] - py),
                 reverse=prefer_far)
    pool = anchors[: max(count + 2, 4)]
    random.shuffle(pool)
    spawned = []
    for col, row in pool[:count]:
        # Small tile jitter so successive waves don't stack identically.
        jc = col + random.randint(-1, 1)
        jr = row + random.randint(-1, 1)
        e = _spawn_enemy(jc, jr, hp=100, speed=2.05)
        e.fire_cooldown = 70 + random.randint(0, 50)
        spawned.append(e)
        # Visual cue: dust kick + spark so the spawn reads as "running in."
        ex, ey = e.rect.center
        _burst(ex, ey, (120, 110, 95), 14, speed_mul=1.6)
        _burst(ex, ey, (255, 200, 110), 4, speed_mul=2.1)
    if spawned:
        audio.play("radio_static", volume=0.18)
        audio.schedule(8, "footstep", volume=0.35)
    return spawned

def handle_interact():
    """Mission-specific [E] interactions. Called through game_core."""
    if not STATE.player or STATE.player_dead:
        return
    if getattr(STATE, "mission2_card", None):
        if STATE.mission2_card.get("age", 0) >= 18:
            _close_card()
        return

    stage = STATE.mission2_stage
    if stage == STAGE_SNIPER:
        gabi_pos = _sniper_actor_pos(0)
        if _dist(_player_pos(), gabi_pos) < 90:
            # Point 10: Player can try to save Gabi during the 40-frame window
            # Input is read but does nothing; acknowledged later in Vasile line
            window_start = _beat("movement_one", "tried_to_save_window_start", 340)
            window_end = _beat("movement_one", "tried_to_save_window_end", 380)
            pt = int(getattr(STATE, "mission2_sniper_phase_timer", 0))
            if window_start <= pt < window_end:
                if not getattr(STATE, "mission2_tried_to_save", False):
                    STATE.mission2_tried_to_save = True
                    # Store that player tried - will be acknowledged later
                    STATE.mission2_tried_to_save_acknowledged = False
            return
    if stage == STAGE_INVESTIGATE:
        for clue in STATE.mission2_clues:
            if not clue["done"] and _near(clue["pos"], 86):
                clue["done"] = True
                STATE.mission2_clues_found.add(clue["id"])
                _open_card(clue["title"], clue["lines"], color=(245, 190, 65), kind=clue.get("kind", "document"))
                if _clue_count() >= REQUIRED_INVESTIGATION_CLUES:
                    _set_objective("Regroup at the blue toll-barrier marker")
                return
    elif stage == STAGE_TRUCK:
        truck = STATE.mission2_truck_manifest
        if not truck["done"] and _near(truck["pos"], 100):
            truck["done"] = True
            _open_card(truck["title"], truck["lines"], color=(255, 190, 90), after="truck_done", kind=truck.get("kind", "manifest"))
            return
    elif stage == STAGE_EVIDENCE:
        if STATE.enemies:
            if any(_near(ev["pos"], 82) for ev in STATE.mission2_evidence if not ev["done"]):
                STATE.floating_texts.append([STATE.player.rect.centerx, STATE.player.rect.centery - 42, "CLEAR IPA CONTACTS FIRST", (255, 120, 75), 90])
            return
        for ev in STATE.mission2_evidence:
            if not ev["done"] and _near(ev["pos"], 82):
                ev["done"] = True
                after = "evidence_ready" if _evidence_count() >= 3 else None

    STATE.mission2_timer += 1

        return

                _start_sniper_sequence()

    elif stage == STAGE_SNIPER:

    elif stage == STAGE_COMBAT:
        _update_post_sniper_gameplay()

            STATE.mission2_post_sniper_phase = ""
            STATE.mission2_stage = STAGE_TRUCK
            STATE.mission_phase = "m2_truck"

            _radio('VASILE: "No MAG soldiers. Find out why that IPA truck is here."', mark="truck")
    elif stage == STAGE_TRUCK:
        _set_objective("Inspect the IPA truck manifest with [E]")
    elif stage == STAGE_TANKER:
        if _near(TANKER_POS, 170):
            _start_tanker_run()
        else:
            _set_objective("Approach the leaking fuel tanker")
    elif stage == STAGE_RUN:
        if _near(SAFE_ZONE, 105):
            STATE.mission2_countdown = min(STATE.mission2_countdown, 90)
            _set_objective("Stay in cover — brace for the tanker blast")
        else:
            _set_objective("RUN to the blue shelter marker before the tanker detonates")
        STATE.mission2_countdown -= 1
        if STATE.mission2_countdown <= 0:
            _explode_tanker()
    elif stage == STAGE_BLACKOUT:
        if STATE.mission2_timer >= 210:
            _open_card(
                "COMMS LOST",

                    "Vasile cuts out mid-sentence.",
                    "The checkpoint loudspeakers are dead.",
                    "Only the control building still has power.",
                ],
                color=(165, 170, 180),
                after="blast_done",
                kind="static",
            )
    elif stage == STAGE_EVIDENCE:
        found = _evidence_count()

            _set_objective(f"Clear IPA cleanup team ({len(STATE.enemies)} left)")
        elif found < 3:
            _set_objective(f"Inspect green evidence terminals with [E] ({found}/3)")
        else:
            _set_objective("Hold [E] at the upload terminal")
            if _near(UPLOAD_TERMINAL_POS, 95) and keys[pygame.K_e]:
                STATE.mission2_stage = STAGE_UPLOAD
                STATE.mission_phase = "m2_upload"
                STATE.mission2_timer = 0
    elif stage == STAGE_UPLOAD:
        if not _near(UPLOAD_TERMINAL_POS, 110):
            _set_objective("Return to the upload terminal")
            STATE.mission2_upload_progress = max(0.0, STATE.mission2_upload_progress - 1 / 360)

def _update_player(keys, solid_walls):
    if not STATE.player or STATE.player_dead:
        return
    if STATE.mission2_stage == STAGE_INVESTIGATE:
        STATE.player.speed = 3.45
    elif STATE.mission2_stage == STAGE_RUN:
        STATE.player.speed = 4.45
    else:
        STATE.player.speed = 3.6

    mx, my = get_scaled_mouse()
    world_mx = mx + STATE.camera_x
    world_my = my + STATE.camera_y
    STATE.player.angle = math.atan2(world_my - STATE.player.rect.centery, world_mx - STATE.player.rect.centerx)

    dx = dy = 0
    if getattr(STATE, "mission2_input_freeze", 0) > 0:
        keys_for_move = {pygame.K_w: False, pygame.K_s: False, pygame.K_a: False, pygame.K_d: False}
    else:
        keys_for_move = _last_keys
    if not _movement_locked():
        if keys_for_move[pygame.K_w]:
            dy -= STATE.player.speed
        if keys_for_move[pygame.K_s]:
            dy += STATE.player.speed
        if keys_for_move[pygame.K_a]:
            dx -= STATE.player.speed
        if keys_for_move[pygame.K_d]:
            dx += STATE.player.speed
    if dx and dy:
        dx *= 0.707
        dy *= 0.707
    STATE.player.move(dx, dy, solid_walls)

    if dx or dy:
        STATE._footstep_timer -= 1
        if STATE._footstep_timer <= 0:
            audio.play("footstep", volume=0.25)
            STATE._footstep_timer = 27 + random.randint(0, 8)


def _update_weapon_timers():
    if STATE.shot_cooldown > 0:
        STATE.shot_cooldown -= 1
    if STATE.is_reloading:
        STATE.reload_timer -= 1
        if STATE.reload_timer <= 0:
            STATE.is_reloading = False
            if STATE.spare_mags > 0:
                STATE.spare_mags -= 1
                STATE.ammo_in_mag = STATE.mag_size
    if STATE.melee_active:
        STATE.melee_timer -= 1
        if STATE.player and not STATE.melee_hit_done and STATE.melee_timer <= 10:
            px, py = _player_pos()
            for e in STATE.enemies:
                if math.hypot(e.rect.centerx - px, e.rect.centery - py) < 55:
                    e.hp -= 75
                    e.on_hit(px, py)
                    STATE.melee_hit_done = True
                    STATE.shots_hit += 1
                    _burst(e.rect.centerx, e.rect.centery, C_BLOOD, 12)
                    break
        if STATE.melee_timer <= 0:
            STATE.melee_active = False
            STATE.melee_hit_done = False


def _update_molotovs(solid_walls):
    player_hit = False
    for b in STATE.thrown_bottles[:]:
        prev_x, prev_y = b["x"], b["y"]
        b["t"] = min(1.0, b["t"] + b["speed"])
        nx = b["start_x"] + (b["tx"] - b["start_x"]) * b["t"]
        ny = b["start_y"] + (b["ty"] - b["start_y"]) * b["t"]
        ix, iy = nx, ny
        hit_wall = False
        for wall in solid_walls:
            clipped = wall.clipline((prev_x, prev_y), (nx, ny))
            if clipped:
                ix, iy = clipped[0]
                hit_wall = True
                break
        if b["t"] >= 1.0 or hit_wall:
            _spawn_molotov_fire(ix, iy, solid_walls)
            STATE.thrown_bottles.remove(b)
        else:
            b["x"], b["y"] = nx, ny

    for fire in STATE.molotov_fires[:]:
        fire["life"] -= 1
        fx, fy = fire["x"], fire["y"]
        if random.random() < 0.55:
            STATE.particles.append(Particle(fx + random.randint(-55, 55), fy + random.randint(-45, 45), random.choice([(255, 95, 25), (255, 160, 45), (70, 65, 60)])))
        for e in STATE.enemies:
            d = math.hypot(e.rect.centerx - fx, e.rect.centery - fy)
            if d < MOLOTOV_RADIUS and not _wall_blocks_line(fx, fy, e.rect.centerx, e.rect.centery, solid_walls):
                e.hp -= MOLOTOV_ENEMY_DAMAGE * (0.5 + (1 - d / MOLOTOV_RADIUS))
                e.alerted = True
                e.state = "CHASE"
        if STATE.player and not STATE.player_dead and not _movement_locked():
            d = math.hypot(STATE.player.rect.centerx - fx, STATE.player.rect.centery - fy)
            if d < MOLOTOV_RADIUS * 0.7 and not _wall_blocks_line(fx, fy, STATE.player.rect.centerx, STATE.player.rect.centery, solid_walls):
                STATE.player.hp -= MOLOTOV_PLAYER_DAMAGE
                player_hit = True
        if fire["life"] <= 0:
            STATE.molotov_fires.remove(fire)

    if STATE.player and not STATE.player_dead:
        STATE.fire_zones[:] = [fz for fz in STATE.fire_zones if fz.update(STATE.enemies, STATE.player.rect, audio)]
    return player_hit


def _update_sniper_enemy(e, solid_walls):
    """Mission 2 sniper: fixed nest, long sightline, deliberate shots."""
    if not STATE.player or STATE.player_dead:
        return
    px, py = _player_pos()
    dx = px - e.rect.centerx
    dy = py - e.rect.centery
    e.angle = math.atan2(dy, dx)
    if e.fire_cooldown > 0:
        e.fire_cooldown -= 1
    if _wall_blocks_line(e.rect.centerx, e.rect.centery, px, py, solid_walls):
        return
    if e.fire_cooldown <= 0:
        dist = max(1.0, math.hypot(dx, dy))
        vx, vy = dx / dist, dy / dist
        STATE.bullets.append({
            "x": e.rect.centerx + vx * 22,
            "y": e.rect.centery + vy * 22,
            "prev_x": e.rect.centerx + vx * 22,
            "prev_y": e.rect.centery + vy * 22,
            "dx": vx * 28,
            "dy": vy * 28,
            "life": 34,
            "owner": "enemy",
            "damage": 26,
        })
        STATE.mission2_sniper_traces.append({"from": (e.rect.centerx, e.rect.centery), "to": (px, py), "life": 10})
        audio.play("gunshot_distant", volume=0.55)
        e.fire_cooldown = 115 + random.randint(0, 45)


def _update_bullets_enemies(solid_walls):
    player_hit = False
    for b in STATE.bullets[:]:
        b["prev_x"], b["prev_y"] = b["x"], b["y"]
        b["x"] += b["dx"]
        b["y"] += b["dy"]
        b["life"] -= 1
        hit = False
        for wall in solid_walls:
            if wall.clipline((b["prev_x"], b["prev_y"]), (b["x"], b["y"])):
                hit = True
                _burst(b["x"], b["y"], (130, 125, 110), 4)
                audio.play("bullet_impact", volume=0.16)
                break
        if not hit and b.get("owner") == "player":
            for e in STATE.enemies:
                if math.hypot(b["x"] - e.rect.centerx, b["y"] - e.rect.centery) < 22:

                    # You cannot pistol-snipe the sniper from the open road.
                    # If your bullet can reach him, he has already seen you.
                    if getattr(e, "is_mission2_sniper", False) and not getattr(STATE, "mission2_sniper_suppressed", False):
                        hit = True

                        STATE.mission2_sniper_pressure_timer = 0
                        STATE.mission2_sniper_pending_timer = min(
                            getattr(STATE, "mission2_sniper_pending_timer", 999),
                            SNIPER_CLOSE_AIM_FRAMES,
                        )
                        STATE.mission2_sniper_target = _player_pos()
                        _add_sniper_laser(STATE.mission2_sniper_target, life=20)

                        STATE.floating_texts.append([
                            STATE.player.rect.centerx,
                            STATE.player.rect.centery - 52,
                            "SNIPER HAS YOUR ANGLE",
                            (255, 60, 40),
                            70,
                        ])

                        _burst(b["x"], b["y"], (140, 135, 120), 10, speed_mul=1.4)
                        audio.play("concrete_impact", volume=0.45)
                        break

                    e.hp -= 50
                    e.on_hit(STATE.player.rect.centerx, STATE.player.rect.centery)
                    hit = True
                    STATE.shots_hit += 1
                    STATE.blood_decals.append([b["x"], b["y"], 140, random.randint(8, 16)])
                    _burst(b["x"], b["y"], C_BLOOD, 14)
                    audio.play("bullet_impact_enemy", volume=0.42)
                    break
        if not hit and b.get("owner") == "enemy" and STATE.player and not STATE.player_dead:
            if math.hypot(b["x"] - STATE.player.rect.centerx, b["y"] - STATE.player.rect.centery) < 22:
                if getattr(STATE, "mission2_grace_timer", 0) <= 0:
                    STATE.player.hp -= b.get("damage", ENEMY_DAMAGE)
                    player_hit = True
                hit = True
                STATE.screen_shake = max(STATE.screen_shake, 6)
                _burst(b["x"], b["y"], C_BLOOD, 8)
                audio.play("bullet_impact_enemy", volume=0.5)
        if hit or b["life"] <= 0 or not (-500 < b["x"] < MAP_W * TILE_SIZE + 500 and -500 < b["y"] < MAP_H * TILE_SIZE + 500):
            if b in STATE.bullets:
                STATE.bullets.remove(b)

    for e in STATE.enemies[:]:
        if STATE.player and not STATE.player_dead:
            e.alerted = True
            if getattr(e, "is_mission2_sniper", False):
                # Mission 2's post-ambush sniper is handled by the authored
                # kill-box pressure system, not generic enemy fire. Keep the
                # actor facing the player for drawing/feedback.
                px, py = _player_pos()
                e.angle = math.atan2(py - e.rect.centery, px - e.rect.centerx)
            else:
                if e.state in ("IDLE", "PATROL", "SEARCH"):
                    e.state = "CHASE"
                e.last_seen_pos = _player_pos()
                e.update(STATE.player.rect.centerx, STATE.player.rect.centery, solid_walls, STATE.bullets, STATE.particles, STATE.enemies, STATE.fire_zones)
        if e.hp <= 0:
            STATE.hostiles_killed += 1
            STATE.blood_decals.append([e.rect.centerx, e.rect.centery, 165, random.randint(12, 22)])
            _burst(e.rect.centerx, e.rect.centery, C_BLOOD, 24)
            roll = random.random()
            if roll < 0.28:
                STATE.pickups.append([e.rect.centerx, e.rect.centery, "bandage"])
            elif roll < 0.56 and STATE.spare_mags < 5:
                STATE.pickups.append([e.rect.centerx, e.rect.centery, "mag"])
            STATE.enemies.remove(e)
    return player_hit


def _update_pickups():
    if not STATE.player or STATE.player_dead:
        return
    for pu in STATE.pickups[:]:
        if math.hypot(pu[0] - STATE.player.rect.centerx, pu[1] - STATE.player.rect.centery) > 32:
            continue
        kind = pu[2]
        if kind == "bandage":
            old = STATE.player.hp
            STATE.player.hp = min(STATE.player.max_hp, STATE.player.hp + 35)
            STATE.floating_texts.append([pu[0], pu[1] - 16, f"+{int(STATE.player.hp - old)} HP", (100, 220, 100), 70])
        elif kind == "mag":
            STATE.spare_mags = min(5, STATE.spare_mags + 1)
            STATE.floating_texts.append([pu[0], pu[1] - 16, "+1 MAG", C_TERMINAL, 70])
        audio.play("pickup", volume=0.45)
        STATE.pickups.remove(pu)

def _update_camera():
    target = STATE.mission2_camera_focus or _player_pos()
    tx, ty = target

    if getattr(STATE, "mission2_camera_snap", None):
        tx, ty = STATE.mission2_camera_snap
        STATE.mission2_camera_snap = None

    sx = sy = 0
    if getattr(STATE, "reduced_motion", False):
        STATE.screen_shake = 0
    elif STATE.screen_shake > 0:
        sx = random.randint(-STATE.screen_shake, STATE.screen_shake)
        sy = random.randint(-STATE.screen_shake, STATE.screen_shake)
        STATE.screen_shake = max(0, STATE.screen_shake - 1)

    STATE.cam_kick_x *= 0.82
    STATE.cam_kick_y *= 0.82

    desired_x, desired_y = _camera_offset_for_target((
        tx + STATE.cam_kick_x + sx,
        ty + STATE.cam_kick_y + sy,
    ))

    if not getattr(STATE, "mission2_camera_initialized", False):
        STATE.camera_x = desired_x
        STATE.camera_y = desired_y
        STATE.mission2_camera_initialized = True
        return STATE.camera_x, STATE.camera_y

    if getattr(STATE, "mission2_stage", "") == STAGE_SNIPER:
        # Smoothstep easing — slow start, slow end, fast middle
        dx = desired_x - STATE.camera_x
        dy = desired_y - STATE.camera_y
        dist = math.hypot(dx, dy)
        if dist < 1.0:
            STATE.camera_x = desired_x
            STATE.camera_y = desired_y
        else:
            factor = min(1.0, 0.08 * (1.0 + dist / 600.0))
            factor = factor * factor * (3.0 - 2.0 * factor)
            STATE.camera_x += dx * factor
            STATE.camera_y += dy * factor
    elif getattr(STATE, "mission2_camera_focus_timer", 0) > 0:
        ease = 0.38
        STATE.camera_x += (desired_x - STATE.camera_x) * ease
        STATE.camera_y += (desired_y - STATE.camera_y) * ease
    else:
        ease = 0.14
        STATE.camera_x += (desired_x - STATE.camera_x) * ease
        STATE.camera_y += (desired_y - STATE.camera_y) * ease

    max_x = max(0, MAP_W * TILE_SIZE - WIDTH)
    max_y = max(0, MAP_H * TILE_SIZE - HEIGHT)
    STATE.camera_x = max(0, min(max_x, STATE.camera_x))
    STATE.camera_y = max(0, min(max_y, STATE.camera_y))

    if getattr(STATE, "mission2_stage", "") == STAGE_SNIPER:
        if not getattr(STATE, "reduced_motion", False):
            timer = getattr(STATE, "mission2_sniper_phase_timer", 0)
            STATE.camera_x += math.sin(timer * 0.13) * 0.8
            STATE.camera_y += math.cos(timer * 0.17) * 0.6

    return STATE.camera_x, STATE.camera_y


def game_update(keys):
    global _last_keys
    _last_keys = keys
    if STATE.gc_flash_timer > 0:
        STATE.gc_flash_timer -= 1
    if STATE.player and STATE.player.hp <= 0 and not STATE.player_dead:
        STATE.player_dead = True
        STATE.death_timer = 120
        STATE.screen_shake = 22
        _burst(STATE.player.rect.centerx, STATE.player.rect.centery, C_BLOOD, 48, speed_mul=1.6)
    if STATE.player_dead:
        STATE.death_timer -= 1
        if STATE.death_timer <= 0:
            if not STATE.mission_end_time:
                STATE.mission_end_time = pygame.time.get_ticks()
            STATE.request_death_screen = True

    solid_walls = _solid_walls()
    _update_player(keys, solid_walls)
    _update_travel_triggers()
    _update_squad_ambient()
    _update_weapon_timers()
    player_hit = _update_molotovs(solid_walls)
    if STATE.mission2_stage not in (STAGE_SNIPER, STAGE_BLACKOUT, STAGE_DONE) and not getattr(STATE, "mission2_card", None):
        player_hit = _update_bullets_enemies(solid_walls) or player_hit
    if player_hit:
        STATE.screen_shake = max(STATE.screen_shake, 4)
    _update_pickups()
    _update_stage(keys)
    _update_particles_texts(solid_walls)

    if getattr(STATE, "mission2_chromatic_timer", 0) > 0:
        STATE.mission2_chromatic_timer -= 1
    if getattr(STATE, "mission2_sniper_silhouette_flash", 0) > 0:
        STATE.mission2_sniper_silhouette_flash -= 1
    if getattr(STATE, "mission2_radio_muffle_timer", 0) > 0:
        STATE.mission2_radio_muffle_timer -= 1
    if getattr(STATE, "mission2_input_freeze", 0) > 0:
        STATE.mission2_input_freeze -= 1

    if getattr(STATE, "mission2_heartbeat_active", False):
        STATE.mission2_heartbeat_timer += 1
        if STATE.mission2_heartbeat_timer % 72 == 0:
            audio.play("heartbeat", volume=0.18)
        # Point 12: Fade heartbeat as player takes first action (moves, shoots, etc.)
        # Also hard cap at ~12 seconds (720 frames)
        player_acting = False
        if STATE.player and not STATE.player_dead:
            keys = _last_keys
            if keys and (keys[pygame.K_w] or keys[pygame.K_s] or keys[pygame.K_a] or keys[pygame.K_d] or keys[pygame.K_e] or keys[pygame.K_SPACE] or keys[pygame.K_r] or keys[pygame.K_g] or keys[pygame.K_v]):
                player_acting = True
        if (STATE.mission2_stage == STAGE_COMBAT and (STATE.mission2_heartbeat_timer > 720 or player_acting)):
            STATE.mission2_heartbeat_active = False

    return _update_camera()

def _world(pos, fcx, fcy):
    # Snap world-to-screen coordinates to whole pixels.  Mission 2 camera values
    # can be fractional because the player speed is fractional; drawing tiles and
    # clue icons at fractional positions causes shimmering/"glitchy" textures
    # while walking.
    return int(round(pos[0] - fcx)), int(round(pos[1] - fcy))

def _draw_actor(surface, actor, fcx, fcy):
    """Top-down KRAP squadmate. Pose is driven entirely by activity string."""
    x = actor["pos"][0] - fcx
    y = actor["pos"][1] - fcy
    name  = actor.get("name", "")
    activity = (actor.get("activity") or "standing").lower()

    # alive is ONLY used for label colour and blood — never for pose branch.
    alive = actor.get("alive", True)

    accent = {
        "GABI":   (220, 175,  70),
        "SEBY":   ( 85, 170, 225),
        "DARIUS": (110, 220, 145),
    }.get(name, C_KRAP_BLUE)
    name_col = (150, 185, 230) if alive else (130, 80, 80)

    # --- Pose flags ---------------------------------------------------
    is_down      = "down"   in activity or "dead" in activity
    is_drag_body = "dragging_body" in activity
    is_drag      = ("drag" in activity) and not is_drag_body
    is_reach     = "reach"  in activity or "radio" in activity
    is_frozen    = "frozen" in activity or "shock" in activity
    is_crouch    = is_drag or is_reach or is_frozen or "crouch" in activity or "cover" in activity
    is_shout     = "shout"  in activity or "yell" in activity

    if is_down:
        # ---- DOWNED body ------------------------------------------------
        pygame.draw.ellipse(surface, (0, 0, 0, 110),
                            (x - 28, y + 2, 56, 16))
        pygame.draw.rect(surface, (36, 44, 52),
                         (x - 22, y - 10, 44, 18), border_radius=6)

                         (x - 12, y - 10, 24, 4), border_radius=2)

        pygame.draw.line(surface, (16, 16, 15),
                         (x - 26, y + 6), (x + 26, y - 4), 4)
        # Head
        pygame.draw.circle(surface, C_PLAYER_SKIN,
                           (int(x - 20), int(y - 3)), 5)

        pygame.draw.ellipse(surface, (90, 14, 14, 190),

def _draw_prop_icon(surface, kind, cx, cy, scale=1.0, active=True, color=C_TERMINAL):

    s = max(0.35, float(scale))

def _draw_world_prop(surface, item, fcx, fcy, active=True):
    sx, sy = _world(item["pos"], fcx, fcy)
    kind = item.get("kind", item.get("id", "document"))
    if item in getattr(STATE, "mission2_clues", []):
        prop_col = (245, 190, 65)

        prop_col = (255, 190, 90)
    else:
        prop_col = C_TERMINAL

    # Small object, readable silhouette: visibility comes from a restrained halo

    halo_r = PROP_WORLD_HALO_RADIUS

    pygame.draw.circle(halo, (*prop_col, halo_alpha), (halo_r + 4, halo_r + 4), halo_r)

    surface.blit(halo, (sx - halo_r - 4, sy - halo_r - 4))

    _draw_prop_icon(surface, kind, sx, sy, scale=PROP_WORLD_SCALE, active=active, color=prop_col)


def _draw_marker(surface, pos, label, col, fcx, fcy, radius=MARKER_RADIUS):
    sx, sy = _world(pos, fcx, fcy)
    pulse = int(80 + math.sin(time.time() * 5) * 35)
    if -60 < sx < WIDTH + 60 and -60 < sy < HEIGHT + 60:
        ring = pygame.Surface((radius * 3, radius * 3), pygame.SRCALPHA)
        pygame.draw.circle(ring, (*col, pulse), (radius * 3 // 2, radius * 3 // 2), radius, 2)
        pygame.draw.circle(ring, (*col, pulse // 2), (radius * 3 // 2, radius * 3 // 2), radius + 5, 1)
        surface.blit(ring, (sx - radius * 3 // 2, sy - radius * 3 // 2))
        txt = font_terminal_small.render(label, True, col)
        surface.blit(txt, (sx - txt.get_width() // 2, sy - radius - 23))
    else:
        # Edge arrow.
        cx = max(32, min(WIDTH - 32, sx))
        cy = max(90, min(HEIGHT - 58, sy))
        ang = math.atan2(sy - HEIGHT / 2, sx - WIDTH / 2)
        pts = [
            (cx + math.cos(ang) * 16, cy + math.sin(ang) * 16),
            (cx + math.cos(ang + 2.45) * 12, cy + math.sin(ang + 2.45) * 12),
            (cx + math.cos(ang - 2.45) * 12, cy + math.sin(ang - 2.45) * 12),
        ]
        pygame.draw.polygon(surface, col, pts)
        txt = font_terminal_small.render(label, True, col)
        surface.blit(txt, (cx - txt.get_width() // 2, cy + 16))

def _draw_card(surface):
    card = getattr(STATE, "mission2_card", None)
    if not card:
        return
    age = card.get("age", 0)
    open_t = min(1.0, age / 18.0)
    ease = 1 - (1 - open_t) ** 3
    overlay = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA)
    overlay.fill((0, 0, 0, int(170 * ease)))
    surface.blit(overlay, (0, 0))
    col = card["color"]
    w, h = 720, 330
    x = WIDTH // 2 - w // 2
    y = int(HEIGHT // 2 - h // 2 + (1 - ease) * 80)
    shadow = pygame.Surface((w + 22, h + 22), pygame.SRCALPHA)
    shadow.fill((0, 0, 0, int(105 * ease)))
    surface.blit(shadow, (x + 10, y + 12))
    pygame.draw.rect(surface, (5, 10, 13), (x, y, w, h), border_radius=6)
    pygame.draw.rect(surface, col, (x, y, w, h), 2, border_radius=6)
    pygame.draw.rect(surface, (2, 5, 8), (x, y, w, 54), border_radius=6)
    pygame.draw.rect(surface, col, (x, y, 5, h), border_radius=3)
    title = font_speaker.render(card["title"], True, col)
    surface.blit(title, (x + 28, y + 13))
    preview_w, preview_h = 142, 158
    prop_x, prop_y = x + 102, y + 178
    preview = pygame.Surface((preview_w, preview_h), pygame.SRCALPHA)
    scan_y = int((age * 2) % preview_h)
    pygame.draw.rect(preview, (0, 0, 0, 70), (0, 0, preview_w, preview_h), border_radius=8)
    pygame.draw.rect(preview, (*col, 110), (0, 0, preview_w, preview_h), 1, border_radius=8)
    _draw_prop_icon(preview, card.get("kind", "document"), preview_w // 2, preview_h // 2, scale=PROP_CARD_SCALE, active=True, color=col)
    pygame.draw.line(preview, (*col, 125), (8, scan_y), (preview_w - 8, scan_y), 1)
    surface.blit(preview, (prop_x - preview_w // 2, prop_y - preview_h // 2))
    yy = y + 82
    text_x = x + 195
    max_w = w - 225
    for line in card["lines"]:
        words = line.split()
        cur = ""
        wrapped = []
        for word in words:
            test = (cur + " " + word).strip()
            if font_dialogue.size(test)[0] > max_w and cur:
                wrapped.append(cur)
                cur = word
            else:
                cur = test
        if cur:
            wrapped.append(cur)
        for part in wrapped:
            surf = font_dialogue.render(part, True, C_WHITE)
            surface.blit(surf, (text_x, yy))
            yy += 32
        yy += 8
    hint = font_terminal.render("PRESS [E] TO CLOSE", True, col)
    if age >= 18 or (pygame.time.get_ticks() // 400) % 2 == 0:
        surface.blit(hint, (x + w - hint.get_width() - 26, y + h - 40))

def _wrap_text(text, font, max_width):
    """Simple word-wrap that respects an existing pygame font."""
    words = text.split()
    if not words:
        return [""]
    lines = []
    cur = words[0]
    for w in words[1:]:
        trial = cur + " " + w
        if font.size(trial)[0] <= max_width:
            cur = trial
        else:
            lines.append(cur)
            cur = w
    lines.append(cur)
    return lines

def game_draw(fcx: float, fcy: float) -> None:
    screen.fill((25, 26, 26))
    # Render from an integer camera.  This removes the tile/prop seam shimmer
    # that happens when the camera scrolls by fractional pixels.
    cam_x, cam_y = int(round(fcx)), int(round(fcy))

    start_col = max(0, int(cam_x // TILE_SIZE) - 1)
    end_col = min(MAP_W, int((cam_x + WIDTH) // TILE_SIZE) + 2)
    start_row = max(0, int(cam_y // TILE_SIZE) - 1)
    end_row = min(MAP_H, int((cam_y + HEIGHT) // TILE_SIZE) + 2)

    tile_base = {
        ".": (32, 33, 33),   # wet asphalt / highway
        "g": (40, 36, 32),   # muddy shoulder
        "d": (29, 35, 34),   # drainage ditch
        "s": (34, 37, 35),   # service road
        "O": (29, 36, 38),   # interior/control sector
        "X": (29, 38, 32),
        "T": (35, 32, 28),
        "C": (54, 54, 52),
        "B": (82, 82, 76),
    }
    for row in range(start_row, end_row):
        for col in range(start_col, end_col):
            sx, sy = col * TILE_SIZE - cam_x, row * TILE_SIZE - cam_y
            tile = MISSION2_MAP[row][col]
            base = tile_base.get(tile, (31, 32, 32))
            pygame.draw.rect(screen, base, (sx, sy, TILE_SIZE, TILE_SIZE))
            if (row + col) % 3 == 0:
                seam = (25, 26, 26) if tile in ".TOX" else (31, 30, 28)
                pygame.draw.line(screen, seam, (sx, sy + TILE_SIZE - 1), (sx + TILE_SIZE, sy + TILE_SIZE - 1), 1)
            # Deterministic micro-detail keeps the big map from reading as an
            # empty grid while avoiding per-frame random shimmer.
            seed = row * 97 + col * 131
            if tile == ".":
                if seed % 11 == 0:
                    pygame.draw.line(screen, (22, 22, 22), (sx + 10, sy + 18), (sx + 42, sy + 28), 1)
                if seed % 17 == 0:
                    pygame.draw.line(screen, (45, 42, 38), (sx + 4, sy + 50), (sx + 58, sy + 44), 2)
            elif tile == "g":
                if seed % 7 == 0:
                    pygame.draw.circle(screen, (55, 48, 38), (sx + 18, sy + 38), 3)
                if seed % 13 == 0:
                    pygame.draw.line(screen, (48, 58, 42), (sx + 45, sy + 25), (sx + 49, sy + 18), 1)
            elif tile == "d":
                if seed % 5 == 0:
                    pygame.draw.ellipse(screen, (24, 31, 38), (sx + 8, sy + 34, 34, 13))
            elif tile in "sO":
                if seed % 9 == 0:
                    pygame.draw.rect(screen, (23, 27, 28), (sx + 8, sy + 8, 12, 3))

    # Vertical lane stripes for the expanded Route 12 corridor.
    for xcol in (28, 36, 44):
        x = xcol * TILE_SIZE + TILE_SIZE // 2 - cam_x
        for y in range(-int(cam_y) % 150 - 150, HEIGHT + 150, 150):
            pygame.draw.rect(screen, (145, 130, 60), (x, y, 5, 72))

    for v in STATE.mission2_vehicles:
        _draw_vehicle(screen, v["rect"], v["kind"], cam_x, cam_y)

    barrier_ids = {id(r) for r in STATE.mission2_barriers}
    for wall in STATE.walls:
        if wall is STATE.mission2_tanker_rect:
            continue
        dr = pygame.Rect(wall.x - cam_x, wall.y - cam_y, wall.w, wall.h)
        if id(wall) in barrier_ids:
            pygame.draw.rect(screen, (92, 92, 84), dr, border_radius=4)
            pygame.draw.rect(screen, (38, 38, 36), dr, 2, border_radius=4)
            pygame.draw.line(screen, (130, 130, 120), (dr.x + 6, dr.y + 8), (dr.right - 6, dr.y + 8), 1)
        else:
            pygame.draw.rect(screen, (63, 61, 58), dr)
            pygame.draw.rect(screen, (106, 100, 90), dr, 2)

    _draw_gantry(screen, cam_x, cam_y)

    for decal in STATE.blood_decals[:]:
        surf = pygame.Surface((decal[3] * 2, decal[3] * 2), pygame.SRCALPHA)
        pygame.draw.circle(surf, (130, 10, 10, int(decal[2])), (decal[3], decal[3]), decal[3])
        screen.blit(surf, (decal[0] - decal[3] - cam_x, decal[1] - decal[3] - cam_y))
        decal[2] -= 0.25
        if decal[2] <= 0:
            STATE.blood_decals.remove(decal)

    _draw_tanker(screen, cam_x, cam_y)
    for fz in STATE.fire_zones:
        fz.draw(screen, cam_x, cam_y)
    _draw_story_props(screen, cam_x, cam_y)

    _draw_interactables(screen, cam_x, cam_y)

    for b in STATE.bullets:
        pygame.draw.circle(screen, C_BULLET, (int(b["x"] - cam_x), int(b["y"] - cam_y)), 2)
        pygame.draw.line(screen, (255, 220, 140), (b["x"] - cam_x, b["y"] - cam_y), (b.get("prev_x", b["x"] - b["dx"]) - cam_x, b.get("prev_y", b["y"] - b["dy"]) - cam_y), 2)

    for tr in STATE.mission2_sniper_traces:
        sx1, sy1 = _world(tr["from"], cam_x, cam_y)
        sx2, sy2 = _world(tr["to"], cam_x, cam_y)
        life_pct = max(0, min(1, tr["life"] / max(tr.get("max_life", 1), 1)))
        a = max(0, min(255, int(life_pct * 200)))
        kind = tr.get("kind", "body")
        if kind == "near_miss":
            base_c = (255, 220, 180)
        elif kind == "concrete":
            base_c = (200, 185, 160)
        else:
            base_c = (220, 180, 170)
        segments = 9
        seg_dist = 6
        trace_surf = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA)
        for i in range(segments):
            frac = i / segments
            jfrac = (i + 1) / segments
            mx1 = sx1 + (sx2 - sx1) * frac + random.randint(-3, 3)
            my1 = sy1 + (sy2 - sy1) * frac + random.randint(-3, 3)
            mx2 = sx1 + (sx2 - sx1) * jfrac + random.randint(-3, 3)
            my2 = sy1 + (sy2 - sy1) * jfrac + random.randint(-3, 3)
            seg_a = max(0, a - i * 15)
            thickness = max(1, 4 - i // 3)
            pygame.draw.line(trace_surf, (*base_c, seg_a), (mx1, my1), (mx2, my2), thickness)
        screen.blit(trace_surf, (0, 0))
        if life_pct > 0.6:
            spark = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA)
            end_a = int((life_pct - 0.6) / 0.4 * 180)
            pygame.draw.circle(spark, (255, 200, 150, end_a), (int(sx2), int(sy2)), 4)
            pygame.draw.circle(spark, (255, 240, 220, end_a // 2), (int(sx2), int(sy2)), 8)
            screen.blit(spark, (0, 0))

    _draw_sniper_lasers_world(screen, cam_x, cam_y)
    _draw_sniper_muzzle_flashes(screen, cam_x, cam_y)

    for b in STATE.thrown_bottles:
        t = b["t"]
        bx = b["x"] - cam_x
        by = b["y"] - cam_y - math.sin(t * math.pi) * 40
        pygame.draw.circle(screen, (255, 120, 35), (int(bx), int(by)), 5)
        pygame.draw.line(screen, (90, 55, 25), (bx, by), (bx + 3, by - 12), 3)

    for actor in STATE.mission2_squad:
        _draw_actor(screen, actor, cam_x, cam_y)

    for e in STATE.enemies:
        e.draw(screen, cam_x, cam_y)

    reload_progress = 1.0 - STATE.reload_timer / 60.0 if STATE.is_reloading and STATE.reload_timer > 0 else 0.0
    melee_progress = 1.0 - STATE.melee_timer / 18.0 if STATE.melee_active and STATE.melee_timer > 0 else 0.0
    if STATE.player:
        STATE.player.draw(
            screen,
            cam_x,
            cam_y,
            STATE.walls,
            STATE.is_reloading,
            reload_progress,
            STATE.melee_active,
            melee_progress,
            is_moving=not STATE.player_dead and _last_keys and (_last_keys[pygame.K_w] or _last_keys[pygame.K_s] or _last_keys[pygame.K_a] or _last_keys[pygame.K_d]),
        )

    for p in STATE.particles:
        p.draw(screen, cam_x, cam_y)
    for c in STATE.casings:
        c.draw(screen, cam_x, cam_y)
    for pu in STATE.pickups:
        px = pu[0] - cam_x
        py = pu[1] - cam_y + math.sin(time.time() * 3 + pu[0]) * 3
        if pu[2] == "bandage":
            pygame.draw.rect(screen, (240, 240, 240), (px - 10, py - 10, 20, 20), border_radius=3)
            pygame.draw.rect(screen, (220, 40, 40), (px - 2, py - 6, 4, 12))
            pygame.draw.rect(screen, (220, 40, 40), (px - 6, py - 2, 12, 4))
        else:
            pygame.draw.rect(screen, (95, 90, 82), (px - 7, py - 12, 14, 24))

    # Rain overlay.  Suppressed during the kill-box silence; intensity > 1.0
    # spawns extra drops (rain spike on first kill -- Point 9 of the rework).
    rain_suppressed = getattr(STATE, "mission2_rain_suppressed", False)
    rain_intensity = getattr(STATE, "mission2_rain_intensity", 1.0)
    if not rain_suppressed:
        for drop in STATE.mission2_rain_drops:
            x, y, length = drop
            pygame.draw.line(screen, C_RAIN, (x, y), (x - 4, y + length), 1)
            drop[0] -= 3
            drop[1] += length
            if drop[1] > HEIGHT + 30:
                drop[0] = random.randint(0, WIDTH)
                drop[1] = random.randint(-90, -10)
        # Extra sheets of rain when intensity is boosted (first-kill spike).
        if rain_intensity > 1.05:
            extra = int(len(STATE.mission2_rain_drops) * (rain_intensity - 1.0))
            for _ in range(extra):
                x = random.randint(0, WIDTH)
                y = random.randint(0, HEIGHT)
                length = random.randint(10, 24)
                pygame.draw.line(screen, C_RAIN, (x, y), (x - 4, y + length), 1)
        # Decay intensity back toward 1.0 so the spike fades naturally.
        if rain_intensity > 1.01:
            STATE.mission2_rain_intensity = max(1.0, rain_intensity - 0.004)

    if STATE.muzzle_flash_timer > 0 and STATE.player:
        tip_x, tip_y = STATE.player.barrel_tip()
        screen.blit(MUZZLE_FLASH, (int(tip_x - cam_x - 40), int(tip_y - cam_y - 40)))
        STATE.muzzle_flash_timer -= 1

    ch_x, ch_y = get_scaled_mouse()
    if not getattr(STATE, "mission2_card", None) and STATE.mission2_stage not in (STAGE_SNIPER, STAGE_BLACKOUT):
        pygame.draw.circle(screen, C_ORANGE, (ch_x, ch_y), 14, 1)
        pygame.draw.circle(screen, C_ORANGE, (ch_x, ch_y), 2)
        for x1, y1, x2, y2 in [(ch_x - 12, ch_y, ch_x - 3, ch_y), (ch_x + 3, ch_y, ch_x + 12, ch_y), (ch_x, ch_y - 12, ch_x, ch_y - 3), (ch_x, ch_y + 3, ch_x, ch_y + 12)]:
            pygame.draw.line(screen, C_ORANGE, (x1, y1), (x2, y2), 1)

    if STATE.player_dead:
        overlay = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA)
        overlay.fill((120, 10, 10, 90))
        screen.blit(overlay, (0, 0))
    if STATE.gc_flash_timer > 0 and not getattr(STATE, "reduced_motion", False):
        flash = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA)
        flash.fill((255, 245, 220, min(255, STATE.gc_flash_timer * 18)))
        screen.blit(flash, (0, 0))

    if getattr(STATE, "mission2_sniper_kill_flash", 0) > 0:
        a = min(220, STATE.mission2_sniper_kill_flash * 10)
        flash = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA)
        flash.fill((255, 40, 25, a))
        screen.blit(flash, (0, 0))

    draw_vignette(screen, strength=150)
    draw_scanlines(screen)

    if STATE.mission2_stage == STAGE_SNIPER:
        _draw_sniper_overlay(screen)
    elif STATE.mission2_stage == STAGE_BLACKOUT:
        _draw_blackout_overlay(screen)
    else:
        _draw_hud(screen)
        _draw_radio_text(screen)
        if STATE.show_minimap and STATE.minimap:
            STATE.minimap.draw(screen, STATE.player, STATE.enemies, cam_x, cam_y, notebook_pos=UPLOAD_TERMINAL_POS, exfil_pos=EXFIL_POS)
    _draw_card(screen)

    # Hard black cut (Point 11): overrides everything for a few frames at the
    # cutscene -> gameplay handoff.  Decrement here so the counter ticks once
    # per drawn frame regardless of update timing.
    if getattr(STATE, "mission2_hard_black", 0) > 0:
        screen.fill((0, 0, 0))
        STATE.mission2_hard_black -= 1

def get_mission2_overview():
    return {
        "id": MISSION2_ID,
        "title": MISSION2_TITLE,
        "location": MISSION2_LOCATION,
        "style": MISSION2_PRIMARY_STYLE,
        "story_flags": MISSION2_STORY_FLAGS.copy(),
    }


def get_opening_cinematic():
    return list(OPENING_CINEMATIC)


def get_squadmate_death_cutscene():
    return list(SQUADMATE_DEATH_CUTSCENE)


def get_mission_phases():
    return list(MISSION_PHASES)


def get_mission2_zones():
    return {
        name: {
            "label": data["label"],
            "tile_rect": (data["tile_rect"].x, data["tile_rect"].y, data["tile_rect"].w, data["tile_rect"].h),
        }
        for name, data in ZONE_DEFS.items()
    }


def get_mission2_clue_defs():
    return [
        {"id": c["id"], "name": c["name"], "zone": c["zone"], "pos": c["pos"]}
        for c in MISSION2_CLUE_DEFS
    ]


def get_ending_cinematic():
    return list(mission2_ending_shots)


def get_all_dialogue_lines():
    out = []
    for group in (mission2_intro_shots, mission2_ending_shots, SQUADMATE_DEATH_CUTSCENE):
        for beat in group:
            if beat.get("speaker") and beat.get("text"):
                out.append({"speaker": beat["speaker"], "text": beat["text"]})
    return out
