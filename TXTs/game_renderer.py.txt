"""
game_renderer.py — Unified game drawing for all three missions.

Replaces the 3 separate game_draw() functions in mission.py, mission2.py,
and mission3.py with a single pipeline-driven renderer.

ARCHITECTURE:
  1. Each mission registers "draw submissions" into the RenderPipeline
     at the correct layer.
  2. Mission-specific elements are registered by the mission's own
     `populate_pipeline()` method.
  3. The shared skeleton (bullets, player, particles, HUD, post-FX)
     is drawn by this module — missions don't duplicate it.
  4. `flush()` draws everything in layer order with Y-sorting.

MIGRATION PATH:
  - mission.py: replace game_draw() body with:
        renderer = GameRenderer(fcx, fcy)
        renderer.populate_m1()
        renderer.flush()
  - mission2.py: replace game_draw() body with:
        renderer = GameRenderer(fcx, fcy)
        renderer.populate_m2()
        renderer.flush()
  - mission3.py: replace game_draw() body with:
        renderer = GameRenderer(fcx, fcy)
        renderer.populate_m3()
        renderer.flush()
"""

import math
import pygame
from render_pipeline import RenderPipeline
from config import (
    WIDTH, HEIGHT, TILE_SIZE, screen, C_BG_GAME,
    font_terminal_small, C_RAIN, C_BULLET, bullet_glow, MUZZLE_FLASH,
    draw_scanlines, draw_vignette,
)
from art_cache import get_blood_decal, get_glow, get_fullscreen_overlay
from state import STATE
from user_settings import SETTINGS
from camera import draw_bullets


class GameRenderer:
    """Unified game renderer using the RenderPipeline.

    Attributes:
        fcx, fcy: Raw camera floats (from update_camera).
        cam_x, cam_y: Integer-snapped camera (prevents tile shimmer).
        pipeline: The RenderPipeline instance for this frame.
    """

    def __init__(self, fcx, fcy, snap_camera=True):
        self.fcx = fcx
        self.fcy = fcy
        # Integer-snapped camera prevents tile/prop seam shimmer when
        # the camera scrolls by fractional pixels. M1 used raw floats;
        # M2/M3 snapped. We snap by default for consistency.
        if snap_camera:
            self.cam_x = int(round(fcx))
            self.cam_y = int(round(fcy))
        else:
            self.cam_x = fcx
            self.cam_y = fcy
        self.pipeline = RenderPipeline()

    # ─────────────────────────────────────────
    # SHARED SKELETON SUBMISSIONS
    # ─────────────────────────────────────────

    def _submit_shared_world(self):
        """Submit draw calls shared by all missions: bullets, thrown bottles,
        enemies, player, particles, casings, pickups, floating texts."""

        cam_x, cam_y = self.cam_x, self.cam_y

        # ── Bullets (shared helper from camera.py) ──
        self.pipeline.submit(
            'projectiles',
            lambda s: draw_bullets(s, STATE.bullets, cam_x, cam_y, with_glow=True),
        )

        # ── Thrown bottles ──
        self.pipeline.submit(
            'projectiles',
            lambda s: self._draw_thrown_bottles(s, cam_x, cam_y),
            sort_key=0,
        )

        # ── Enemies (Y-sorted by centery) ──
        for e in STATE.enemies:
            sort_key = e.rect.centery
            self.pipeline.submit(
                'entities',
                lambda s, _e=e: _e.draw(s, cam_x, cam_y),
                sort_key=sort_key,
            )

        # ── Player ──
        if STATE.player:
            import death_anim
            if not death_anim.should_skip_player_draw():
                reload_max_p = 90 if getattr(STATE, "current_weapon", "pistol") == "os12" else 60
                reload_progress = (1.0 - STATE.reload_timer / reload_max_p
                                   if STATE.is_reloading and STATE.reload_timer > 0 else 0.0)
                melee_progress = (1.0 - STATE.melee_timer / 18.0
                                  if STATE.melee_active and STATE.melee_timer > 0 else 0.0)
                # Import the module-level _last_keys from whichever mission is active.
                # Each mission module defines its own; we read it from game_core proxy.
                import game_core as gc
                last_keys = getattr(gc, '_last_keys', None)
                is_moving = (not STATE.player_dead and last_keys and (
                    last_keys[pygame.K_w] or last_keys[pygame.K_s] or
                    last_keys[pygame.K_a] or last_keys[pygame.K_d]
                ))
                self.pipeline.submit(
                    'entities',
                    lambda s: STATE.player.draw(
                        s, cam_x, cam_y, STATE.walls,
                        STATE.is_reloading, reload_progress,
                        STATE.melee_active, melee_progress,
                        is_moving=is_moving,
                    ),
                    sort_key=STATE.player.rect.centery,
                )

        # ── Particles ──
        for p in STATE.particles:
            self.pipeline.submit(
                'effects',
                lambda s, _p=p: _p.draw(s, cam_x, cam_y),
            )

        # ── Casings ──
        for c in STATE.casings:
            self.pipeline.submit(
                'effects',
                lambda s, _c=c: _c.draw(s, cam_x, cam_y),
            )

        # ── Pickups ──
        self.pipeline.submit(
            'items',
            lambda s: self._draw_pickups(s, cam_x, cam_y),
        )

        # ── Floating texts ──
        self.pipeline.submit(
            'floating',
            lambda s: self._draw_floating_texts(s, cam_x, cam_y),
        )

    def _submit_shared_overlays(self):
        """Submit overlay draw calls shared by all missions:
        muzzle flash, death overlay, flash overlay, post-FX, HUD."""

        cam_x, cam_y = self.cam_x, self.cam_y

        # ── Muzzle flash ──
        if STATE.muzzle_flash_timer > 0 and STATE.player:
            self.pipeline.submit(
                'overlay',
                lambda s: self._draw_muzzle_flash(s, cam_x, cam_y),
            )

        # ── Death overlay ──
        if STATE.player_dead:
            import death_anim
            self.pipeline.submit(
                'overlay',
                lambda s: death_anim.draw_death_overlay(s, cam_x, cam_y),
            )

        # ── GC flash overlay (grenade/blast white flash) ──
        if STATE.gc_flash_timer > 0 and not getattr(STATE, "reduced_motion", False):
            self.pipeline.submit(
                'overlay',
                lambda s: self._draw_gc_flash(s),
            )

        # ── Post-processing (scanlines + vignette) ──
        # Vignette strength varies by mission — set via self.vignette_strength
        vignette_strength = getattr(self, 'vignette_strength', 130)
        self.pipeline.submit(
            'postprocess',
            lambda s: draw_vignette(s, strength=vignette_strength),
        )
        self.pipeline.submit(
            'postprocess',
            lambda s: draw_scanlines(s),
        )

        # ── HUD (drawn after post-FX so it sits on top) ──
        self.pipeline.submit(
            'hud',
            lambda s: self._draw_hud(s),
        )

        # ── Camera tilt (death) — drawn absolutely last ──
        if STATE.player_dead:
            import death_anim
            self.pipeline.submit(
                'hud',
                lambda s: death_anim.apply_camera_tilt(s),
            )

    # ─────────────────────────────────────────
    # SHARED DRAW HELPERS
    # ─────────────────────────────────────────

    @staticmethod
    def _draw_thrown_bottles(surface, cam_x, cam_y):
        """Draw molotov bottles in flight."""
        for b in STATE.thrown_bottles:
            t = b["t"]
            bx = b["x"] - cam_x
            by = b["y"] - cam_y - math.sin(t * math.pi) * 40
            # Flame
            pygame.draw.circle(surface, (255, 120, 35), (int(bx), int(by)), 5)
            # Bottle neck
            pygame.draw.line(surface, (90, 55, 25), (bx, by), (bx + 3, by - 12), 3)

    @staticmethod
    def _draw_pickups(surface, cam_x, cam_y):
        """Draw pickup items. Missions can override by not calling this."""
        for pu in STATE.pickups:
            px = pu["x"] - cam_x
            py = pu["y"] - cam_y
            if -50 < px < WIDTH + 50 and -50 < py < HEIGHT + 50:
                # Generic pickup glow + rect
                glow = get_glow(14, (200, 200, 100), 40)
                surface.blit(glow, (px - 14, py - 14))
                rect = pygame.Rect(px - 10, py - 10, 20, 20)
                pygame.draw.rect(surface, (40, 40, 38), rect)
                pygame.draw.rect(surface, (100, 95, 80), rect, 1)

    @staticmethod
    def _draw_floating_texts(surface, cam_x, cam_y):
        """Draw floating text popups."""
        pinned = [ft for ft in STATE.floating_texts if len(ft) >= 6 and ft[5] == "pin"]
        unpinned = [ft for ft in STATE.floating_texts if not (len(ft) >= 6 and ft[5] == "pin")]
        for ft in unpinned:
            txt = font_terminal_small.render(ft[2], True, ft[3])
            surface.blit(txt, (ft[0] - cam_x - txt.get_width() // 2, ft[1] - cam_y))

    @staticmethod
    def _draw_muzzle_flash(surface, cam_x, cam_y):
        """Draw the player's muzzle flash at barrel tip."""
        if STATE.player:
            tip_x, tip_y = STATE.player.barrel_tip()
            surface.blit(
                MUZZLE_FLASH,
                (int(tip_x - cam_x - 40), int(tip_y - cam_y - 40)),
            )
            STATE.muzzle_flash_timer -= 1

    @staticmethod
    def _draw_gc_flash(surface):
        """Draw the grenade/blast white flash overlay."""
        flash_surf = get_fullscreen_overlay((255, 255, 255, 255))
        flash_surf.set_alpha(min(255, STATE.gc_flash_timer * 22))
        surface.blit(flash_surf, (0, 0))

    def _draw_hud(self, surface):
        """Draw the HUD. Overridden per-mission via self.hud_drawer."""
        hud_drawer = getattr(self, 'hud_drawer', None)
        if hud_drawer:
            hud_drawer(surface)

            'hud',
            lambda s: self._m1_draw_exfil_truck(s, cam_x, cam_y),
        )

        # Hotwire overlay (drawn on top of gameplay but below HUD)
        if getattr(STATE, 'truck_hotwiring', False):
            self.pipeline.submit(
                'overlay',
                lambda s: self._m1_draw_hotwire(s, cam_x, cam_y),
            )

            self.pipeline.submit(
                'overlay',
                lambda s: self._m1_draw_breach_flicker(s),
            )

        # Set HUD drawer for M1
        self.hud_drawer = self._m1_draw_hud

        # Shared overlays (post-FX, death, etc.)
        self._submit_shared_overlays()

    # ── M1-specific draw helpers ──

    @staticmethod
    def _m1_draw_floors(surface, cam_x, cam_y):
        """M1 floor tiles: warehouse tiles vs slum tiles."""
        from config import FLOOR_TILE_SIZE, FLOOR_TILE, SLUM_FLOOR_TILE
        start_tx = int(cam_x // FLOOR_TILE_SIZE) * FLOOR_TILE_SIZE
        start_ty = int(cam_y // FLOOR_TILE_SIZE) * FLOOR_TILE_SIZE
        warehouse_y_start = 11 * TILE_SIZE
        warehouse_y_end = 20 * TILE_SIZE
        for tx in range(
            start_tx - FLOOR_TILE_SIZE,
            int(cam_x + WIDTH + FLOOR_TILE_SIZE * 2),
            FLOOR_TILE_SIZE,
        ):
            for ty in range(
                start_ty - FLOOR_TILE_SIZE,
                int(cam_y + HEIGHT + FLOOR_TILE_SIZE * 2),
                FLOOR_TILE_SIZE,
            ):
                tile = (
                    FLOOR_TILE
                    if (warehouse_y_start <= ty <= warehouse_y_end)
                    else SLUM_FLOOR_TILE
                )
                surface.blit(tile, (tx - cam_x, ty - cam_y))

    @staticmethod
    def _m1_draw_walls(surface, cam_x, cam_y):
        """M1 walls: warehouse walls vs colored container tiles."""
        from config import WAREHOUSE_WALL
        container_tiles = []  # Will be populated from config
        try:
            from config import (CONTAINER_TILE_RED, CONTAINER_TILE_BLUE,
                                CONTAINER_TILE_RUST, CONTAINER_TILE_GREY, wall_shadow)
            container_tiles = [CONTAINER_TILE_RED, CONTAINER_TILE_BLUE,
                               CONTAINER_TILE_RUST, CONTAINER_TILE_GREY]
        except ImportError:
            pass
        for wall in STATE.walls:
            dr = pygame.Rect(wall.x - cam_x, wall.y - cam_y, TILE_SIZE, TILE_SIZE)
            try:
                surface.blit(wall_shadow, (dr.x - 4, dr.y - 4))
            except NameError:
                pass
            col = wall.x // TILE_SIZE
            row = wall.y // TILE_SIZE
            if (col, row) in getattr(STATE, 'warehouse_walls_set', set()):
                surface.blit(WAREHOUSE_WALL, (dr.x, dr.y))
            elif container_tiles:
                tile_idx = (col + row) % len(container_tiles)
                surface.blit(container_tiles[tile_idx], (dr.x, dr.y))

    @staticmethod
    def _m1_draw_breach_markers(surface, cam_x, cam_y):
        """M1 breach markers: pulsing orange circles on breachable walls."""
        for marker in STATE.breach_markers[:]:
            mx, my, timer = marker
            mx -= cam_x
            my -= cam_y
            if -50 < mx < WIDTH + 50 and -50 < my < HEIGHT + 50:
                alpha = min(255, timer * 2)
                for i in range(3):
                    radius = 8 + i * 4
                    s = get_glow(radius, (255, 80, 20), 255)
                    s.set_alpha(alpha // (i + 1))
                    surface.blit(s, (mx - radius, my - radius))
            marker[2] -= 1
            if marker[2] <= 0:
                STATE.breach_markers.remove(marker)

    @staticmethod
    def _m1_draw_desks(surface, cam_x, cam_y):
        """M1 desks with papers, notes, coffee mugs, and read indicators.
        NOTE: The full desk drawing is ~30 lines of detailed pygame.draw calls.
        This is a simplified placeholder — replace with the original code
        from mission.py when integrating."""
        for desk in STATE.desks:
            dx = desk["x"] - cam_x
            dy = desk["y"] - cam_y
            if -50 < dx < WIDTH + 50 and -50 < dy < HEIGHT + 50:

                # Read indicator
                dot_col = (100, 220, 100) if desk.get("read") else (240, 190, 50)
                pygame.draw.circle(surface, dot_col, (dx, dy - 22), 4)

    # ─────────────────────────────────────────
    # MISSION 3 POPULATION
    # ─────────────────────────────────────────

    def populate_m3(self):
        """Populate the pipeline with Mission 3 (The Vostral) draws."""
        cam_x, cam_y = self.cam_x, self.cam_y

        # M3 uses stronger vignette during interrogation
        try:
            from mission3 import STAGE_INTERROGATION
            stage = getattr(STATE, "mission3_stage", STAGE_INTERROGATION)
            self.vignette_strength = 180 if stage == STAGE_INTERROGATION else 130
        except (ImportError, AttributeError):
            self.vignette_strength = 180

        # Chair-break cutscene takes over the entire frame
        anim_t = getattr(STATE, "mission3_chair_break_anim", -1)
        if anim_t >= 0:
            try:
                from mission3 import _draw_chair_break_cutscene
                self.pipeline.submit('overlay', lambda s: _draw_chair_break_cutscene(anim_t))
                # Early return — nothing else draws
                self.vignette_strength = 0  # skip shared post-FX
                self.hud_drawer = None
                return
            except ImportError:
                pass

        # Background fill
        self.pipeline.submit('floor', lambda s: s.fill((18, 18, 22)))

        # Cell background tiles
        self.pipeline.submit(
            'floor',
            lambda s: self._m3_draw_cell_bg(s, cam_x, cam_y),
        )

        # M3 player drawn with no_gun=True
        if STATE.player:
            import death_anim
            if not death_anim.should_skip_player_draw():
                import game_core as gc
                last_keys = getattr(gc, '_last_keys', None)
                is_moving = (not STATE.player_dead and last_keys and (
                    last_keys[pygame.K_w] or last_keys[pygame.K_s] or
                    last_keys[pygame.K_a] or last_keys[pygame.K_d]
                ))
                self.pipeline.submit(
                    'entities',
                    lambda s: STATE.player.draw(
                        s, cam_x, cam_y,
                        walls=[],  # M3 uses its own wall view
                        no_gun=True,
                        is_moving=is_moving,
                    ),
                    sort_key=STATE.player.rect.centery,
                )

        # Bullets (shared)
        self.pipeline.submit(
            'projectiles',
            lambda s: draw_bullets(s, STATE.bullets, cam_x, cam_y, with_glow=True),
        )

        # Cell room overlay (atmosphere, monitor, Volkov, door, breach flash)
        self.pipeline.submit(
            'overlay',
            lambda s: self._m3_draw_cell_room(s, cam_x, cam_y),
        )

        # Shared overlays (post-FX, etc.) — M3 skips shared world since
        # there are no enemies/particles in the cell phase
        self._submit_shared_overlays()

        # Objective panel
        self.pipeline.submit(
            'hud',
            lambda s: self._m3_draw_objective(s),
        )

        # Dialogue panel (interrogation phase)
        try:
            from mission3 import STAGE_INTERROGATION
            if getattr(STATE, "mission3_stage", STAGE_INTERROGATION) == STAGE_INTERROGATION:
                self.pipeline.submit(
                    'hud',
                    lambda s: self._m3_draw_dialogue(s),
                )
        except ImportError:
            pass

        # Drain prompt
        self.pipeline.submit(
            'hud',
            lambda s: self._m3_draw_drain_prompt(s, cam_x, cam_y),
        )

        # Crafting overlay
        if getattr(STATE, "mission3_crafting_active", False):
            self.pipeline.submit(
                'hud',
                lambda s: self._m3_draw_crafting(s),
            )

        # Override HUD drawer — M3 handles its own HUD via pipeline submits
        self.hud_drawer = None

    # ─────────────────────────────────────────

    @staticmethod
    def _draw_blood_decals(surface, cam_x, cam_y):
        """Blood decals — shared by M1 and M2."""
        for decal in STATE.blood_decals[:]:
            surf = get_blood_decal(decal[3])
            surf.set_alpha(int(decal[2]))
            surface.blit(surf, (decal[0] - decal[3] - cam_x, decal[1] - decal[3] - cam_y))
            decal[2] -= 0.25
            if decal[2] <= 0:
                STATE.blood_decals.remove(decal)

    # ─────────────────────────────────────────
    # FLUSH
    # ─────────────────────────────────────────

    def flush(self):
        """Render all submitted draw calls and clear the pipeline.
        
        Uses pygame.display.get_surface() instead of a cached import
        so that display mode changes (windowed ↔ fullscreen) don't
        leave us rendering to a stale/destroyed surface.
        """
        surface = pygame.display.get_surface()
        self.pipeline.flush(surface)
