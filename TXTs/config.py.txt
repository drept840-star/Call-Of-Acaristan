# `screen` IS the display surface, always at 1000x700. We use pygame.SCALED
# which makes pygame handle the upscaling to the physical window size via
# the GPU — this is the fast path (the manual transform.scale approach we
# tried earlier was software-only and caused severe lag at high resolutions).
#
# With SCALED, the display surface stays at 1000x700 logical pixels no matter
# what physical size the window is. All game code draws to it using the
# hardcoded WIDTH x HEIGHT bounds. The OS window can be resized freely by
# the user (RESIZABLE flag) and pygame re-scales automatically.
#
# The Resolution setting in Settings → Display controls the FULLSCREEN
# physical resolution (passed to set_mode when mode=1). In windowed mode,
# the window starts at 1000x700 and the user can resize it by dragging.

def draw_sky_gradient(
    surface, top_color, bot_color, y_start=SCENE_TOP, y_end=SCENE_BOT
):
    for i in range(y_end - y_start):
        t = i / (y_end - y_start)
        r = int(top_color[0] + (bot_color[0] - top_color[0]) * t)
        g = int(top_color[1] + (bot_color[1] - top_color[1]) * t)
        b = int(top_color[2] + (bot_color[2] - top_color[2]) * t)
        pygame.draw.line(surface, (r, g, b), (0, y_start + i), (WIDTH, y_start + i))
def draw_building(surface, x, ground_y, w, h, base_col, win_col, win_lit_col, seed=0):
    rng = random.Random(seed)
    shadow = pygame.Surface((w + 8, h + 8), pygame.SRCALPHA)
    shadow.fill((0, 0, 0, 60))
    surface.blit(shadow, (x + 4, ground_y - h + 4))
    pygame.draw.rect(surface, base_col, (x, ground_y - h, w, h))
    for row in range(0, h, 18):
        pygame.draw.line(
            surface,
            (
                max(0, base_col[0] - 10),
                max(0, base_col[1] - 10),
                max(0, base_col[2] - 10),
            ),
            (x, ground_y - h + row),
            (x + w, ground_y - h + row),
            1,
        )
    for wy in range(ground_y - h + 12, ground_y - 10, 22):
        for wx in range(x + 8, x + w - 8, 16):
            lit = rng.random() > 0.45
            col = win_lit_col if lit else win_col
            pygame.draw.rect(surface, col, (wx, wy, 8, 12))
            if lit:
                glow = pygame.Surface((12, 16), pygame.SRCALPHA)
                glow.fill((*win_lit_col, 30))
                surface.blit(glow, (wx - 2, wy - 2))
    pygame.draw.rect(
        surface,
        (max(0, base_col[0] + 15), max(0, base_col[1] + 15), max(0, base_col[2] + 15)),
        (x, ground_y - h, w, 6),
    )

def _make_vignette(strength=130):
    vig = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA)
    cx, cy = WIDTH // 2, (SCENE_TOP + SCENE_BOT) // 2
    max_r = int(math.hypot(WIDTH // 2, SCENE_H // 2)) + 10
    step = 6
    for r in range(max_r, 0, -step):
        t = 1.0 - (r / max_r)
        alpha = int(strength * t * t)
        alpha = min(alpha, strength)
        rw = int(r * 2)
        rh = int(r * 1.15)
        rect = (cx - r, cy - rh // 2, rw, rh)
        pygame.draw.ellipse(vig, (0, 0, 0, alpha), rect)
    return vig

def draw_scanlines(surface):
    # Honor the Graphics → Scanlines toggle. Settings are pushed into the
    # shared SETTINGS singleton by acaristan.py; this is the single render-side
    # gate that actually turns the CRT overlay off when the user disables it.
    # Late import to avoid a config ↔ user_settings cycle at module load time.
    try:
        from user_settings import SETTINGS
        if not SETTINGS.scanlines:
            return
    except Exception:
        pass
    surface.blit(SCANLINES, (0, 0))

class Particle:
    def __init__(self, x, y, color):
        self.x, self.y = x, y
        angle = random.uniform(0, math.pi * 2)
        speed = random.uniform(1, 6)
        self.dx = math.cos(angle) * speed
        self.dy = math.sin(angle) * speed
        self.life = random.randint(15, 35)
        self.color = color
        self.size = random.randint(2, 4)
    def update(self):
        self.x += self.dx
        self.y += self.dy
        self.dx *= 0.9
        self.dy *= 0.9
        self.life -= 1
        self.size = max(1, int(self.life / 8))
    def draw(self, surface, offset_x, offset_y):
        if self.life > 0:
            pygame.draw.rect(
                surface,
                self.color,
                (self.x - offset_x, self.y - offset_y, self.size, self.size),
            )
class Casing:
    CASING_W, CASING_H = 8, 4
    LIFETIME = 180
    def __init__(self, x, y, eject_angle):
        self.x, self.y = x, y
        spread = random.uniform(-0.3, 0.3)
        speed = random.uniform(5, 9)
        self.dx = math.cos(eject_angle + spread) * speed
        self.dy = math.sin(eject_angle + spread) * speed
        self.rot = random.uniform(0, math.pi * 2)
        self.rot_speed = random.uniform(-0.4, 0.4)
        self.life = self.LIFETIME
        self.grounded = False
    def rect(self):
        return pygame.Rect(
            self.x - self.CASING_W // 2,
            self.y - self.CASING_H // 2,
            self.CASING_W,
            self.CASING_H,
        )
    def update(self, walls=()):
        if not self.grounded:
            self.rot += self.rot_speed
            new_x = self.x + self.dx
            new_rect = pygame.Rect(
                new_x - self.CASING_W // 2,
                self.y - self.CASING_H // 2,
                self.CASING_W,
                self.CASING_H,
            )
            for w in walls:
                if new_rect.colliderect(w):
                    if self.dx > 0:
                        new_rect.right = w.left
                    elif self.dx < 0:
                        new_rect.left = w.right
                    self.x = new_rect.centerx
                    self.dx *= -0.3
                    self.rot_speed *= 0.5
                    break
            else:
                self.x = new_x
            new_y = self.y + self.dy
            new_rect = pygame.Rect(
                self.x - self.CASING_W // 2,
                new_y - self.CASING_H // 2,
                self.CASING_W,
                self.CASING_H,
            )
            for w in walls:
                if new_rect.colliderect(w):
                    if self.dy > 0:
                        new_rect.bottom = w.top
                    elif self.dy < 0:
                        new_rect.top = w.bottom
                    self.y = new_rect.centery
                    self.dy *= -0.3
                    self.rot_speed *= 0.5
                    break
            else:
                self.y = new_y
            self.dx *= 0.85
            self.dy *= 0.85
            self.rot_speed *= 0.88
            if abs(self.dx) < 0.05 and abs(self.dy) < 0.05:
                self.dx = 0
                self.dy = 0
                self.grounded = True
        self.life -= 1
    def draw(self, surface, offset_x, offset_y):
        if self.life <= 0:
            return
        sx = self.x - offset_x
        sy = self.y - offset_y
        hw = self.CASING_W / 2
        hh = self.CASING_H / 2
        cos_a = math.cos(self.rot)
        sin_a = math.sin(self.rot)
        corners = []
        for lx, ly in ((-hw, -hh), (hw, -hh), (hw, hh), (-hw, hh)):
            rx = lx * cos_a - ly * sin_a
            ry = lx * sin_a + ly * cos_a
            corners.append((sx + rx, sy + ry))
        pygame.draw.polygon(surface, (190, 155, 70), corners)
        pygame.draw.polygon(surface, (140, 110, 40), corners, 1)

# Per-frame cache for get_scaled_mouse(). With pygame.SCALED, pygame already
# maps mouse coordinates to the logical 1000x700 render space for us, so
# get_scaled_mouse() is essentially just pygame.mouse.get_pos(). We keep the
# function for backward compatibility (it's called from ~20 sites) and cache
# the result per-frame for the hover-check hot path.
_scaled_mouse_cache = None
_scaled_mouse_raw = None


def get_scaled_mouse():
    """Return mouse position in the 1000x700 logical render space.

    Normally pygame.SCALED already returns logical coordinates. Some Windows /
    SDL paths lose that logical scaling after F11 -> windowed -> maximize; then
    the display surface becomes the physical window size while the game still
    draws a 1000x700 frame in the top-left. acaristan._present_frame() rescales
    that frame at presentation time, and this function mirrors the same
    letterbox math for mouse input.
    """
    global _scaled_mouse_cache, _scaled_mouse_raw
    raw_pos = pygame.mouse.get_pos()
    display = pygame.display.get_surface()
    display_size = display.get_size() if display is not None else (WIDTH, HEIGHT)
    cache_key = (raw_pos, display_size)
    if _scaled_mouse_raw == cache_key and _scaled_mouse_cache is not None:
        return _scaled_mouse_cache

    if display_size != (WIDTH, HEIGHT):
        dw, dh = display_size
        scale = min(dw / WIDTH, dh / HEIGHT) if WIDTH and HEIGHT else 1.0
        scaled_w = int(WIDTH * scale)
        scaled_h = int(HEIGHT * scale)
        ox = (dw - scaled_w) // 2
        oy = (dh - scaled_h) // 2
        if scale > 0:
            lx = int((raw_pos[0] - ox) / scale)
            ly = int((raw_pos[1] - oy) / scale)
        else:
            lx, ly = raw_pos
    else:
        lx, ly = raw_pos

    result = (
        max(0, min(WIDTH, lx)),
        max(0, min(HEIGHT, ly)),
    )
    _scaled_mouse_cache = result
    _scaled_mouse_raw = cache_key
    return result
