        return
    if idx % 2 != 0 and ch not in ".!?":
        return
    try:
        audio.play("type_tick", volume=volume)
    except Exception:
        pass

def update_walking_sim(keys):
    global walk_camera_x, walk_camera_y, walk_firefight_triggered, walk_firefight_timer
    global walk_dialogue_active, walk_dialogue_speaker, walk_dialogue_text
    global walk_dialogue_char_idx, walk_dialogue_timer, walk_complete
    global walk_active_hover, walk_after_firefight, walk_fp_cutscene, _walk_player_moving, walk_return_timer
    if walk_player is None:
        return walk_camera_x, walk_camera_y, False
    if walk_return_timer > 0:
        walk_return_timer -= 1
    if walk_fp_cutscene:
        _update_walk_fp_cutscene()
        return walk_camera_x, walk_camera_y, False
    movement_locked = (walk_firefight_triggered and not walk_after_firefight) or walk_return_timer > 0
    if not movement_locked:
        dx, dy = 0, 0
        spd = 3
        # Delta-time aware: spd is in px/frame, multiply by (60*dt).
        _dt_mul = 60.0 * STATE.dt
        if keys[pygame.K_w]:
            dy -= spd * _dt_mul
        if keys[pygame.K_s]:
            dy += spd * _dt_mul
        if keys[pygame.K_a]:
            dx -= spd * _dt_mul
        if keys[pygame.K_d]:
            dx += spd * _dt_mul
        if dx != 0 and dy != 0:
            dx *= 0.707
            dy *= 0.707
        is_moving = dx != 0 or dy != 0
        _walk_player_moving = is_moving
        walk_player.move(dx, dy, walk_walls)
        global _walk_footstep_timer
        if is_moving:
            _walk_footstep_timer -= 1
            if _walk_footstep_timer <= 0:
                audio.play("footstep_walk", volume=0.2)
                _walk_footstep_timer = 35 + random.randint(0, 10)
        else:
            _walk_footstep_timer = 0
        walk_camera_x += (walk_player.rect.centerx - WIDTH // 2 - walk_camera_x) * 0.1
        walk_camera_y += (walk_player.rect.centery - HEIGHT // 2 - walk_camera_y) * 0.1
    if walk_firefight_triggered and not walk_after_firefight:
        target_angle = 0.0
        cur = walk_player.angle
        diff = (target_angle - cur + math.pi) % (2 * math.pi) - math.pi
        walk_player.angle += diff * 0.08
    else:
        # Use get_scaled_mouse() so the player faces toward the cursor
        # correctly at non-default resolutions. Raw pygame.mouse.get_pos()
        # returns window pixels which don't match render-space coords
        # after a resolution change.
        mx, my = get_scaled_mouse()
        world_mx = mx + walk_camera_x
        world_my = my + walk_camera_y
        walk_player.angle = math.atan2(
            world_my - walk_player.rect.centery, world_mx - walk_player.rect.centerx
        )
    walk_active_hover = None
    for sx, sy, key, label, text in walk_story_points:
        if (
            math.hypot(walk_player.rect.centerx - sx, walk_player.rect.centery - sy)
            < 80
        ):
            walk_active_hover = (label, text)
            break
    if not walk_firefight_triggered:
        overlook_x = 38 * TILE_SIZE
        if walk_player.rect.centerx > overlook_x:
            walk_firefight_triggered = True
            STATE.walk_attack_active = True
            STATE.walk_attack_timer = -90        # pre-roll before anything happens
            STATE.walk_attack_events_done.clear()
            STATE.walk_boat_active = False
            STATE.walk_boat_x = -60.0
            STATE.walk_boat_y = SCENE_TOP + 165
            STATE.rig_status = ["ONLINE", "ONLINE", "ONLINE"]
            STATE.walk_missiles = []
            STATE.walk_explosions = []
            STATE.walk_terminal_active = False
            STATE.walk_terminal_status = ""
            walk_dialogue_active = False
            audio.play("radio_static", volume=0.25)
            audio.play("wind", volume=0.4, loops=-1)
    if STATE.walk_attack_active:
        STATE.walk_attack_timer += 1
        if STATE.walk_attack_timer >= 0:
            _update_walk_attack_timeline()
    if walk_dialogue_active and walk_dialogue_char_idx < len(walk_dialogue_text):
        walk_dialogue_timer += 1
        # First few characters type fast so it doesn't sit on "Thi"
        if walk_dialogue_char_idx < 8:
            threshold = 1
        else:
            threshold = 2
        if walk_dialogue_timer >= threshold:
            walk_dialogue_timer = 0
            next_ch = walk_dialogue_text[walk_dialogue_char_idx] if walk_dialogue_char_idx < len(walk_dialogue_text) else ""
            _play_walk_type_tick(next_ch, walk_dialogue_char_idx, volume=0.09)
            walk_dialogue_char_idx += 1
    if STATE.walk_dialogue_active and STATE.walk_dialogue_char_idx < len(STATE.walk_dialogue_text):
        STATE.walk_dialogue_timer += 1
        # Attack-cutscene dialogue should not crawl one letter at a time.
        state_dialogue_speed = 2 if STATE.walk_attack_active else 5
        if STATE.walk_dialogue_timer >= state_dialogue_speed:
            STATE.walk_dialogue_timer = 0
            next_ch = STATE.walk_dialogue_text[STATE.walk_dialogue_char_idx] if STATE.walk_dialogue_char_idx < len(STATE.walk_dialogue_text) else ""
            _play_walk_type_tick(next_ch, STATE.walk_dialogue_char_idx, volume=0.10 if STATE.walk_attack_active else 0.09)
            STATE.walk_dialogue_char_idx += 1
    if walk_after_firefight or STATE.walk_after_firefight:
        if walk_player.rect.centerx > walk_van_trigger_x:
            clear_walk_dialogue()
            walk_complete = True
    return walk_camera_x, walk_camera_y, walk_complete
def _ease_out_cubic(t):
    t = max(0.0, min(1.0, t))
    return 1 - (1 - t) ** 3
def _lerp(a, b, t):
    return a + (b - a) * t
def _start_walk_fp_cutscene(scene, reward, npc=None, stash=None):
    global walk_fp_cutscene, walk_dialogue_active
    durations = {
        "kid_kevlar": 520,
        "crate_ammo": 190,
        "crate_kevlar": 200,
    }
    _slow = {"kid_kevlar": 1.7}
    walk_dialogue_active = False
    speed = _slow.get(scene, 1.0)
    walk_fp_cutscene = {
        "scene": scene,
        "reward": reward,
        "npc": npc,
        "stash": stash,
        "timer": 0,
        "duration": durations.get(scene, 180),
        "speed": speed,
    }
    if scene.startswith("crate"):
        audio.play("door_creak", volume=0.45)
    else:
        audio.play("pickup", volume=0.25)
def _finish_walk_fp_cutscene():
    global walk_fp_cutscene
    global warehouse_hp_bonus, warehouse_mag_bonus, warehouse_intel_given
    global walk_dialogue_active, walk_dialogue_speaker, walk_dialogue_text
    global walk_dialogue_char_idx, walk_player
    if not walk_fp_cutscene:
        return
    scene = walk_fp_cutscene["scene"]
    reward = walk_fp_cutscene["reward"]
    npc = walk_fp_cutscene.get("npc")
    stash = walk_fp_cutscene.get("stash")
    walk_dialogue_active = True
    walk_dialogue_char_idx = 0
    if reward in ("kid_kevlar", "stash_kevlar"):
        warehouse_hp_bonus += 10
        if walk_player:
            walk_player.max_hp = 100 + warehouse_hp_bonus
            walk_player.hp = walk_player.max_hp
        walk_dialogue_speaker = npc["name"] if npc else "SYSTEM"
        if npc:
            walk_dialogue_text = (
                "The kid watches you strap the plate on.\n"
                "He gives a small nod, then disappears into the alley.\n"
                "Kevlar equipped. (+10 Max HP)"
            )
        else:
            walk_dialogue_text = (
                "You pry open the damaged container.\n"
                "Inside: a cracked but usable Kevlar plate.\n"
                "Kevlar salvaged. (+10 Max HP)"
            )
    elif reward == "stash_ammo":
        warehouse_mag_bonus += 1
        walk_dialogue_speaker = "SYSTEM"
        walk_dialogue_text = (
            "You open the KRAP ammo crate.\n"
            "Loose 9mm magazines sit under a torn supply manifest.\n"
            "Extra magazine acquired. (+1 starting mag)"
        )
    audio.play("pickup", volume=0.55)
    walk_fp_cutscene = None
def _update_walk_fp_cutscene():
    global walk_fp_cutscene
    if not walk_fp_cutscene:
        return
    walk_fp_cutscene["timer"] += 1.0 / walk_fp_cutscene.get("speed", 1.0)
    if walk_fp_cutscene["timer"] >= walk_fp_cutscene["duration"]:
        _finish_walk_fp_cutscene()

def _draw_fp_kevlar(surface, x, y, scale=1.0):
    s = scale
    w = int(118 * s)
    h = int(92 * s)
    vest = pygame.Surface((w, h), pygame.SRCALPHA)
    pts = [
        (int(w * 0.18), 0),
        (int(w * 0.82), 0),
        (w - 4, int(h * 0.28)),
        (w - 8, h - 6),
        (int(w * 0.68), h),
        (int(w * 0.32), h),
        (8, h - 6),
        (4, int(h * 0.28)),
    ]
    pygame.draw.polygon(vest, (64, 76, 58), pts)
    pygame.draw.polygon(vest, (35, 45, 34), pts, 2)
    # Collar
    pygame.draw.rect(vest, (42, 50, 38), (int(w * 0.32), 2, int(w * 0.36), int(h * 0.18)), border_radius=3)
    # Center seam
    pygame.draw.line(vest, (34, 42, 30), (w // 2, 8), (w // 2, h - 8), 2)
    # Pouches
    for px in (int(w * 0.18), int(w * 0.52)):
        pygame.draw.rect(vest, (48, 58, 42), (px, int(h * 0.38), int(w * 0.28), int(h * 0.34)), border_radius=3)
        pygame.draw.rect(vest, (30, 38, 28), (px, int(h * 0.38), int(w * 0.28), int(h * 0.34)), 1, border_radius=3)
        pygame.draw.line(vest, (72, 84, 60), (px + 4, int(h * 0.48)), (px + int(w * 0.28) - 4, int(h * 0.48)), 1)
    # Scratches / wear
    pygame.draw.line(vest, (105, 110, 85), (int(w * 0.18), int(h * 0.2)), (int(w * 0.38), int(h * 0.28)), 1)
    pygame.draw.line(vest, (25, 30, 22), (int(w * 0.62), int(h * 0.74)), (int(w * 0.78), int(h * 0.7)), 1)
    surface.blit(vest, (int(x - w // 2), int(y - h // 2)))

def _draw_fp_crate(surface, cx, cy, open_t, contents="ammo"):
    cw, ch = 260, 130
    # Crate shadow
    shadow = pygame.Surface((cw + 40, ch + 40), pygame.SRCALPHA)
    pygame.draw.ellipse(shadow, (0, 0, 0, 80), (0, 16, cw + 40, ch + 18))
    surface.blit(shadow, (cx - cw // 2 - 20, cy - ch // 2 + 45))
    # Crate body
    pygame.draw.rect(surface, (92, 62, 35), (cx - cw // 2, cy - ch // 2, cw, ch), border_radius=5)
    pygame.draw.rect(surface, (52, 34, 20), (cx - cw // 2, cy - ch // 2, cw, ch), 3, border_radius=5)
    # Wood planks
    for i in range(4):
        py = cy - ch // 2 + 20 + i * 26
        pygame.draw.line(surface, (67, 43, 24), (cx - cw // 2 + 8, py), (cx + cw // 2 - 8, py), 2)
    # Metal bands
    pygame.draw.rect(surface, (56, 58, 58), (cx - cw // 2 + 28, cy - ch // 2, 12, ch))
    pygame.draw.rect(surface, (56, 58, 58), (cx + cw // 2 - 40, cy - ch // 2, 12, ch))
    # Interior glow once opening
    if open_t > 0.25:
        glow_alpha = int(120 * open_t)
        glow = pygame.Surface((cw - 30, ch - 20), pygame.SRCALPHA)
        pygame.draw.ellipse(glow, (255, 175, 50, glow_alpha), glow.get_rect())
        surface.blit(glow, (cx - cw // 2 + 15, cy - ch // 2 + 12))
        pygame.draw.rect(surface, (25, 20, 14), (cx - cw // 2 + 18, cy - ch // 2 + 18, cw - 36, ch - 30), border_radius=3)
        if contents == "ammo":
            _draw_fp_mag(surface, cx - 35, cy + 5, rot=-8)
            _draw_fp_mag(surface, cx + 5, cy + 3, rot=5)
            _draw_fp_mag(surface, cx + 42, cy + 8, rot=-3)
        else:
            _draw_fp_kevlar(surface, cx, cy + 10, scale=0.75)
    # Lid opening upward
    lid_h = 42
    lid_y = cy - ch // 2 - int(open_t * 95)
    lid_squash = max(10, int(lid_h * (1.0 - open_t * 0.65)))
    pygame.draw.rect(surface, (110, 75, 42), (cx - cw // 2, lid_y, cw, lid_squash), border_radius=4)
    pygame.draw.rect(surface, (55, 36, 22), (cx - cw // 2, lid_y, cw, lid_squash), 2, border_radius=4)
    pygame.draw.line(surface, (70, 45, 25), (cx - cw // 2 + 10, lid_y + lid_squash // 2), (cx + cw // 2 - 10, lid_y + lid_squash // 2), 2)

    scene = walk_fp_cutscene["scene"]

    # Dark cinematic focus overlay

def handle_walk_interact():
    global \
        walk_dialogue_active, \
        walk_dialogue_speaker, \
        walk_dialogue_text, \
        walk_dialogue_char_idx
    global \
        player_rations, \
        warehouse_hp_bonus, \
        warehouse_mag_bonus, \
        walk_npcs, \
        walk_stashes, \
        walk_fp_cutscene
    if walk_fp_cutscene:
        return
    if walk_dialogue_active:
        return
    for npc in walk_npcs:
        dist = math.hypot(
            walk_player.rect.centerx - npc["x"], walk_player.rect.centery - npc["y"]
        )
        if dist < 80:
            walk_dialogue_active = True
            walk_dialogue_speaker = npc["name"]
            walk_dialogue_char_idx = 0
            if npc["has_given"]:
                walk_dialogue_text = npc["done_dialogue"]
            else:
                if player_rations > 0:
                    npc["has_given"] = True
                    player_rations -= 1
                    if npc["reward"] == "kevlar":
                        _start_walk_fp_cutscene("kid_kevlar", "kid_kevlar", npc=npc)
                        return
                    else:
                        warehouse_intel_given = True
                        audio.play("pickup", volume=0.5)
                        walk_dialogue_text = (
                            npc["thanks"][0]
                            + "\n"
                            + npc["thanks"][1]
                            + "\n"
                            + npc["thanks"][2]
                            + "\n"
                            + npc["thanks"][3]
                        )
                else:
                    walk_dialogue_text = npc["dialogue"][0] + "\n" + npc["dialogue"][1]
            return
    for stash in walk_stashes:
        dist = math.hypot(
            walk_player.rect.centerx - stash["x"], walk_player.rect.centery - stash["y"]
        )
        if dist < 80:
            if stash["searched"]:
                walk_dialogue_active = True
                walk_dialogue_speaker = "SYSTEM"
                walk_dialogue_text = "Already searched."
                walk_dialogue_char_idx = 0
            else:
                stash["searched"] = True
                if stash["type"] == "ammo":
                    _start_walk_fp_cutscene("crate_ammo", "stash_ammo", stash=stash)
                    return
                elif stash["type"] == "kevlar":
                    _start_walk_fp_cutscene("crate_kevlar", "stash_kevlar", stash=stash)
                    return
            return

def _gen_walk_props():
    rng = random.Random(31)
    props = []
    map_w = len(walk_map[0]) * TILE_SIZE
    map_h = len(walk_map) * TILE_SIZE
    def is_walkable(wx, wy):
        col = wx // TILE_SIZE
        row = wy // TILE_SIZE
        if 0 <= row < len(walk_map) and 0 <= col < len(walk_map[row]):
            return walk_map[row][col] not in ("C", "W")
        return False
    lamps = [
        (3 * TILE_SIZE + 32, 4 * TILE_SIZE + 32),
        (9 * TILE_SIZE + 32, 7 * TILE_SIZE + 16),
        (16 * TILE_SIZE, 4 * TILE_SIZE + 48),
        (22 * TILE_SIZE + 16, 8 * TILE_SIZE + 16),
        (30 * TILE_SIZE + 16, 4 * TILE_SIZE + 32),
        (36 * TILE_SIZE, 7 * TILE_SIZE + 16),
    ]
    for lx, ly in lamps:
        props.append(("lamp", lx, ly))
    for _ in range(50):
        for _try in range(10):
            wx = rng.randint(2 * TILE_SIZE, map_w - 3 * TILE_SIZE)
            wy = rng.randint(2 * TILE_SIZE, map_h - 2 * TILE_SIZE)
            if is_walkable(wx, wy):
                kind = rng.choice(
                    [
                        "drum",
                        "drum",
                        "trash",
                        "trash",
                        "pallet",
                        "puddle",
                        "puddle",
                        "puddle",
                        "pipe",
                        "cable",
                        "debris",
                        "debris",
                    ]
                )
                props.append((kind, wx, wy))
                break
    for _ in range(25):
        for _try in range(10):
            col = rng.randint(2, len(walk_map[0]) - 3)
            row = rng.randint(2, len(walk_map) - 3)
            tile = walk_map[row][col]
            if tile == ".":
                if row > 0 and walk_map[row - 1][col] == "C":
                    wx = col * TILE_SIZE + rng.randint(8, TILE_SIZE - 8)
                    wy = row * TILE_SIZE + 4
                    kind = rng.choice(
                        ["wall_graffiti", "wall_poster", "wall_pipe", "wall_stain"]
                    )
                    props.append((kind, wx, wy))
                    break
    return props

# ─── Attack sequence timing (frames @ 60fps) ─────────────────────────
# Three-act cinematic:
#   ACT 1 — Setup     ( 0 – 360 / ~6 s)
#   ACT 2 — The Strike (400 – 980 / ~10 s)
#   ACT 3 — Aftermath (1100 – 1700 / ~10 s)
ATK_BOAT_APPEAR      = 90    # Boat enters frame from the left
ATK_FISHERMAN_1     = 135   # early warning before the terminal feed boots
ATK_TERMINAL_ON      = 260   # KRAP observation feed boots up
ATK_FISHERMAN_2     = 360   # warning just before the first missile run
ATK_TARGET_CENTER    = 400
ATK_MISSILE_1_LAUNCH = 460
ATK_IMPACT_CENTER    = 540
ATK_TARGET_LEFT      = 620
ATK_MISSILE_2_LAUNCH = 680
ATK_IMPACT_LEFT      = 760
ATK_TARGET_RIGHT     = 840
ATK_MISSILE_3_LAUNCH = 900
ATK_IMPACT_RIGHT     = 980
ATK_FISHERMAN_3     = 1220  # aftermath line after the last rig has burned for a beat
ATK_FISHERMAN_4     = 1600  # final fisherman line
ATK_END             = 2180  # extra breath before handoff to avoid an abrupt return
# ── Canonical world positions (used by BOTH update and draw) ──
HORIZON_Y_OVERLOOK  = SCENE_TOP + 180
BOAT_BASE_Y         = SCENE_TOP + 165
RIG_X_POSITIONS     = (225, 505, 760)   # LEFT, CENTER, RIGHT
RIG_NAMES           = ("LEFT RIG", "CENTER RIG", "RIGHT RIG")
RIG_IMPACT_FRAMES   = (ATK_IMPACT_LEFT, ATK_IMPACT_CENTER, ATK_IMPACT_RIGHT)
def _update_walk_attack_timeline():
    """Drives the attack cutscene: boat, missiles, rig destruction, dialogue.
    All side-effects (sounds, screen shake, dialogue, STATE mutations) happen
    here. The draw function only reads STATE — so the HUD and the visuals
    can never disagree with what actually happened.
    """
    if not STATE.walk_attack_active:
        return
    t = STATE.walk_attack_timer
    done = STATE.walk_attack_events_done
    # ── ACT 1: Setup ─────────────────────────────────────────────
    if t >= ATK_BOAT_APPEAR and "boat" not in done:
        done.add("boat")
        STATE.walk_boat_active = True
        STATE.walk_boat_x = 0.0
        STATE.walk_boat_y = BOAT_BASE_Y
        STATE.walk_terminal_status = ""
    if t >= ATK_FISHERMAN_1 and "fish1" not in done:
        done.add("fish1")
        _trigger_walk_dialogue(
            "OLD FISHERMAN",
            "Hear that engine?\nToo fast for a fishing boat.",
        )
    if t >= ATK_TERMINAL_ON and "terminal" not in done:
        done.add("terminal")
        STATE.walk_terminal_active = True
        STATE.walk_terminal_status = "CONTACT: UNKNOWN VESSEL — TRACKING"
        audio.play("menu_click", volume=0.5)
        audio.play("radio_static", volume=0.3)
    if t >= ATK_FISHERMAN_2 and "fish2" not in done:
        done.add("fish2")
        _trigger_walk_dialogue(
            "OLD FISHERMAN",
            "They come in low during the rain.\n"
            "The rigs never see them until the first rocket.",
        )
    # ── ACT 2: The strike (CENTER → LEFT → RIGHT) ───────────────
    _process_rig_strike(t, done, "center", 1,
        ATK_TARGET_CENTER, ATK_MISSILE_1_LAUNCH, ATK_IMPACT_CENTER)
    _process_rig_strike(t, done, "left", 0,
        ATK_TARGET_LEFT,   ATK_MISSILE_2_LAUNCH, ATK_IMPACT_LEFT)
    _process_rig_strike(t, done, "right", 2,
        ATK_TARGET_RIGHT,  ATK_MISSILE_3_LAUNCH, ATK_IMPACT_RIGHT)
    # ── ACT 3: Aftermath ────────────────────────────────────────
    if t >= ATK_FISHERMAN_3 and "fish3" not in done:
        done.add("fish3")
        STATE.walk_terminal_status = "ALL RIGS OFFLINE — OIL FIELD LOST"
        _trigger_walk_dialogue(
            "OLD FISHERMAN",
            "Third time this month.\n"
            "KRAP watches until the oil stops moving.",
        )
    if t >= ATK_FISHERMAN_4 and "fish4" not in done:
        done.add("fish4")
        _trigger_walk_dialogue(
            "OLD FISHERMAN",
            "You're new.\nYou'll learn.",
        )
    # Extra human reactions; timed right after visible events, not before them.
    if t >= ATK_IMPACT_CENTER + 34 and "react_center" not in done:
        done.add("react_center")
        _trigger_walk_dialogue(
            "OLD FISHERMAN",
            "There. Center rig's folding.",
        )
    if t >= ATK_IMPACT_LEFT + 35 and "react_left" not in done:
        done.add("react_left")
        _trigger_walk_dialogue(
            "OLD FISHERMAN",
            "Left rig's falling into the sea.",
        )
    if t >= ATK_IMPACT_RIGHT + 50 and "react_right" not in done:
        done.add("react_right")
        _trigger_walk_dialogue(
            "OLD FISHERMAN",
            "Now the last one burns.\nThree fires on the horizon.",
        )
    # Delayed sound layers: launch -> impact -> collapse -> steam/fire spread.
    for _idx, _impact in enumerate(RIG_IMPACT_FRAMES):
        if t >= _impact + 36 and f"collapse_sound_{_idx}" not in done:
            done.add(f"collapse_sound_{_idx}")
            STATE.walk_terminal_status = f"STRUCTURAL COLLAPSE — {RIG_NAMES[_idx]}"
            audio.play("rig_metal_collapse", volume=0.24)
            audio.schedule(10, "terminal_glitch", volume=0.14)
            STATE.screen_shake = max(STATE.screen_shake, 22 - _idx * 2)
        if t >= _impact + 68 and f"water_sound_{_idx}" not in done:
            done.add(f"water_sound_{_idx}")
            STATE.walk_terminal_status = f"FIRE SPREAD / STEAM PLUME — {RIG_NAMES[_idx]}"
            audio.play("rig_water_splash", volume=0.22)
            audio.schedule(8, "rig_fire_burst", volume=0.16)
            audio.schedule(4, "terminal_glitch", volume=0.12)
    # ── Boat runs left-to-right, then accelerates out of frame ──
    if STATE.walk_boat_active:
        if t < ATK_TARGET_CENTER:
            boat_speed = 0.32
        elif t < ATK_IMPACT_RIGHT + 120:
            boat_speed = 0.60
        elif t < ATK_FISHERMAN_4:
            boat_speed = 0.78
        else:
            boat_speed = 1.55
        STATE.walk_boat_x += boat_speed
    # ── Advance missile and explosion lifetimes ─────────────────
    for m in STATE.walk_missiles[:]:
        m["t"] = min(1.0, m["t"] + m["speed"])
        if m["t"] >= 1.0:
            STATE.walk_missiles.remove(m)
    for ex in STATE.walk_explosions[:]:
        ex["life"] -= 1
        if ex["life"] <= 0:
            STATE.walk_explosions.remove(ex)
    # ── End cutscene ────────────────────────────────────────────
    if t >= ATK_END and "end" not in done:
        done.add("end")
        STATE.walk_terminal_status = "SIGNAL LOST — RETURNING TO HARBOR"
        clear_walk_dialogue()
        STATE.walk_attack_active = False
        STATE.walk_terminal_active = False
        global walk_after_firefight, walk_return_timer
        walk_after_firefight = True
        walk_return_timer = 240
        STATE.walk_after_firefight = True
def _process_rig_strike(t, done, prefix, rig_index,
                        target_frame, launch_frame, impact_frame):
    """One full TARGETED → LAUNCH → DESTROYED sequence for a single rig.
    Uses RIG_X_POSITIONS as the single source of truth so the missile
    physically aims at the rig you can actually see.
    """
    rig_x    = RIG_X_POSITIONS[rig_index]
    rig_name = RIG_NAMES[rig_index]
    # TARGETED
    if t >= target_frame and f"{prefix}_target" not in done:
        done.add(f"{prefix}_target")
        STATE.rig_status[rig_index] = "TARGETED"
        STATE.walk_terminal_status = f"TARGET LOCK: {rig_name}"
        audio.play("menu_hover", volume=0.45)
    # LAUNCH — missile fires from current boat position
    if t >= launch_frame and f"{prefix}_launch" not in done:
        done.add(f"{prefix}_launch")
        STATE.walk_missiles.append({
            "start_x": float(STATE.walk_boat_x + 50),
            "start_y": float(STATE.walk_boat_y - 2),
            "tx":      float(rig_x),
            "ty":      float(HORIZON_Y_OVERLOOK - 70),
            "t":       0.0,
            "speed":   1.0 / max(1, impact_frame - launch_frame),
            "rig_index": rig_index,
        })
        STATE.walk_terminal_status = f"MISSILE INBOUND — {rig_name}"
        audio.play("missile_launch", volume=0.42)
        audio.schedule(8, "missile_flyby", volume=0.30)
        audio.schedule(2, "terminal_glitch", volume=0.16)
    # IMPACT
    if t >= impact_frame and f"{prefix}_impact" not in done:
        done.add(f"{prefix}_impact")
        STATE.rig_status[rig_index] = "DESTROYED"
        _life = 190 if rig_index == 2 else 155
        STATE.walk_explosions.append({
            "x":        float(rig_x),
            "y":        float(HORIZON_Y_OVERLOOK - 60),
            "life":     _life,
            "max_life": _life,
            "rig_index": rig_index,
        })
        STATE.walk_terminal_status = f"*** IMPACT — {rig_name} DESTROYED ***"
        audio.duck_ambience(frames=150, factor=0.25)
        audio.play("rig_impact_crack", volume=0.62)
        audio.schedule(4, "rig_low_boom", volume=0.82)
        audio.schedule(13, "rig_secondary_explosion", volume=0.42)

def _draw_attack_explosion(surface, cx, cy, t, scale=1.0):
    """Big cinematic oil-rig impact: flash, shockwave, fireball, smoke, debris and spray."""
    if t < 0:
        return
    s = max(0.35, float(scale))
    t = float(t)
    early = max(0.0, 1.0 - t / 22.0)
    burn = max(0.0, 1.0 - t / 135.0)
    grow = min(1.0, t / 50.0)
    # Sudden hot impact flash and expanding blast ring.
    if t < 30:
        flash_w = int(300 * s)
        flash_h = int(190 * s)
        flash = pygame.Surface((flash_w, flash_h), pygame.SRCALPHA)
        fw, fh = flash.get_size()
        center = (fw // 2, int(fh * 0.58))
        for rad, alpha, col in [
            (int((28 + t * 2.2) * s), int(235 * early), (255, 248, 212)),
            (int((58 + t * 3.0) * s), int(150 * early), (255, 178, 72)),
            (int((96 + t * 4.5) * s), int(70 * early), (255, 70, 24)),
        ]:
            if rad > 2 and alpha > 0:
                pygame.draw.circle(flash, (*col, alpha), center, rad)
        ring_rect = pygame.Rect(0, 0, int((78 + t * 7.0) * s), int((18 + t * 1.8) * s))
        ring_rect.center = (center[0], center[1] + int(22 * s))
        ring_alpha = int(170 * max(0.0, 1.0 - t / 30.0))
        pygame.draw.ellipse(flash, (255, 214, 135, ring_alpha), ring_rect, max(1, int(2 * s)))
        surface.blit(flash, (int(cx - fw // 2), int(cy - fh * 0.70)))
    # Main fireball / black smoke column.
    plume_w = max(80, int((190 + grow * 90) * s))
    plume_h = max(120, int((210 + grow * 110) * s))
    plume = pygame.Surface((plume_w, plume_h), pygame.SRCALPHA)
    pw, ph = plume.get_size()
    base_x = pw // 2
    base_y = ph - int(48 * s)
    # Core roiling fireballs. Deterministic per frame so it flickers without teleporting wildly.
    for i in range(24):
        frac = i / 23.0
        phase = t * (0.052 + frac * 0.018) + i * 1.73
        column_spread = (13 + frac * 44 + grow * 18) * s
        ox = math.sin(phase) * column_spread + math.sin(phase * 0.41) * 8 * s
        oy = (frac * (118 + grow * 85) + math.sin(phase * 0.8) * 8) * s
        rr = max(4, int((21 - frac * 8 + early * 9 + math.sin(phase) * 2.0) * s))
        heat = max(0.0, 1.0 - frac * 0.88)
        alpha = int((215 * burn + 55 * early) * (1.0 - frac * 0.35))
        if alpha <= 0:
            continue
        col = (
            255,
            int(92 + 140 * heat),
            int(24 + 94 * heat),
            max(0, min(255, alpha)),
        )
        pos = (int(base_x + ox), int(base_y - oy))
        pygame.draw.circle(plume, col, pos, rr)
        if i < 9 and t < 58:
            pygame.draw.circle(plume, (255, 245, 198, int(125 * max(0.0, 1.0 - t / 70.0))), pos, max(2, rr // 2))
    # Secondary fuel pops, offset from the main blast.
    for pop_i, pop_t in enumerate((13, 27, 44, 67)):
        dt = t - pop_t
        if 0 <= dt < 20:
            pop_alpha = int(180 * (1.0 - dt / 20.0))
            off = (-42, 36, -18, 50)[pop_i] * s
            py = base_y - (26 + pop_i * 18 + dt * 1.8) * s
            for r, col in [(22 + dt * 1.5, (255, 230, 175)), (38 + dt * 2.2, (255, 105, 36))]:
                pygame.draw.circle(plume, (*col, max(0, pop_alpha // (1 if r < 30 else 2))), (int(base_x + off), int(py)), max(3, int(r * s)))
    # Thick rolling smoke, rising above the fire.
    if t > 8:
        smoke_age = min(1.0, (t - 8) / 100.0)
        for i in range(22):
            frac = i / 21.0
            phase = i * 1.21 + t * 0.027
            sx = base_x + int(math.sin(phase) * (18 + frac * 62) * s + math.sin(phase * 0.33) * 18 * s)
            sy = base_y - int((58 + frac * (110 + smoke_age * 125) + smoke_age * 38) * s)
            sr = max(8, int((18 + frac * 24 + smoke_age * 15) * s))
            shade = int(36 + frac * 22)
            sa = max(0, int((120 - frac * 42) * burn + 28 * smoke_age))
            pygame.draw.circle(plume, (shade, shade - 4, shade - 2, min(150, sa)), (sx, sy), sr)
    # Sparks and debris arcing out from the impact.
    # Upgraded: more particles (34 → 50), longer ember trails, flicker
    # modulation on hot debris, and lingering spark clusters after the
    # main blast fades.
    rng = random.Random(int(cx * 13 + cy * 7))
    for i in range(50):  # was 34 — more debris for richer blast
        seed_ang = -math.pi * 0.94 + (math.pi * 1.88) * (i / 49.0)
        ang = seed_ang + math.sin(i * 2.31) * 0.24
        speed = (2.8 + rng.random() * 5.2) * s
        age = t - rng.randint(0, 10)
        if age < 0 or age > 90:  # was 78 — debris lives longer
            continue
        dx = math.cos(ang) * speed * age
        dy = math.sin(ang) * speed * age + 0.055 * age * age * s
        px = base_x + dx
        py = base_y + dy - 20 * s
        if 0 <= px < pw and 0 <= py < ph:
            hot = max(0, 1.0 - age / 60.0)  # was /55 — hot longer
            # ── Flicker modulation: brightness varies per frame ──
            # Each particle flickers independently based on its index + time,
            # so the fireball reads as alive/flame-like, not static.
            flicker = 0.75 + 0.25 * math.sin(t * 0.8 + i * 1.7)
            if i % 3:
                col = (255, int((118 + 82 * hot) * flicker), 48, int(190 * hot * flicker))
            else:
                col = (78, 80, 84, int(170 * max(0, 1 - age / 90)))
            pygame.draw.circle(plume, col, (int(px), int(py)), max(1, int((2.6 if i % 3 else 3.6) * s * max(0.55, hot))))
            if hot > 0.25:
                # ── Longer ember trail (was 7px, now 12px) ──
                trail_len = 12 * s
                tx = int(px - math.cos(ang) * trail_len)
                ty = int(py - math.sin(ang) * trail_len)
                pygame.draw.line(plume, (255, 150, 56, int(110 * hot)), (int(px), int(py)), (tx, ty), max(1, int(1.5 * s)))
                # ── Bright spark tip (hottest point of the trail) ──
                pygame.draw.circle(plume, (255, 230, 150, int(160 * hot * flicker)), (int(px), int(py)), max(1, int(1.2 * s)))
    # ── Lingering spark clusters (upgraded) ──
    # After the main fireball fades, small ember clusters linger and
    # flicker around the blast site — reads as residual heat / burning
    # debris that hasn't gone out yet.
    if t > 30:
        spark_fade = max(0, 1.0 - (t - 30) / 120.0)
        rng_sparks = random.Random(int(cx * 17 + cy * 11 + int(t / 8)))
        for i in range(15):
            sa = rng_sparks.uniform(0, math.tau)
            sr = rng_sparks.uniform(15, 50) * s
            sx = base_x + math.cos(sa) * sr
            sy = base_y - math.sin(sa) * sr * 0.6 - 10 * s
            if 0 <= sx < pw and 0 <= sy < ph:
                flicker = 0.5 + 0.5 * math.sin(t * 1.2 + i * 2.1)
                alpha = int(120 * spark_fade * flicker)
                if alpha > 0:
                    pygame.draw.circle(plume, (255, 180, 60, alpha), (int(sx), int(sy)), max(1, int(1.5 * s)))
    surface.blit(plume, (int(cx - pw // 2), int(cy - ph + 44 * s)))
    # Water impact: white-hot steam, vertical spray, oily orange reflection.
    if t < 70:
        splash_w = int(260 * s)
        splash_h = int(88 * s)
        splash = pygame.Surface((splash_w, splash_h), pygame.SRCALPHA)
        sw, sh = splash.get_size()
        fade = max(0.0, 1.0 - t / 70.0)
        for i in range(5):
            rect = pygame.Rect(0, 0, int((70 + t * (4.2 + i * 0.8)) * s), int((12 + t * 0.45 + i * 2) * s))
            rect.center = (sw // 2, sh - int((22 - i * 2) * s))
            pygame.draw.ellipse(splash, (190, 220, 240, int((120 - i * 16) * fade)), rect, 1)
        for i in range(20):
            frac = i / 19.0
            side = -1 if i < 10 else 1
            local = frac if side > 0 else 1.0 - frac
            px = sw // 2 + side * int((14 + local * 88 + t * 0.55) * s)
            py = sh - int((22 + math.sin(local * math.pi) * (28 + t * 0.28)) * s)
            pygame.draw.circle(splash, (205, 226, 238, int(145 * fade)), (px, py), max(1, int((3 - local * 1.4) * s)))
        for i in range(8):
            alpha = int((80 - i * 7) * fade)
            pygame.draw.line(splash, (255, 122, 42, alpha), (sw // 2 - int((34 - i * 2) * s), sh - int((18 - i) * s)), (sw // 2 + int((34 - i * 2) * s), sh - int((18 - i) * s)), 1)
        surface.blit(splash, (int(cx - sw // 2), int(cy + 4 * s)))

def _draw_attack_water(surface, elapsed, horizon_y):
    """Perspective-biased ocean with converging guide-lines, layered chop,
    animated ripples, and foam highlights.

    Upgraded (Task: ocean improvement):
      - Animated concentric ripple rings that expand from random points
        (reads as raindrops / wavelets on the surface).
      - Foam highlight specks on wave crests (white-tipped waves).
      - Deeper color gradient (more depth variation).
      - More wave bands for richer texture.
    """
    w, h = surface.get_size()
    # ── Depth gradient (darker near horizon, lighter near foreground) ──
    for y in range(horizon_y, h):
        p = (y - horizon_y) / max(1, h - horizon_y)
        r = int(8 + p * 14)
        g = int(18 + p * 20)
        b = int(30 + p * 28)
        pygame.draw.line(surface, (r, g, b), (0, y), (w, y))
    # ── Perspective grid lines (converging toward vanishing point) ──
    grid = pygame.Surface((w, h), pygame.SRCALPHA)
    for lane in range(-8, 9):
        x0 = ATTACK_VANISH_X + lane * 30
        x1 = ATTACK_VANISH_X + lane * 112
        pygame.draw.line(grid, (28, 46, 72, 22), (x0, horizon_y + 6), (x1, h - 34), 1)
    for band in range(15):
        frac = band / 14.0
        y = horizon_y + int((frac ** 1.86) * (h - horizon_y - 34))
        alpha = int(58 * (1.0 - frac * 0.55))
        pygame.draw.line(grid, (56, 84, 122, alpha), (0, y), (w, y), 1)
    surface.blit(grid, (0, 0))
    # ── Wave bands (animated horizontal chop) ──
    wave_layer = pygame.Surface((w, h), pygame.SRCALPHA)
    for band in range(16):  # was 12 — more bands for richer texture
        frac = band / 15.0
        y = horizon_y + 12 + int((frac ** 1.75) * (h - horizon_y - 90))
        seg_w = 14 + int(frac * 28)
        alpha = int(60 - frac * 22)
        step = 34 + int(frac * 28)
        for wx in range(-50, w + 50, step):
            wave_y = y + math.sin(elapsed * 0.028 + wx * 0.03 + band) * (1.2 + frac * 3.2)
            pygame.draw.line(
                wave_layer,
                (64, 98, 136, alpha),
                (wx, int(wave_y)),
                (wx + seg_w, int(wave_y + 1)),
                1,
            )
            # ── Foam highlight on wave crests (white-tipped waves) ──
            # Every 3rd wave segment gets a small white speck on the crest
            # for a "whitecap" read. Only in the foreground (frac > 0.3).
            if frac > 0.3 and (wx + band) % 3 == 0:
                foam_alpha = int(40 * frac)
                pygame.draw.circle(
                    wave_layer,
                    (180, 200, 215, foam_alpha),
                    (wx + seg_w // 2, int(wave_y - 1)),
                    max(1, int(frac * 1.5)),
                )
    surface.blit(wave_layer, (0, 0))
    # ── Animated ripple rings (raindrops / wavelets spreading outward) ──
    # Deterministic positions that shift slowly over time so the ripples
    # appear at new locations periodically. Each ripple expands and fades.
    ripple_layer = pygame.Surface((w, h), pygame.SRCALPHA)
    rng_ripples = random.Random(99)
    for i in range(6):
        # Base position for this ripple source (drifts slightly over time)
        base_x = rng_ripples.randint(50, w - 50)
        base_y = horizon_y + rng_ripples.randint(40, h - horizon_y - 60)
        # Each ripple source emits a new ring every ~60 frames
        for ring in range(3):
            phase = (elapsed + i * 23 + ring * 20) % 60
            radius = int(phase * 1.2)
            alpha = max(0, int(50 * (1.0 - phase / 60.0)))
            if alpha > 0 and radius > 2:
                pygame.draw.circle(
                    ripple_layer,
                    (90, 130, 170, alpha),
                    (base_x, base_y),
                    radius,
                    1,
                )
    surface.blit(ripple_layer, (0, 0))


def _draw_attack_water_fire_reflection(surface, rig_x, water_y, scale, elapsed, heat=1.0):
    """Draw a fire-glow reflection on the water under a burning/destroyed rig.

    Upgraded ocean feature: when a rig is destroyed, its fire casts a warm
    orange reflection on the water below that shimmers and wobbles like
    light on waves. This is the "stronger fire-reflection glow" the VLM
    asked for — it ties the explosion to the ocean visually.

    Args:
      rig_x: center x of the rig
      water_y: y of the waterline at the rig
      scale: rig scale (reflection scales with it)
      elapsed: animation timer (for shimmer)
      heat: 0.0..1.0 — how hot the fire still is (fades over time)
    """
    if heat <= 0:
        return
    s = scale
    refl_w = int(180 * s)
    refl_h = int(90 * s)
    refl = pygame.Surface((refl_w, refl_h), pygame.SRCALPHA)
    rw, rh = refl.get_size()
    # Layered horizontal streaks that narrow and fade toward the bottom,
    # with a wobble that simulates light dancing on waves.
    for i in range(12):
        frac = i / 11.0
        ry = int(frac * (rh - 4))
        half = int((rw * 0.40 * (1.0 - frac * 0.55)) * heat)
        if half < 1:
            continue
        # Wobble: each line sways differently based on elapsed time + row
        wobble = int(math.sin(elapsed * 0.06 + i * 0.7 + rig_x * 0.01) * (3 + frac * 6))
        alpha = int(85 * heat * (1.0 - frac * 0.4))
        if alpha <= 0:
            continue
        # Color: hot orange near the top (closest to fire), fading to dark red
        r = int(255 * heat)
        g = int(100 + 40 * (1.0 - frac))
        b = int(30 * (1.0 - frac))
        pygame.draw.line(
            refl,
            (r, g, b, alpha),
            (rw // 2 - half + wobble, ry),
            (rw // 2 + half + wobble, ry),
            1,
        )
    # Bright core glow at the base of the reflection (hottest point)
    core_r = int(40 * s * heat)
    if core_r > 2:
        core_glow = pygame.Surface((core_r * 2, core_r * 2), pygame.SRCALPHA)
        pygame.draw.circle(
            core_glow,
            (255, 140, 40, int(60 * heat)),
            (core_r, core_r),
            core_r,
        )
        refl.blit(
            core_glow,
            (rw // 2 - core_r, 0),
            special_flags=pygame.BLEND_RGBA_ADD,
        )
    surface.blit(refl, (int(rig_x - rw // 2), int(water_y + 6)))

def _draw_destroyed_rig(surface, cx, water_y, scale, elapsed_since_impact, tilt=0.0, rig_index=None):
    """Three different rig deaths.
    LEFT   rig: side-slam collapse, it folds hard into the sea.
    CENTER rig: vertical implosion, legs fold inward and the deck pancakes down.
    RIGHT  rig: twisting break-up, the upper deck tears in two and corkscrews down.
    """
    if elapsed_since_impact < 0:
        return False
    s = max(0.35, float(scale))
    ix = float(cx)
    waterline = float(water_y)
    t = float(max(0, elapsed_since_impact))
    if rig_index is None:
        if tilt < -0.15:
            style = 0      # left / side-slam
        elif tilt > 0.15:
            style = 2      # right / twist-break
        else:
            style = 1      # center / implosion
    else:
        style = int(rig_index) % 3
    # Style tuning. These few values make the three silhouettes behave differently.
    if style == 0:          # LEFT: slow heavy fall to the left
        fall_sign = -1.0
        collapse_delay = 3.0
        collapse_len = 78.0
        sink_start = 92.0
        twist_mult = 1.05
        smoke_drift = -1.0
        deck_mode = "side"
    elif style == 1:        # CENTER: fast straight-down implosion
        fall_sign = 0.0
        collapse_delay = 0.0
        collapse_len = 48.0
        sink_start = 64.0
        twist_mult = 0.22
        smoke_drift = 0.0
        deck_mode = "implode"
    else:                   # RIGHT: delayed twisting break-up to the right
        fall_sign = 1.0
        collapse_delay = 10.0
        collapse_len = 92.0
        sink_start = 105.0
        twist_mult = 1.45
        smoke_drift = 1.0
        deck_mode = "twist"
    buckle = min(1.0, t / 34.0)
    collapse = min(1.0, max(0.0, (t - collapse_delay) / collapse_len))
    sink = min(1.0, max(0.0, (t - sink_start) / 165.0))
    burn = max(0.34, 1.0 - t / 520.0)
    pulse = 0.72 + 0.28 * math.sin(t * 0.105 + ix * 0.01 + style)
    # ── Oil slick and fiery reflections ───────────────────────────────
    slick_w = int((105 + min(210, t * (1.15 + 0.22 * style))) * s)
    slick_h = int((34 + min(46, t * (0.24 + 0.03 * style))) * s)
    oil = pygame.Surface((slick_w, slick_h), pygame.SRCALPHA)
    pygame.draw.ellipse(oil, (4, 6, 9, min(180, int(95 + t * 0.42))), oil.get_rect())
    if style == 1:
        inner = pygame.Rect(int(slick_w * 0.22), int(slick_h * 0.18), int(slick_w * 0.56), int(slick_h * 0.50))
    elif style == 0:
        inner = pygame.Rect(int(slick_w * 0.08), int(slick_h * 0.26), int(slick_w * 0.68), int(slick_h * 0.34))
    else:
        inner = pygame.Rect(int(slick_w * 0.25), int(slick_h * 0.23), int(slick_w * 0.67), int(slick_h * 0.36))
    pygame.draw.ellipse(oil, (44, 25, 15, int(54 * burn)), inner)
    for i in range(6):
        y = int(slick_h * (0.35 + i * 0.075))
        wob = int(math.sin(t * 0.04 + i + style) * 7 * s)
        side_pull = int(smoke_drift * i * 4 * s)
        pygame.draw.line(
            oil,
            (255, 102, 30, int(24 * burn)),
            (slick_w // 2 - int(42 * s) + wob + side_pull, y),
            (slick_w // 2 + int(42 * s) + wob + side_pull, y),
            1,
        )
    surface.blit(oil, (int(ix - slick_w // 2), int(waterline - 4 * s)))
    # ── Different water hits/ripples ──────────────────────────────────
    if t < 155:
        ripple = pygame.Surface((int(280 * s), int(96 * s)), pygame.SRCALPHA)
        rw, rh = ripple.get_size()
        for i in range(4):
            age = max(0.0, t - i * (12.0 if style == 1 else 16.0))
            if age <= 0:
                continue
            alpha = int(100 * max(0.0, 1.0 - age / 155.0))
            if style == 1:
                wmul, hmul, offx = 2.5, 0.42, 0
            elif style == 0:
                wmul, hmul, offx = 2.25, 0.34, -i * 12 * s
            else:
                wmul, hmul, offx = 2.8, 0.48, i * 13 * s
            rect = pygame.Rect(0, 0, int((58 + age * wmul) * s), int((10 + age * hmul) * s))
            rect.center = (rw // 2 + int(offx), rh // 2 + int(10 * s))
            pygame.draw.ellipse(ripple, (126, 170, 205, alpha), rect, 1)
        surface.blit(ripple, (int(ix - rw // 2), int(waterline - rh // 2 + 16 * s)))
    deck_start_y = waterline - 104 * s
    # ── Collapsing legs and deck ──────────────────────────────────────
    if collapse < 0.985:
        if deck_mode == "implode":
            # Center rig: legs fold inward and deck drops almost vertically.
            deck_y = deck_start_y + (waterline - 16 * s - deck_start_y) * (collapse ** 0.78)
            deck_w = (150 - 74 * collapse) * s
            deck_d = (30 + 14 * collapse) * s
            twist = math.sin(t * 0.16) * 7 * s
            leg_roots = [ix - 62 * s, ix - 24 * s, ix + 22 * s, ix + 62 * s]
            for li, root_x in enumerate(leg_roots):
                side = -1 if root_x < ix else 1
                top_x = ix + (-56 + li * 37) * s * (1.0 - 0.42 * collapse)
                top_y = deck_y + (12 + li * 1.4) * s
                knee_x = ix + side * (18 + li % 2 * 10) * s * (1.0 - 0.55 * buckle)
                knee_y = waterline - (50 - li * 5) * s + collapse * 62 * s
                pygame.draw.line(surface, (31, 34, 41), (int(top_x), int(top_y)), (int(knee_x), int(knee_y)), max(1, int(4 * s)))
                pygame.draw.line(surface, (25, 28, 35), (int(knee_x), int(knee_y)), (int(root_x), int(waterline + 20 * s)), max(1, int(3 * s)))
            # Two deck halves collapsing toward the middle, with a visible crack.
            for half, side in enumerate((-1, 1)):
                half_w = deck_w * 0.52
                center_x = ix + side * deck_w * 0.20 * (1.0 - collapse)
                dip = (12 + 28 * collapse) * s
                pts = [
                    (center_x - side * half_w * 0.78, deck_y - deck_d * 0.22),
                    (ix + side * 6 * s + twist, deck_y - deck_d * 0.48 + dip),
                    (ix + side * 12 * s + twist, deck_y + deck_d * 0.35 + dip),
                    (center_x - side * half_w * 0.25, deck_y + deck_d * 0.48),
                ]
                pygame.draw.polygon(surface, (27, 30, 37), pts)
                pygame.draw.polygon(surface, (78, 80, 86), pts, 1)
            pygame.draw.line(surface, (255, 126, 40, int(120 * burn)), (int(ix), int(deck_y - 20 * s)), (int(ix), int(deck_y + 36 * s)), max(1, int(2 * s)))
            # Mast crumples downward into the hole.
            mast_base = (ix + twist, deck_y - 8 * s)
            mast_tip = (ix + math.sin(t * 0.08) * 18 * s, deck_y - (82 - 72 * collapse) * s)
            pygame.draw.line(surface, (44, 48, 56), (int(mast_base[0]), int(mast_base[1])), (int(mast_tip[0]), int(mast_tip[1])), max(1, int(5 * s)))
        else:
            # Left/right rigs: side topples, but right twists harder and tears apart.
            deck_y = deck_start_y + (waterline - 18 * s - deck_start_y) * (collapse ** (1.22 if style == 0 else 1.04))
            deck_w = (148 - (24 if style == 0 else 42) * collapse) * s
            deck_d = (30 - 4 * collapse) * s
            twist = fall_sign * (10 + (72 if style == 0 else 108) * collapse + math.sin(t * 0.09) * 12) * s * twist_mult
            split_gap = (0 if style == 0 else 28 * collapse * s)
            leg_roots = [ix - 62 * s, ix - 25 * s, ix + 18 * s, ix + 58 * s]
            for li, root_x in enumerate(leg_roots):
                top_x = ix + (-56 + li * 35) * s + twist * (0.16 + li * 0.09)
                top_y = deck_y + (12 + li * 1.4) * s
                knee_x = (top_x + root_x) * 0.5 + fall_sign * (12 + li * 6) * buckle * s
                knee_y = waterline - (42 - li * 4) * s + collapse * (52 if style == 0 else 44) * s
                col = (32, 35, 42) if li % 2 else (25, 28, 35)
                pygame.draw.line(surface, col, (int(top_x), int(top_y)), (int(knee_x), int(knee_y)), max(1, int(4 * s)))

def _draw_attack_terminal_glitches(surface, elapsed, hud_rect):
    """Terminal interference bursts around targeting and impact moments."""
    hx, hy, hw, hh = hud_rect
    burst = 0
    for frame in (ATK_TARGET_CENTER, ATK_MISSILE_1_LAUNCH, ATK_IMPACT_CENTER,
                  ATK_TARGET_LEFT, ATK_MISSILE_2_LAUNCH, ATK_IMPACT_LEFT,
                  ATK_TARGET_RIGHT, ATK_MISSILE_3_LAUNCH, ATK_IMPACT_RIGHT):
        dt = elapsed - frame
        if 0 <= dt < 24:
            burst = max(burst, 24 - dt)
    if burst <= 0:
        return
    rng_g = random.Random(9000 + elapsed // 2)
    for _ in range(14 + burst // 2):
        gy = hy + rng_g.randint(-8, hh + 10)
        gh = rng_g.randint(1, 5)
        gx = hx + rng_g.randint(-18, 18)
        gw = hw + rng_g.randint(-25, 45)
        strip = pygame.Surface((max(20, gw), gh), pygame.SRCALPHA)
        col = rng_g.choice([(80, 180, 240), (255, 130, 45), (210, 60, 45)])
        strip.fill((*col, rng_g.randint(70, 150)))
        surface.blit(strip, (gx, gy))
    if burst > 8:
        warn = font_terminal_small.render("SIGNAL BREAKUP", True, (255, 120, 60))
        warn_bg = pygame.Surface((warn.get_width() + 12, warn.get_height() + 4), pygame.SRCALPHA)
        warn_bg.fill((0, 0, 0, 145))
        surface.blit(warn_bg, (hx + hw - warn_bg.get_width() - 8, hy + hh + 6))
        surface.blit(warn, (hx + hw - warn_bg.get_width() - 2, hy + hh + 8))

def _draw_walk_attack_overlook(surface, force=False):
    """Pseudo-3D overlook panorama for the rig strike sequence.
    force=True keeps drawing the final observation feed during the return
    transition, preventing an abrupt snap back to top-down gameplay.
    """
    if not force and not (STATE.walk_attack_active or STATE.walk_terminal_active):
        return
    elapsed = STATE.walk_attack_timer
    w, h = surface.get_size()
    horizon_y = HORIZON_Y_OVERLOOK
    # Hard-clear the destination first so nothing from the walking-sim scene can
    # ever bleed through this cinematic layer.
    surface.fill((0, 0, 0))
    intro_alpha = 0
    if elapsed < 60:
        intro_alpha = int(255 * (1 - max(0, elapsed) / 60))
    shake_mag = max(0, int(STATE.screen_shake))
    if STATE.screen_shake > 0:
        STATE.screen_shake -= 1
    # Do not offset the entire rendered frame; that was causing visible bands.
    shake_x = 0
    shake_y = 0
    # Pre-fill an alpha scene with opaque black so nothing from the walking-sim
    # map can bleed through, while still letting us use per-pixel alpha effects.
    scene = pygame.Surface((w, h), pygame.SRCALPHA)
    scene.fill((0, 0, 0, 255))
    sky_top = (12, 10, 24)
    sky_bot = (56, 38, 72)
    for y in range(0, horizon_y):
        p = y / max(1, horizon_y)
        r = int(sky_top[0] + (sky_bot[0] - sky_top[0]) * p)
        g = int(sky_top[1] + (sky_bot[1] - sky_top[1]) * p)
        b = int(sky_top[2] + (sky_bot[2] - sky_top[2]) * p)
        pygame.draw.line(scene, (r, g, b), (0, y), (w, y))
    rng_stars = random.Random(42)
    for _ in range(38):
        sx = rng_stars.randint(0, w)
        sy = rng_stars.randint(0, horizon_y // 2)
        sa = rng_stars.randint(70, 170)
        sr = rng_stars.randint(1, 2)
        pygame.draw.circle(scene, (190, 205, 235, sa), (sx, sy), sr)
    # Very subtle upper-sky haze only; avoid any strong horizontal banding.
    haze = pygame.Surface((w, 70), pygame.SRCALPHA)
    for i, (cx, cy, rx, ry, alpha) in enumerate([
        (w * 0.28, 8, 120, 34, 8),
        (w * 0.62, 4, 145, 40, 7),
        (w * 0.84, 10, 110, 30, 6),
    ]):
        drift = math.sin(elapsed * 0.006 + i * 1.4) * 6
        pygame.draw.ellipse(
            haze,
            (18, 20, 34, alpha),
            pygame.Rect(int(cx - rx + drift), cy, rx * 2, ry * 2),
        )
    scene.blit(haze, (0, 0))
    _draw_attack_water(scene, elapsed, horizon_y)
    rng_sheen = random.Random(37)
    for _ in range(8):
        sx = (rng_sheen.randint(0, w) + int(elapsed * 0.35)) % w
        sy = horizon_y + rng_sheen.randint(42, 240)
        sw = rng_sheen.randint(24, 54)
        sheen = pygame.Surface((sw, 8), pygame.SRCALPHA)
        sc = rng_sheen.choice([(180, 80, 60), (170, 120, 90), (135, 95, 165)])
        pygame.draw.ellipse(sheen, (*sc, rng_sheen.randint(10, 28)), sheen.get_rect())
        scene.blit(sheen, (int(sx), int(sy)))
    rig_draw_data = [_attack_rig_pose(i, elapsed) for i in range(3)]
    destroyed_count = sum(1 for s in STATE.rig_status if s == "DESTROYED")
    if destroyed_count > 0:
        glow_layer = pygame.Surface((w, 120), pygame.SRCALPHA)
        glow_intensity = min(1.0, destroyed_count / 3.0)
        pulse = 0.78 + 0.22 * math.sin(elapsed * 0.06)
        for gy in range(120):
            alpha = int(40 * glow_intensity * pulse * (1 - gy / 120))
            pygame.draw.line(glow_layer, (255, 105, 30, alpha), (0, gy), (w, gy), 1)
        scene.blit(glow_layer, (0, horizon_y - 46))
        # Long oil-fire reflections under destroyed rigs. These persist even after the blast sprite fades.
        for idx, rig in enumerate(rig_draw_data):
            if STATE.rig_status[idx] != "DESTROYED":
                continue
            since = max(0, elapsed - RIG_IMPACT_FRAMES[idx])
            refl = pygame.Surface((int(190 * rig["scale"]), int(150 * rig["scale"])), pygame.SRCALPHA)
            rw, rh = refl.get_size()
            heat = max(0.25, 1.0 - since / 760.0)
            for ry in range(0, rh, 3):
                frac = ry / max(1, rh)
                alpha = int(72 * heat * (1.0 - frac) * (0.72 + 0.28 * math.sin(elapsed * 0.06 + ry)))
                if alpha <= 0:
                    continue
                half = int((58 * rig["scale"]) * (1.0 - frac * 0.72))
                wob = int(math.sin(elapsed * 0.045 + ry * 0.15 + idx) * (7 + frac * 10))

    for idx in draw_order:
        rig = rig_draw_data[idx]
        if STATE.rig_status[idx] == "DESTROYED":
            # ── Fire-reflection glow on the water under burning rigs ──
            # Drawn BEFORE the destroyed rig so the rig sits on top of its
            # own reflection. Heat fades over time (760 frames = ~12.7s).
            since_impact = elapsed - RIG_IMPACT_FRAMES[idx]
            heat = max(0.0, 1.0 - since_impact / 760.0)
            if heat > 0.05:
                _draw_attack_water_fire_reflection(

            _draw_destroyed_rig(
                scene,
                rig["x"],

        else:

                scene,
                rig["x"],
                rig["water_y"],
                rig["scale"],
                elapsed,
                state=STATE.rig_status[idx],
                tilt=rig["tilt"],
            )
    # Waterline occlusion to seat the structures into the ocean a bit better.
    for rig in rig_draw_data:
