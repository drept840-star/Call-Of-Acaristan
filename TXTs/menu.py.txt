class ModernButton:
    """Menu button with dt-based scale, glow, and click feedback.

    Polish layers (applied on top of the existing slide-on-hover behavior):
      - Hover:   text scales up to 1.04x, left accent bar gets a soft orange
                 glow halo, chevron brightens.
      - Click:   button visually dips +2px on Y and emits a brief 180ms white
                 flash overlay.
      - Keyboard selection (arrow keys): same visual as hover, plus a brighter
        left bar so the user can always see which item is focused.

    All animation state is updated with delta time so the feel is identical at
    60Hz, 144Hz, or under frame drops. The lerp constants below are tuned for
    a 60Hz frame; the per-frame coefficient is recomputed from dt so motion
    stays frame-rate independent.
    """

    # Lerp speeds in "fraction of remaining distance per second".
    # Higher = snappier. Tune here, used by _lerp_dt.
    _SLIDE_SPEED = 12.0      # hover slide-in
    _SCALE_SPEED = 14.0      # hover scale-up
    _GLOW_SPEED = 10.0       # left-bar glow ramp
    _PRESS_SPEED = 18.0      # click Y-dip recover
    _FLASH_DECAY = 5.5       # post-click white flash fade

    def __init__(self, x, y, text):

        self.x, self.y, self.text = x, y, text

        self.width, self.height = 300, 40

        self.rect = pygame.Rect(x, y, self.width, self.height)

        self.is_hovered = False
        self.is_selected = False   # keyboard focus (arrow keys)
        self._prev_hover = False

        # Animated state — all updated via dt-based lerp.
        self.anim_offset = 0.0     # horizontal slide on hover
        self.scale = 1.0           # text scale, 1.0 idle → 1.04 hover
        self.glow_alpha = 0.0      # 0..255, left accent bar glow
        self.press_dip = 0.0       # 0..2px Y dip on click
        self.flash_timer = 0.0     # seconds remaining of click flash

        # Pre-rendered glow surface for the left accent bar (regenerated only
        # if the bar geometry changes — currently fixed at 9px wide).
        self._glow_surf = pygame.Surface((24, self.height + 20), pygame.SRCALPHA)

    # ---- public API -------------------------------------------------------

    def update(self, dt):
        """Advance animation state by dt seconds.

        Called once per frame by draw_menu() for every visible button. Idempotent
        when called with dt=0 — safe to skip if the frame is paused.
        """
        active = self.is_hovered or self.is_selected
        # Hover/selected drive the slide + scale + glow in unison so the
        # keyboard-focus and mouse-focus feel identical.
        self.anim_offset = _lerp_dt(self.anim_offset, 20.0 if active else 0.0,
                                    self._SLIDE_SPEED, dt)
        self.scale = _lerp_dt(self.scale, 1.04 if active else 1.0,
                              self._SCALE_SPEED, dt)
        self.glow_alpha = _lerp_dt(self.glow_alpha, 220.0 if active else 0.0,
                                   self._GLOW_SPEED, dt)

        # Click dip recovers on its own.
        if self.press_dip > 0.0:
            self.press_dip = _lerp_dt(self.press_dip, 0.0,
                                      self._PRESS_SPEED, dt)
            if self.press_dip < 0.05:
                self.press_dip = 0.0

        # Flash decays linearly with time.
        if self.flash_timer > 0.0:
            self.flash_timer = max(0.0, self.flash_timer - dt)

    def press(self):
        """Trigger click feedback: 2px Y dip + 180ms white flash."""
        self.press_dip = 2.0
        self.flash_timer = 0.18

    def draw(self, surface):

        text_color = C_WHITE if (self.is_hovered or self.is_selected) else C_GREY

        actual_x = self.x + int(self.anim_offset)
        actual_y = self.y + int(self.press_dip)

        # Always-visible tick mark (orange accent bar on the left)
        tick_col = C_ORANGE if (self.is_hovered or self.is_selected) else (90, 55, 20)
        pygame.draw.rect(surface, tick_col,
                         (self.x - 25, self.y + 6, 3, self.height - 12))

        # Soft glow halo around the left bar — only when active.
        if self.glow_alpha > 4.0:
            self._glow_surf.fill((0, 0, 0, 0))
            for r, a_mult in [(12, 0.25), (8, 0.45), (5, 0.75)]:
                a = int(self.glow_alpha * a_mult)
                if a <= 0:
                    continue
                pygame.draw.rect(
                    self._glow_surf, (255, 120, 0, a),
                    (12 - r, 10 - r + 2, r * 2, self.height + r * 2 - 4),
                    border_radius=r,
                )
            surface.blit(self._glow_surf, (self.x - 36, self.y - 10))

        # Always-visible chevron prefix
        chevron_col = C_ORANGE if (self.is_hovered or self.is_selected) else (110, 110, 110)
        chevron = font_btn.render(">", True, chevron_col)
        surface.blit(chevron, (self.x - 15, actual_y + 4))

        # Solid orange left bar + hover background — only when active.
        if self.is_hovered or self.is_selected:
            pygame.draw.rect(surface, C_ORANGE,
                             (actual_x - 15, actual_y, 5, self.height))
            surface.blit(_btn_hover_bg, (actual_x - 5, actual_y))

        # Scaled text. Pygame doesn't do sub-pixel scale cleanly on SysFont,
        # so we round to the nearest int. The 1.04x gain is small but reads
        # clearly in motion.
        text_surf = font_btn.render(self.text, True, text_color)
        if self.scale > 1.01:
            new_w = max(1, int(text_surf.get_width() * self.scale))
            new_h = max(1, int(text_surf.get_height() * self.scale))
            text_surf = pygame.transform.smoothscale(text_surf, (new_w, new_h))
        surface.blit(text_surf, (actual_x, actual_y + 4))

        # Click flash overlay — white tint that fades over 180ms.
        if self.flash_timer > 0.0:
            flash_alpha = int(140 * (self.flash_timer / 0.18))
            flash = pygame.Surface((self.width, self.height), pygame.SRCALPHA)
            flash.fill((255, 240, 200, flash_alpha))
            surface.blit(flash, (actual_x - 5, actual_y))

    def check_hover(self, mp):

        self._prev_hover = self.is_hovered

        self.is_hovered = self.rect.collidepoint(mp)

        if self.is_hovered and not self._prev_hover:

            audio.play("menu_hover", volume=0.3)

    def is_clicked(self, event, mp):

        if event.type == pygame.MOUSEBUTTONDOWN and event.button == 1 and self.is_hovered:

            audio.play("menu_click", volume=0.5)
            self.press()
            return True

        return False

def draw_menu():

    # ── Compute dt and advance all polish animation systems ──
    # Per PYGAME_MASTER_ENGINEER Module 4: dt-based motion so the feel is
    # identical at 60Hz, 144Hz, or under frame drops.
    now_ms = pygame.time.get_ticks()
    if not _anim.initialized:
        _init_menu_anim()
        _anim.last_ms = now_ms
    dt = (now_ms - _anim.last_ms) / 1000.0
    _anim.last_ms = now_ms
    # Clamp dt to a max of 1/15s so a long frame stall (window drag, etc.)
    # doesn't catapult rain drops across the screen.
    dt = min(dt, 1.0 / 15.0)
    _update_menu_anim(dt)

    # ── Detect mouse motion → release keyboard focus ──
    # If the mouse actually moved since last frame, hand control back to the
    # mouse. pygame.mouse.get_rel() returns the delta since the last call.
    mx, my = pygame.mouse.get_rel()
    if mx != 0 or my != 0:
        reset_keyboard_selection()

    # ── Background + rain + embers (under the dark panel) ──
    draw_menu_background()
    _draw_rain(screen)
    _draw_embers(screen)

# ═══════════════════════════════════════════════════════════════════════════
# MENU POLISH SYSTEMS
# ─────────────────────────────────────────────────────────────────────────
# Applied per PYGAME_MASTER_ENGINEER Module 2 + Module 4:
#   - dt-based lerp for all motion (frame-rate independent)
#   - separate update() and draw() passes (MVC)
#   - pre-rendered surfaces for static art (vignette, glow sprites)
#
# Systems:
#   _lerp_dt          — frame-rate-independent lerp helper
#   _anim             — module-level state for all polish systems
#   _init_menu_anim   — populate rain drops + embers (idempotent)
#   _update_menu_anim — step all animations forward by dt seconds
#   _draw_rain        — 3-layer parallax rain + ground splashes
#   _draw_embers      — drifting fire embers from right-side glow
#   _draw_vignette    — radial dark edge overlay
#   _draw_title_glow  — pulsing orange halo behind the title
#   _draw_scanline_drift — moving horizontal scanline over the title
#   _draw_boot_banner — typewriter KRAPNET boot text on first menu show
#   _draw_news_ticker — scrolling lore ticker above the footer
#
# Keyboard navigation (Module 2 accessibility):
#   keyboard_idx                  — currently focused button index, -1 = mouse
#   handle_menu_keydown(event)    — call from acaristan MENU-state KEYDOWN
#   consume_keyboard_trigger()    — returns button index to fire, or -1
# ═══════════════════════════════════════════════════════════════════════════


def _lerp_dt(current, target, speed_per_sec, dt):
    """Frame-rate-independent exponential lerp.

    `speed_per_sec` is the approximate "fraction of remaining distance closed
    per second". At 60fps with speed=12, this behaves like the old `* 0.2`
    per-frame lerp but stays correct at any frame rate.
    """
    if dt <= 0:
        return current
    # Equivalent to applying `* k` per-frame where k = 1 - exp(-speed * dt).
    k = 1.0 - math.exp(-speed_per_sec * dt)
    return current + (target - current) * k


# ── Module-level animation state ─────────────────────────────────────────
# All long-lived state for the polish systems lives here, initialized lazily
# on first draw_menu() call so we don't pay for it during gameplay.

class _MenuAnimState:
    """Holds all per-frame animation state for the menu polish systems.

    Kept as a plain class (not a dict) so attribute typos raise immediately.
    """

    def __init__(self):
        self.initialized = False
        self.last_ms = 0           # last pygame.time.get_ticks() value
        self.t = 0.0               # total elapsed time (seconds) since first draw
        self.rain_drops = []       # list of dicts: x, y, length, speed, layer, alpha
        self.splashes = []         # list of dicts: x, y, age, max_age, radius
        self.embers = []           # list of dicts: x, y, vx, vy, age, max_age, brightness
        self.vignette = None       # pre-rendered radial dark edge surface
        self.title_glow_pulse = 0.0  # 0..1, oscillates with sin(t * 1.8)
        self.boot_done = False
        self.boot_t = 0.0          # seconds since first menu draw
        self.boot_chars_typed = 0  # typewriter cursor for boot banner
        self.ticker_offset = 0.0   # pixels scrolled
        # Keyboard nav
        self.keyboard_idx = -1     # -1 = mouse control, 0..N-1 = focused button
        self.pending_trigger = -1  # button index that should fire this frame, or -1

def _init_menu_anim():
    """Populate rain drops, embers, and pre-rendered surfaces.

    Idempotent — safe to call multiple times, only the first call does work.
    """
    if _anim.initialized:
        return
    _anim.initialized = True
    _anim.last_ms = pygame.time.get_ticks()

    # ---- Rain: 3 layers, very subtle ----
    # Tuned WAY down per user feedback ("pretty eh"). Now reads as a faint
    # atmospheric hint rather than a full storm — short streaks, dim alpha,
    # only 40 drops total across all three layers.
    layer_specs = [
        # (count, length_range, speed_range, alpha_range, thickness)
        (18, (6, 10),  (160, 240), (35, 60),   1),   # far — barely visible
        (15, (10, 16), (280, 380), (60, 90),   1),   # mid
        (7,  (16, 22), (440, 560), (90, 130),  1),   # near — slightly brighter
    ]
    for count, lr, sr, ar, thick in layer_specs:
        for _ in range(count):
            _anim.rain_drops.append({
                "x": random.uniform(0, WIDTH + 80),
                "y": random.uniform(0, HEIGHT),
                "length": random.uniform(*lr),
                "speed": random.uniform(*sr),
                "alpha": random.randint(*ar),
                "thickness": thick,
                # All drops share the same wind angle for visual coherence.
                "wind": -0.18,  # radians from vertical, blows slightly left
            })

    # ---- Vignette: pre-rendered radial gradient ----
    # Transparent in the center, dark at the edges. Built once and reused.
    vig = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA)
    cx, cy = WIDTH // 2, HEIGHT // 2
    max_r = math.hypot(cx, cy)
    # Render as concentric rings from outside in — cheaper per-ring than
    # per-pixel, and the gradient is smooth enough at this resolution.
    for r in range(int(max_r), 0, -2):
        # 0 at center, 1 at edge.
        t = r / max_r
        # Ease so the darkening is concentrated near the edges.
        eased = t ** 2.2
        alpha = int(180 * eased)
        if alpha < 2:
            continue
        pygame.draw.circle(vig, (0, 0, 0, alpha), (cx, cy), r)
    _anim.vignette = vig


def _update_menu_anim(dt):
    """Advance all animation systems by dt seconds."""
    if not _anim.initialized:
        _init_menu_anim()

    _anim.t += dt
    _anim.title_glow_pulse = 0.5 + 0.5 * math.sin(_anim.t * 1.8)

    # ---- Rain drops ----
    ground_y = int(HEIGHT * 0.88)
    for d in _anim.rain_drops:
        # Velocity vector: mostly downward, slight wind slant.
        vx = math.sin(d["wind"]) * d["speed"]
        vy = math.cos(d["wind"]) * d["speed"]
        d["x"] += vx * dt
        d["y"] += vy * dt
        # Wrap when off-screen.
        if d["y"] > ground_y:
            # Spawn a splash ripple on impact (rare, only mid/near layers
            # — kept sparse per "rain is pretty eh" feedback).
            if d["thickness"] >= 1 and random.random() < 0.12:
                _anim.splashes.append({
                    "x": d["x"],
                    "y": ground_y + random.uniform(0, 6),
                    "age": 0.0,
                    "max_age": random.uniform(0.35, 0.55),
                    "radius": random.uniform(3, 6),
                })
            d["y"] = random.uniform(-40, 0)
            d["x"] = random.uniform(-40, WIDTH + 80)
        elif d["x"] < -50:
            d["x"] = WIDTH + 40

    # ---- Splashes ----
    # Cheap pooled list — remove in place by swapping with the last element.
    i = 0
    while i < len(_anim.splashes):
        s = _anim.splashes[i]
        s["age"] += dt
        if s["age"] >= s["max_age"]:
            _anim.splashes[i] = _anim.splashes[-1]
            _anim.splashes.pop()
        else:
            i += 1

    # ---- Embers ----
    # Spawn new embers from the right edge (where the bg fire glow lives).
    # Rate scales with dt so spawn density is frame-rate independent.
    spawn_chance = 60.0 * dt  # avg 60 per second
    while spawn_chance > 0:
        if random.random() < min(1.0, spawn_chance):
            _anim.embers.append({
                "x": random.uniform(WIDTH - 80, WIDTH + 10),
                "y": random.uniform(HEIGHT * 0.45, HEIGHT * 0.85),
                "vx": random.uniform(-1.5, -0.4),
                "vy": random.uniform(-1.2, -0.2),
                "age": 0.0,
                "max_age": random.uniform(2.0, 4.0),
                "brightness": random.uniform(0.6, 1.0),
                "size": random.randint(1, 2),
            })
        spawn_chance -= 1.0

    i = 0
    while i < len(_anim.embers):
        e = _anim.embers[i]
        e["age"] += dt
        e["x"] += e["vx"]
        e["y"] += e["vy"]
        # Slight upward acceleration as the ember heats.
        e["vy"] -= 0.02 * dt * 60
        if e["age"] >= e["max_age"] or e["x"] < -10 or e["y"] < -10:
            _anim.embers[i] = _anim.embers[-1]
            _anim.embers.pop()
        else:
            i += 1

    # ---- Boot sequence ----
    if not _anim.boot_done:
        _anim.boot_t += dt
        # Total chars across all lines + delays between lines.
        total_chars = sum(len(line) for line in _BOOT_LINES)
        total_delays = _BOOT_LINE_DELAY * len(_BOOT_LINES)
        total_duration = total_chars / _BOOT_CHAR_RATE + total_delays
        _anim.boot_chars_typed = int(
            max(0.0, _anim.boot_t) * _BOOT_CHAR_RATE
        )
        if _anim.boot_t >= total_duration + 0.4:
            _anim.boot_done = True

    # ---- News ticker ----
    _anim.ticker_offset += _TICKER_SPEED * dt
    # Wrap when we've scrolled one full loop width.
    # Computed lazily in _draw_news_ticker; just keep the offset bounded.
    if _anim.ticker_offset > 10_000:
        _anim.ticker_offset -= 10_000

def _draw_rain(surface):
    """3-layer parallax rain + ground splashes."""
    if not _anim.initialized:
        return
    # Single scratch surface for all drops — drawn once per frame, cheaper
    # than per-drop SRCALPHA surfaces.
    rain_surf = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA)
    for d in _anim.rain_drops:
        vx = math.sin(d["wind"])
        vy = math.cos(d["wind"])
        x2 = d["x"] + vx * d["length"]
        y2 = d["y"] + vy * d["length"]
        col = (180, 200, 220, d["alpha"])
        if d["thickness"] == 1:
            pygame.draw.line(rain_surf, col,
                             (int(d["x"]), int(d["y"])),
                             (int(x2), int(y2)), 1)
        else:
            pygame.draw.line(rain_surf, col,
                             (int(d["x"]), int(d["y"])),
                             (int(x2), int(y2)), 2)
    surface.blit(rain_surf, (0, 0))

    # Splashes on top of rain.
    if _anim.splashes:
        splash_surf = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA)
        for s in _anim.splashes:
            t = s["age"] / s["max_age"]
            r = int(s["radius"] * (1 + t * 3))
            a = int(180 * (1 - t))
            if a <= 0 or r <= 0:
                continue
            pygame.draw.ellipse(
                splash_surf, (180, 200, 220, a),
                (int(s["x"] - r), int(s["y"] - r // 2), r * 2, r),
                1,
            )
        surface.blit(splash_surf, (0, 0))


def _draw_embers(surface):
    """Drifting fire embers from the right-side bg glow."""
    if not _anim.initialized or not _anim.embers:
        return
    em_surf = pygame.Surface((WIDTH, HEIGHT), pygame.SRCALPHA)
    for e in _anim.embers:
        t = e["age"] / e["max_age"]
        # Brightness fades in then out.
        fade = math.sin(t * math.pi) * e["brightness"]
        if fade <= 0:
            continue
        # Color: bright yellow at birth → orange mid → red at death.
        if t < 0.4:
            col = (255, int(180 + 60 * (1 - t / 0.4)), 80, int(220 * fade))
        elif t < 0.75:
            col = (255, 130, 40, int(220 * fade))
        else:
            col = (220, 80, 30, int(180 * fade))
        r = e["size"]
        pygame.draw.circle(em_surf, col, (int(e["x"]), int(e["y"])), r)
        # Soft halo around bright embers.
        if fade > 0.5:
            halo_a = int(60 * fade)
            pygame.draw.circle(
                em_surf, (255, 140, 60, halo_a),
                (int(e["x"]), int(e["y"])), r * 2, 1
            )
    surface.blit(em_surf, (0, 0))


def _draw_vignette(surface):
    """Pre-rendered radial dark edge overlay."""
    if _anim.vignette is not None:
        surface.blit(_anim.vignette, (0, 0))


def _draw_title_glow(surface, title_x, title_y, title_w, title_h):
    """Slow orange glow pulse behind the title."""
    if not _anim.initialized:
        return
    # Glow size covers the title plus a 40px margin.
    glow_w = title_w + 80
    glow_h = title_h + 60
    glow = pygame.Surface((glow_w, glow_h), pygame.SRCALPHA)
    cx, cy = glow_w // 2, glow_h // 2
    max_r = math.hypot(cx, cy)
    pulse = _anim.title_glow_pulse
    # Layered concentric ellipses, each softer than the last.
    for r in range(int(max_r), 4, -8):
        t = r / max_r
        eased = (1 - t) ** 2
        alpha = int(45 * eased * pulse)
        if alpha < 2:
            continue
        pygame.draw.ellipse(
            glow, (255, 120, 0, alpha),
            (cx - r, cy - int(r * 0.55), r * 2, int(r * 1.1))
        )
    surface.blit(
        glow,
        (title_x + title_w // 2 - glow_w // 2,
         title_y + title_h // 2 - glow_h // 2),
        special_flags=pygame.BLEND_RGBA_ADD,
    )


def _draw_scanline_drift(surface, x, y, w, h):
    """A bright horizontal scanline that slowly drifts across the title."""
    if not _anim.initialized:
        return
    # Drift period: 5 seconds top-to-bottom.
    period = 5.0
    t = (_anim.t % period) / period
    line_y = y + int(t * h)
    # Bright line + soft glow above/below.
    line_surf = pygame.Surface((w, 12), pygame.SRCALPHA)
    for i in range(12):
        a = int(80 * (1 - abs(i - 6) / 6))
        if a > 0:
            pygame.draw.line(line_surf, (255, 200, 140, a),
                             (0, i), (w, i), 1)
    surface.blit(line_surf, (x, line_y - 6),
                 special_flags=pygame.BLEND_RGBA_ADD)


def _draw_boot_banner(surface):
    """Typewriter KRAPNET boot sequence shown on first menu show.

    Returns True while the boot is still playing (so draw_menu can suppress
    the normal classification banner), False when done.
    """
    if _anim.boot_done:
        return False

    # Backdrop — same dimensions as the normal classification banner.
    banner = pygame.Surface((460, 22), pygame.SRCALPHA)
    banner.fill((8, 10, 14, 230))
    pygame.draw.line(banner, C_ORANGE, (0, 0), (460, 0), 1)
    pygame.draw.line(banner, C_ORANGE, (0, 21), (460, 21), 1)
    surface.blit(banner, (20, 20))

    # Render lines with a typewriter effect, staggered by _BOOT_LINE_DELAY.
    chars_remaining = _anim.boot_chars_typed
    y_offset = 24
    for line_idx, line in enumerate(_BOOT_LINES):
        # Each line starts typing at line_idx * _BOOT_LINE_DELAY worth of
        # chars consumed.
        start_chars = int(line_idx * _BOOT_LINE_DELAY * _BOOT_CHAR_RATE)
        if chars_remaining <= start_chars:
            break
        chars_on_this_line = min(
            len(line), chars_remaining - start_chars
        )
        visible = line[:chars_on_this_line]
        # Blinking cursor on the line currently being typed.
        if chars_on_this_line < len(line) and \
           (line_idx == len(_BOOT_LINES) - 1 or
            chars_remaining < start_chars + len(line)):
            if int(_anim.t * 3) % 2 == 0:
                visible += "_"
        surf = _classification_font.render(visible, True, C_ORANGE)
        surface.blit(surf, (30, y_offset))
        y_offset += 14
        if y_offset > 64:  # clamp to banner area
            break

    return True


def _draw_news_ticker(surface):
    """Scrolling lore ticker above the footer."""
    # Compose the full ticker string with separators.
    separator = "  ◆  "
    full = separator.join(_TICKER_ITEMS) + separator
    full_surf = _footer_font.render(full, True, (180, 160, 110))
    sep_surf = _footer_font.render(separator, True, C_ORANGE)
    sep_w = sep_surf.get_width()
    full_w = full_surf.get_width()

    # Ticker area: above the footer divider line, full width of the dark panel.
    ticker_y = HEIGHT - 56
    ticker_h = 18
    # Backdrop strip.
    strip = pygame.Surface((WIDTH, ticker_h), pygame.SRCALPHA)
    strip.fill((0, 0, 0, 150))
    surface.blit(strip, (0, ticker_y))

    # KRAPNET WIRE label on the left — static.
    label = _classification_font.render("KRAPNET WIRE", True, C_ORANGE)
    label_w = label.get_width() + 16
    surface.blit(label, (8, ticker_y + 2))
    # Vertical separator after the label.
    pygame.draw.line(surface, (90, 60, 25),
                     (label_w, ticker_y),
                     (label_w, ticker_y + ticker_h), 1)

    # Scrolling text window: from label_w to WIDTH.
    window_w = WIDTH - label_w - 4
    if window_w <= 0 or full_w <= 0:
        return

    # Offset wraps modulo (full_w + sep_w) so the loop is seamless.
    period = full_w + sep_w
    offset = int(_anim.ticker_offset) % period

    # Draw the text twice — once at -offset, once at (period - offset) — so
    # the loop has no visible gap when it wraps.
    clip_rect = pygame.Rect(label_w + 2, ticker_y, window_w, ticker_h)
    prev_clip = surface.get_clip()
    surface.set_clip(clip_rect)
    surface.blit(full_surf, (label_w + 4 - offset, ticker_y + 2))
    surface.blit(full_surf, (label_w + 4 - offset + period, ticker_y + 2))
    surface.set_clip(prev_clip)

    # Top border line on the ticker strip.
    pygame.draw.line(surface, (60, 40, 15),
                     (0, ticker_y), (WIDTH, ticker_y), 1)


# ── Keyboard navigation API ──────────────────────────────────────────────
# Public functions called by acaristan.py. Designed so the existing mouse
# click path keeps working unchanged — keyboard input is layered on top.


def _visible_button_indices():
    """Return list of indices into `buttons[]` that are currently on-screen.

    draw_menu() sets each button's `y` to -999 when it should be hidden
    (e.g. CONTINUE when no save exists). We respect that convention here.
    """
    return [i for i, b in enumerate(buttons) if b.y > 0]


def handle_menu_keydown(event):
    """Process a KEYDOWN event in the MENU state.

    Call from acaristan.py's event loop when game_state == "MENU".
    Returns None — side effects are stored in _anim.pending_trigger, which
    acaristan.py reads via consume_keyboard_trigger().
    """
    if not _anim.initialized:
        _init_menu_anim()

    if event.key in (pygame.K_UP, pygame.K_w):
        visible = _visible_button_indices()
        if not visible:
            return
        if _anim.keyboard_idx == -1:
            # Initializing from mouse control — pick the bottom visible.
            _anim.keyboard_idx = visible[-1]
        elif _anim.keyboard_idx in visible:
            i = visible.index(_anim.keyboard_idx)
            _anim.keyboard_idx = visible[(i - 1) % len(visible)]
        else:
            _anim.keyboard_idx = visible[-1]
        audio.play("menu_hover", volume=0.3)
    elif event.key in (pygame.K_DOWN, pygame.K_s):
        visible = _visible_button_indices()
        if not visible:
            return
        if _anim.keyboard_idx == -1:
            _anim.keyboard_idx = visible[0]
        elif _anim.keyboard_idx in visible:
            i = visible.index(_anim.keyboard_idx)
            _anim.keyboard_idx = visible[(i + 1) % len(visible)]
        else:
            _anim.keyboard_idx = visible[0]
        audio.play("menu_hover", volume=0.3)
    elif event.key in (pygame.K_RETURN, pygame.K_SPACE):
        if _anim.keyboard_idx != -1:
            _anim.pending_trigger = _anim.keyboard_idx
            buttons[_anim.keyboard_idx].press()
            audio.play("menu_click", volume=0.5)


def consume_keyboard_trigger():
    """Return the button index that should fire this frame, or -1.

    acaristan.py should call this after draw_menu() and route the trigger
    through the same action handlers as a mouse click.
    """
    t = _anim.pending_trigger
    _anim.pending_trigger = -1
    return t


def reset_keyboard_selection():
    """Clear keyboard focus — called when the mouse moves."""
    _anim.keyboard_idx = -1

def _fit_text(text, font, max_width):

    if font.size(text)[0] <= max_width:

        return text

    ellipsis = "..."

    while text and font.size(text + ellipsis)[0] > max_width:

        text = text[:-1]

    return text + ellipsis

def update_server_browser_animation():

    """Called once per frame while on multiplayer screen."""

    global _server_animation_timer, _chat_next_spawn, _chat_messages

    global _refresh_active, _refresh_progress

    _server_animation_timer += 1

    # Animate pings + player counts every ~90 frames (1.5s)

    if _server_animation_timer % 90 == 0 and not _refresh_active:

        for i, (name, cur, max_p, ping) in enumerate(server_list):

            delta = random.choice([-1, 0, 0, 0, 0, 1])

            new_cur = max(0, min(max_p, cur + delta))

            new_ping = max(8, min(400, ping + random.randint(-6, 6)))

            server_list[i] = (name, new_cur, max_p, new_ping)

    # Refresh progress

    if _refresh_active:

        _refresh_progress += 0.025

        if _refresh_progress >= 1.0:

            _refresh_active = False

            _refresh_progress = 0.0

            for i, (name, cur, max_p, ping) in enumerate(server_list):

                new_cur = max(0, min(max_p, cur + random.randint(-3, 3)))

                new_ping = max(8, min(400, ping + random.randint(-25, 25)))

                server_list[i] = (name, new_cur, max_p, new_ping)

    # Spawn chat messages

    _chat_next_spawn -= 1

    if _chat_next_spawn <= 0:

        msg = random.choice(CHAT_POOL)

        _chat_messages.append({

            "user": msg[0],

            "text": msg[1],

            "color": msg[2],

            "age": 0,

        })

        _chat_next_spawn = random.randint(60, 130)

        if random.random() < 0.18:

            msg2 = random.choice(CHAT_POOL)

            _chat_messages.append({

                "user": msg2[0],

                "text": msg2[1],

                "color": msg2[2],

                "age": 0,

            })

        if len(_chat_messages) > 50:

            _chat_messages.pop(0)

    # Age messages

    for m in _chat_messages:

        m["age"] += 1

def send_dm_message():

    global _dm_input, _dm_response_delay, _dm_pending_reply

    if _dm_open_user is None or not _dm_input.strip():

        return

    _dm_history[_dm_open_user].append({

        "from": "you",

        "text": _dm_input.strip(),

        "age": 0,

    })

    _dm_input = ""

    _dm_response_delay = random.randint(45, 110)

    _dm_pending_reply = _dm_open_user

    audio.play("menu_click", volume=0.2)

def update_dm_state():

    global _dm_response_delay, _dm_pending_reply

    if _dm_open_user is not None and _dm_open_user in _dm_history:

        for m in _dm_history[_dm_open_user]:

            m["age"] += 1

    if _dm_response_delay > 0:

        _dm_response_delay -= 1

        if _dm_response_delay == 0 and _dm_pending_reply is not None:

            reply = random.choice(CONNECTIONS[_dm_pending_reply]["replies"])

            _dm_history[_dm_pending_reply].append({

                "from": "them",

                "text": reply,

                "age": 0,

            })

            _dm_pending_reply = None

            audio.play("menu_hover", volume=0.3)

def _status_color(status):

    if "Online" in status or status == "always":

        return (80, 220, 100)

    if "Away" in status or "Meeting" in status:

        return (220, 180, 60)

    if "Game" in status:

        return (100, 160, 220)

    if "Offline" in status or "Deleted" in status or "never" in status:

        return (110, 110, 110)

    if "\u2588" in status:

        return (140, 60, 60)

    return (180, 180, 180)

    mouse_pos = get_scaled_mouse()

    list_header = font_terminal_small.render("CONTACTS", True, (80, 130, 160))

    list_bottom_limit = SERVER_PANEL_Y + SERVER_PANEL_H - 90

    visible_friends = (list_bottom_limit - list_y) // row_h

    visible_friends = min(visible_friends, len(CONNECTIONS))

    for vi in range(visible_friends):

        screen.blit(name_surf, (list_x + 28, ry + 4))

        if len(status_text) > 38:

            status_text = status_text[:35] + "..."

        st_surf = font_terminal_small.render(status_text, True, (120, 130, 140))

        screen.blit(st_surf, (list_x + 28, ry + 22))

            f"\u25b2 {remaining_above} above  \u00b7  \u25bc {remaining_below} below  \u00b7  scroll wheel",

            f"\u25bc {remaining_below} more contacts  \u00b7  scroll wheel",

            True, (90, 110, 130),

        )

        screen.blit(more_lbl, (list_x + 6, more_y))

    elif remaining_above > 0:

            f"\u25b2 {remaining_above} above  \u00b7  scroll wheel",

            True, (90, 110, 130),

        )

        screen.blit(more_lbl, (list_x + 6, more_y))

    # ── Scrollbar for contacts ──

def _wrap_server_name(name, font, max_width, max_lines=2):

    """Wrap a server name into up to max_lines lines.

    Returns a list of rendered strings (1 or 2 elements)."""

    if font.size(name)[0] <= max_width:

        return [name]

    words = name.split(" ")

    if len(words) > 1:

        best_split = 1

        for i in range(1, len(words)):

            first = " ".join(words[:i])

            if font.size(first)[0] <= max_width:

                best_split = i

            else:

                break

        line1 = " ".join(words[:best_split])

        line2 = " ".join(words[best_split:])

        if font.size(line2)[0] > max_width:

            ellipsis = "..."

            while line2 and font.size(line2 + ellipsis)[0] > max_width:

                line2 = line2[:-1]

            line2 = line2 + ellipsis

        return [line1, line2]

    mid = len(name) // 2

    while mid > 1 and font.size(name[:mid])[0] > max_width:

        mid -= 1

    line1 = name[:mid]

    line2 = name[mid:]

    if font.size(line2)[0] > max_width:

        ellipsis = "..."

        while line2 and font.size(line2 + ellipsis)[0] > max_width:

            line2 = line2[:-1]

        line2 = line2 + ellipsis

    return [line1, line2]

def _draw_arrow_btn(rect, sym):

    """Arrow button with CRT phosphor glow on hover."""

    hover = rect.collidepoint(get_scaled_mouse())

    bg = (18, 50, 28) if hover else (10, 22, 14)

    pygame.draw.rect(screen, bg, rect)

    border_col = (80, 220, 120) if hover else (30, 70, 40)

    pygame.draw.rect(screen, border_col, rect, 1)

    c = font_btn.render(sym, True, (140, 240, 160) if hover else (90, 170, 110))

    screen.blit(c, (rect.centerx - c.get_width() // 2, rect.centery - c.get_height() // 2))





def _draw_volume_slider(label, bar_rect, left_rect, right_rect, value, row_y):

    """Volume/brightness slider with CRT phosphor fill."""

    screen.blit(font_terminal_small.render(label, True, (110, 180, 130)), (LABEL_X, row_y + 3))

    _draw_arrow_btn(left_rect, "<")

    pygame.draw.rect(screen, (6, 12, 8), bar_rect)

    fill_w = int(bar_rect.width * value / 100)

    if fill_w > 0:

        screen.blit(_VOL_GRADIENT, bar_rect.topleft, (0, 0, fill_w, bar_rect.height))

    pygame.draw.rect(screen, (30, 70, 40), bar_rect, 1)

    _draw_arrow_btn(right_rect, ">")

    pct = font_terminal.render(f"{value:>3}%", True, (140, 240, 160))

    screen.blit(pct, (right_rect.right + 12, row_y))





def _draw_toggle(label, rect, value, row_y):

    """CRT toggle switch — green when ON, dim when OFF."""

    screen.blit(font_terminal_small.render(label, True, (110, 180, 130)), (LABEL_X, row_y + 3))

    hover = rect.collidepoint(get_scaled_mouse())

    on_col = (50, 160, 80) if value else (18, 22, 18)

    border_col = (80, 220, 120) if (hover or value) else (40, 50, 42)

    pygame.draw.rect(screen, on_col, rect)

    pygame.draw.rect(screen, border_col, rect, 1)

    # Knob

    knob_x = rect.right - 22 if value else rect.x + 4

    pygame.draw.rect(screen, (200, 250, 210) if value else (90, 100, 95),

                     (knob_x, rect.y + 4, 18, rect.height - 8))

    state_label = "ON" if value else "OFF"

    s = font_terminal_small.render(state_label, True, (140, 240, 160) if value else (120, 120, 115))

    screen.blit(s, (rect.right + 12, row_y + 3))





def _draw_stepper(label, left_rect, right_rect, value_str, row_y):

    """Stepper control: < VALUE >  for enumerated settings (resolution, mode, etc.)."""

    screen.blit(font_terminal_small.render(label, True, (110, 180, 130)), (LABEL_X, row_y + 3))

    _draw_arrow_btn(left_rect, "<")

    val_surf = font_terminal.render(value_str, True, (140, 240, 160))

    val_cx = (left_rect.right + right_rect.left) // 2

    screen.blit(val_surf, (val_cx - val_surf.get_width() // 2, row_y))

    _draw_arrow_btn(right_rect, ">")

def draw_settings(active_tab, display_mode, res_idx,

                  master_volume, music_volume, sfx_volume, mute_on_focus_loss,

                  crosshair, crosshair_size, auto_reload, aim_assist, subtitles,

                  brightness, scanlines, grain, particle_density,

                  mouse_sensitivity, invert_mouse,

                  reduced_motion, colorblind_mode):

    """Render the settings overlay — CRT terminal style, 6 tabs."""

    screen.blit(_settings_overlay, (0, 0))

    screen.blit(_settings_panel, (SETTINGS_PANEL_X, SETTINGS_PANEL_Y))



    # ── Title with CRT phosphor glow ──

    title_surf = font_speaker.render("SETTINGS", True, (140, 240, 160))

    title_x = SETTINGS_PANEL_X + SETTINGS_PANEL_W // 2 - title_surf.get_width() // 2

    # Glow underlay (blurred green)

    glow = font_speaker.render("SETTINGS", True, (30, 90, 50))

    screen.blit(glow, (title_x + 1, SETTINGS_PANEL_Y + 24))

    screen.blit(title_surf, (title_x, SETTINGS_PANEL_Y + 23))

    pygame.draw.line(screen, (50, 140, 80),

                     (SETTINGS_PANEL_X + 80, SETTINGS_PANEL_Y + 64),

                     (SETTINGS_PANEL_X + SETTINGS_PANEL_W - 80, SETTINGS_PANEL_Y + 64), 1)



    # ── Tab bar ──

    mouse = get_scaled_mouse()

    for i, (rect, label) in enumerate(zip(settings_tab_rects, SETTINGS_TABS)):

        is_active = (i == active_tab)

        hover = rect.collidepoint(mouse)

        bg = (14, 40, 22) if is_active else ((10, 20, 14) if hover else (6, 12, 8))

        pygame.draw.rect(screen, bg, rect)

        if is_active:

            pygame.draw.rect(screen, (80, 220, 120), (rect.x, rect.bottom - 3, rect.width, 3))

        else:

            pygame.draw.line(screen, (25, 50, 32), (rect.x, rect.bottom), (rect.right, rect.bottom), 1)

        col = (140, 240, 160) if is_active else ((110, 160, 120) if hover else (80, 110, 90))

        ts = font_terminal_small.render(label, True, col)

        screen.blit(ts, (rect.centerx - ts.get_width() // 2, rect.centery - ts.get_height() // 2))



    # ── Tab content ──

    if active_tab == 0:

        # DISPLAY

        _draw_stepper("Resolution", settings_res_left_rect, settings_res_right_rect,

                      SETTINGS_RESOLUTIONS[res_idx] if res_idx < len(SETTINGS_RESOLUTIONS) else "N/A",

                      ROW_RES_Y)

        _draw_stepper("Display Mode", settings_display_left_rect, settings_display_right_rect,

                      DISPLAY_MODES[display_mode] if display_mode < len(DISPLAY_MODES) else "N/A",

                      ROW_DISPLAY_Y)

    elif active_tab == 1:

        _draw_section_header("LEVELS", CONTENT_Y)

        _draw_volume_slider("Master Volume", settings_master_bar_rect,

                            settings_master_left_rect, settings_master_right_rect,

        _draw_volume_slider("Music Volume", settings_music_bar_rect,

                            settings_music_left_rect, settings_music_right_rect,

                            music_volume, ROW_MUSIC_Y)

        _draw_volume_slider("SFX Volume", settings_sfx_bar_rect,

                            settings_sfx_left_rect, settings_sfx_right_rect,

                            sfx_volume, ROW_SFX_Y)

        _draw_toggle("Mute on Focus Loss", settings_mute_rect,

                     mute_on_focus_loss, ROW_MUTE_Y)

        _draw_hint("Silences all audio when the window loses focus.", ROW_MUTE_Y + 36)



    elif active_tab == 2:

        # GAMEPLAY

        _draw_section_header("COMBAT", CONTENT_Y)

        _draw_toggle("Crosshair", settings_crosshair_rect, crosshair, ROW_CROSSHAIR_Y)

        _draw_stepper("Crosshair Size", settings_crosshair_size_left, settings_crosshair_size_right,

                      CROSSHAIR_SIZE_LABELS[CROSSHAIR_SIZES.index(crosshair_size)] if crosshair_size in CROSSHAIR_SIZES else "MEDIUM",

def _format_elapsed(seconds):
    """mm:ss for the elapsed-time stat."""
    m = int(seconds) // 60
    s = int(seconds) % 60
    return f"{m:02d}:{s:02d}"


def draw_pause_menu():
    """Draw the pause overlay + framed panel + mission status + buttons.

    Redesigned from the original 3-line placeholder (flat overlay + "PAUSED"
    text + naked buttons) to match the main menu's visual language: framed
    panel with corner ticks, classification banner, Impact title with drop
    shadow + orange accent bar, operation subtitle, live mission status
    block (elapsed time / hostiles / accuracy / ammo), and the same
    ModernButton style used everywhere else.
    """

    screen.blit(_pause_overlay, (0, 0))

    screen.blit(_pause_panel_frame, (_pause_panel_x, _pause_panel_y))

    # ── Operation subtitle (dynamic — depends on current mission) ──
    op_name = getattr(STATE, "active_mission_operation", "WAREHOUSE RAID — BLACK NOTEBOOK")

    op_surf = font_terminal.render(op_name, True, C_TERMINAL)

    screen.blit(op_surf, (WIDTH // 2 - op_surf.get_width() // 2, _pause_panel_y + 165))

    # ── Mission status header ──
    header_surf = _pause_stat_label_font.render("MISSION STATUS", True, C_ORANGE)

    screen.blit(header_surf, (WIDTH // 2 - header_surf.get_width() // 2, _pause_panel_y + 195))

    # Thin divider under the header.
    pygame.draw.line(screen, (90, 50, 20),

                     (WIDTH // 2 - 120, _pause_panel_y + 213),

                     (WIDTH // 2 + 120, _pause_panel_y + 213), 1)

    # ── Status rows ──
    # Compute live values from STATE. Everything degrades gracefully to
    # "—" if the mission hasn't started yet or STATE doesn't have the field.
    try:

        start_t = getattr(STATE, "mission_start_time", 0) or pygame.time.get_ticks()

        elapsed = max(0, (pygame.time.get_ticks() - start_t) // 1000)

        elapsed_str = _format_elapsed(elapsed)

    except Exception:

        elapsed_str = "--:--"

    hostiles = getattr(STATE, "hostiles_killed", 0)

    shots_fired = getattr(STATE, "shots_fired", 0)

    shots_hit = getattr(STATE, "shots_hit", 0)

    accuracy = int((shots_hit / shots_fired) * 100) if shots_fired > 0 else 0

    ammo_in_mag = getattr(STATE, "ammo_in_mag", 0)

    mag_size = getattr(STATE, "mag_size", 0)

    spare_mags = getattr(STATE, "spare_mags", 0)

    # Two columns: labels on the left, values on the right, centered as a block.
    col_label_x = WIDTH // 2 - 140

    col_value_x = WIDTH // 2 + 140

    rows = [

        ("ELAPSED", elapsed_str),

        ("HOSTILES NEUTRALIZED", str(hostiles)),

        ("ACCURACY", f"{accuracy}%  ({shots_hit}/{shots_fired})"),

        ("AMMO", f"{ammo_in_mag}/{mag_size}  +{spare_mags} mags"),

    ]

    row_y = _pause_panel_y + 228

    for label, value in rows:

        val_surf = _pause_stat_value_font.render(value, True, C_WHITE)

        screen.blit(lbl_surf, (col_label_x - lbl_surf.get_width(), row_y))

        screen.blit(val_surf, (col_value_x, row_y))

        row_y += 26

    # ── Buttons ──
    for btn in pause_buttons:

        btn.check_hover(get_scaled_mouse())

        btn.draw(screen)

def _black_ledger_death_button_progress():
    """0..1 reveal progress for Black Ledger death controls."""
    global _death_animation_start
    if _death_animation_start == 0:
        _death_animation_start = pygame.time.get_ticks()
    elapsed = pygame.time.get_ticks() - _death_animation_start
    # Let the Ledger verdict breathe before controls appear.
    delay = 850
    duration = 900
    p = max(0.0, min(1.0, (elapsed - delay) / duration))
    # Smoothstep easing.
    return p * p * (3.0 - 2.0 * p)


def black_ledger_death_buttons_ready():
    """True once the Black Ledger death controls are visually usable."""
    return _black_ledger_death_button_progress() >= 0.92


def draw_black_ledger_death_buttons():
    """Draw only the interactive controls for the Black Ledger death screen.

    They now fade/slide in after the verdict instead of popping into existence.
    This keeps the final Ledger frame feeling like a death screen first and an
    options menu second.
    """
    progress = _black_ledger_death_button_progress()
    if progress <= 0.01:
        # Keep button rects away from the mouse while invisible.
        for btn in death_buttons:
            btn.rect.y = HEIGHT + 200
            btn.is_hovered = False
        return

    alpha = int(225 * progress)
    slide = int((1.0 - progress) * 34)

    controls = pygame.Surface((WIDTH, 112), pygame.SRCALPHA)

    # Bottom shelf fades in softly from black, not as a hard new UI panel.
    shelf_h = 92
    shelf = pygame.Surface((WIDTH, shelf_h), pygame.SRCALPHA)
    shelf.fill((0, 0, 0, int(142 * progress)))
    pygame.draw.line(shelf, (105, 34, 28, int(130 * progress)), (0, 0), (WIDTH, 0), 1)
    controls.blit(shelf, (0, 18 + slide // 2))

    prompt_alpha = int(160 * progress)
    prompt = font_terminal_small.render(
        "death record complete", True, (135, 82, 74)
    )
    prompt.set_alpha(prompt_alpha)
    controls.blit(prompt, (WIDTH // 2 - prompt.get_width() // 2, 10 + slide // 2))

    # Place the two existing death buttons side-by-side under the death verdict.
    positions = [
        (WIDTH // 2 - 320, 54 + slide),
        (WIDTH // 2 + 20, 54 + slide),
    ]

    ready_for_hover = progress >= 0.84
    mp = get_scaled_mouse()
    # Convert mouse into the local controls surface coordinate.
    local_mp = (mp[0], mp[1] - (HEIGHT - controls.get_height()))

    for btn, (x, y) in zip(death_buttons, positions):
        btn.x = int(x)
        btn.y = int(y)
        btn.rect.x = btn.x
        # Actual screen-space rect must include the controls surface offset.
        btn.rect.y = int(HEIGHT - controls.get_height() + btn.y)

        if ready_for_hover:
            # Temporarily test hover against local surface coordinates.
            old_rect_y = btn.rect.y
            btn.rect.y = btn.y
            btn.check_hover(local_mp)
            btn.rect.y = old_rect_y
        else:
            btn.is_hovered = False

        # Draw to local surface with local y, then restore screen-space rect.
        old_rect_y = btn.rect.y
        btn.rect.y = btn.y
        btn.draw(controls)
        btn.rect.y = old_rect_y

    controls.set_alpha(alpha)
    screen.blit(controls, (0, HEIGHT - controls.get_height()))
